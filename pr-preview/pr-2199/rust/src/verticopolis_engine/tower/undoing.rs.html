1//! What a tower needs to apply an undo step (`sim_undo.rs`): the facts of a
2//! unit or a shaft that an edit owns, which a step puts back while leaving
3//! the live state the simulation moved (occupants, car positions), and the
4//! insertions that put removed units and shafts back where they stood.
5#![cfg_attr(not(feature = "wasm"), allow(dead_code))]
6use super::*;
7use crate::economy::screens::{put_screen_policies, screen_policies};
8use crate::undo::{puts_back, UnitFacts};

10impl Unit {
11    /// Whether the facts an edit owns are the same: the price, the market
12    /// flag, the name, the variety, each screen's film policy and the ticket
13    /// price.
14    pub(crate) fn config_eq(&self, other: &Unit) -> bool {
15        self.rent == other.rent
16            && self.no_rate == other.no_rate
17            && self.label == other.label
18            && self.subtype == other.subtype
19            && screen_policies(self) == screen_policies(other)
20            && self.extras().ticket_price == other.extras().ticket_price
21    }

23    /// The unit as a step puts it back: its record, less the people counts
24    /// nobody is left to settle. The customers and diners inside it left
25    /// with it, so their counts start empty, and a venue's attendance is
26    /// read again from them, as a load does (`load/reader.rs`).
27    pub(crate) fn returning(&self) -> Unit {
28        let counted = self.kind.counts_visitors() || self.kind.is_structural();
29        Unit {
30            customers_in: None,
31            hotel_customers_in: None,
32            out_for_meal: None,
33            occupants: if counted { 0 } else { self.occupants },
34            ..self.clone()
35        }
36    }
37
38    /// This unit, with the facts an edit owns (the price, the market flag,
39    /// the name, the variety, each screen's film policy and the ticket price) taken
40    /// from `found`: the unit as it leaves a gesture that changed those facts
41    /// first, which an undo of the gesture puts back.
42    pub(crate) fn with_edit_facts_of(&self, found: &Unit) -> Unit {
43        let mut u = Unit {
44            label: found.label.clone(),
45            subtype: found.subtype,
46            ..self.clone()
47        };
48        // A condo that sold keeps the price it sold at, which no edit can
49        // change any more.
50        let sold_condo = self.kind == Kind::Condo && self.ever_occupied;
51        if !sold_condo {
52            u.take_price_and_policy_from(found);
53        }
54        u
55    }
56
57    /// Put the price, the market flag, each screen's film policy and the ticket price
58    /// back from `other`.
59    /// The name and the variety go through the tower's setters
60    /// ([`Tower::put_unit_config`]), which log a change.
61    fn take_price_and_policy_from(&mut self, other: &Unit) {
62        self.rent = other.rent;
63        self.no_rate = other.no_rate;
64        let policies = screen_policies(other);
65        if screen_policies(self) != policies {
66            put_screen_policies(self, policies);
67        }
68        let price = other.extras().ticket_price;
69        if self.extras().ticket_price != price {
70            self.extras_mut().ticket_price = price;
71        }
72    }
73}

75impl Transport {
76    /// Whether this shaft and `other` would share a column on some floor, by
77    /// the rule a placement follows (`Tower::validate_transport`): two
78    /// flights of stairs or escalators may meet end to end.
79    pub(crate) fn clashes_with(&self, other: &Transport) -> bool {
80        let floors = self.bottom <= other.top && other.bottom <= self.top;
81        let columns = self.x < other.x + other.width && other.x < self.x + self.width;
82        let end_to_end = self.kind.is_fixed_span()
83            && other.kind.is_fixed_span()
84            && self.x == other.x
85            && self.width == other.width
86            && (self.bottom == other.top || self.top == other.bottom);
87        floors && columns && !end_to_end
88    }
89
90    /// The shaft as a step puts it back: its record, with the cars standing
91    /// empty where a new shaft's stand (`Tower::place_transport`). The people
92    /// who rode it left with it, and the dispatch forgot its cars, so no car
93    /// comes back with a load or under way. Its rides stay: they record use.
94    pub(crate) fn returning(&self) -> Transport {
95        Transport {
96            car_positions: (0..self.cars).map(|i| (self.bottom + i) as f64).collect(),
97            car_dir: vec![0; self.cars as usize],
98            car_load: None,
99            ..self.clone()
100        }
101    }
102
103    /// Whether the facts an edit owns are the same: the span, the cars, the
104    /// stops and the schedule.
105    pub(crate) fn config_eq(&self, other: &Transport) -> bool {
106        self.bottom == other.bottom
107            && self.top == other.top
108            && self.cars == other.cars
109            && self.skip_floors == other.skip_floors
110            && self.schedule == other.schedule
111    }
112
113    /// Put the facts an edit owns back from `other`. The cars on the shaft
114    /// stay where they are; a fleet that grows gets its new cars at the
115    /// bottom, as buying one does, and one that shrinks loses the last ones.
116    pub(crate) fn take_config_from(&mut self, other: &Transport) {
117        self.bottom = other.bottom;
118        self.top = other.top;
119        self.skip_floors.clone_from(&other.skip_floors);
120        self.schedule.clone_from(&other.schedule);
121        let cars = other.cars as usize;
122        if self.cars != other.cars {
123            self.car_positions.resize(cars, other.bottom as f64);
124            self.car_dir.resize(cars, 1);
125            if let Some(load) = &mut self.car_load {
126                load.resize(cars, 0.0);
127            }
128            self.cars = other.cars;
129        }
130        for p in self.car_positions.iter_mut() {
131            *p = p.min(other.top as f64).max(other.bottom as f64);
132        }
133    }
134}

136impl Tower {
137    /// Whether the tiles `u` covers are free on its layer (the rooms, or the
138    /// floors and lobbies under them), a tile held by a unit `leaving` names
139    /// counting as free.
140    pub(crate) fn tiles_free_for(&self, u: &Unit, leaving: impl Fn(i64) -> bool) -> bool {
141        let layer = if u.kind.is_structural() {
142            &self.structure
143        } else {
144            &self.rooms
145        };
146        (u.floor..u.floor + u.kind.floors()).all(|floor| {
147            (u.x..u.x + u.width).all(|x| layer.get(&(floor, x)).is_none_or(|&id| leaving(id)))
148        })
149    }
150
151    /// Put back on unit `id` each fact an edit owns that the edit changed
152    /// from `want` to `left` and nothing has changed since
153    /// ([`UnitChange::Restore`](crate::undo::UnitChange::Restore)).
154    pub(crate) fn put_unit_facts(&mut self, id: i64, want: &UnitFacts, left: &UnitFacts) {
155        let Some(&i) = self.by_id.get(&id) else {
156            return;
157        };
158        let u = &mut self.units[i];
159        if puts_back(&want.rent, &left.rent, &u.rent) {
160            u.rent = want.rent;
161        }
162        if puts_back(&want.no_rate, &left.no_rate, &u.no_rate) {
163            u.no_rate = want.no_rate;
164        }
165        let mut policies = screen_policies(u);
166        for (screen, now) in policies.iter_mut().enumerate() {
167            let (want, left) = (&want.film_policies[screen], &left.film_policies[screen]);
168            if puts_back(want, left, now) {
169                *now = *want;
170            }
171        }
172        if policies != screen_policies(u) {
173            put_screen_policies(u, policies);
174        }
175        if puts_back(
176            &want.ticket_price,
177            &left.ticket_price,
178            &u.extras().ticket_price,
179        ) {
180            u.extras_mut().ticket_price = want.ticket_price;
181        }
182        if puts_back(&want.label, &left.label, &self.units[i].label) {
183            self.set_unit_label_at(i, want.label.clone());
184        }
185        if puts_back(&want.subtype, &left.subtype, &self.units[i].subtype) {
186            self.set_unit_subtype(id, want.subtype);
187        }
188    }
189
190    /// Put units back in the list, each at its index `at` in the list they
191    /// leave (or the end, when the list is shorter), as placing them would
192    /// register them. They go on the end of the list and one walk back from
193    /// there moves them into place, shifting the units after each up, in
194    /// place ([`merge_appended`]); one reindex follows, so a step that
195    /// returns a whole row costs one walk of the list and no second copy of
196    /// it.
197    pub(crate) fn insert_units_at(&mut self, mut inserts: Vec<(usize, Unit)>) {
198        if inserts.is_empty() {
199            return;
200        }
201        inserts.sort_by_key(|&(at, _)| at);
202        let old_len = self.units.len();
203        let lands = landing_places(inserts.iter().map(|&(at, _)| at), old_len);
204        let first = lands[0];
205        let total = old_len + inserts.len();
206        // The same headroom rule as a placement (#1259).
207        if total > self.units.capacity() {
208            let more = (total - old_len).max((old_len / 4).max(64));
209            self.units.reserve_exact(more);
210        }
211        // A floor that gains its first lobby tile stops the express shafts
212        // that span it, as a placement does.
213        let mut new_lobby_floors: Vec<i64> = Vec::new();
214        for (_, u) in &inserts {
215            if u.kind == Kind::Lobby
216                && !self.floor_has_lobby(u.floor)
217                && !new_lobby_floors.contains(&u.floor)
218            {
219                new_lobby_floors.push(u.floor);
220            }
221        }
222        self.units.extend(inserts.into_iter().map(|(_, u)| u));
223        merge_appended(&mut self.units, old_len, &lands);
224        self.reindex_from(first);
225        for &i in &lands {
226            self.register(i);
227            if self.units[i].kind == Kind::WeddingHall {
228                self.built_wedding_hall = Some(true);
229            }
230        }
231        for floor in new_lobby_floors {
232            self.sync_express_stops_for_floor(floor);
233        }
234        self.revision += lands.len() as i64;
235    }
236
237    /// Put a shaft back in the list at index `at` (or the end).
238    pub(crate) fn insert_transport_at(&mut self, at: usize, transport: Transport) {
239        let at = at.min(self.transports.len());
240        self.transports.insert(at, transport);
241        self.reindex_transports();
242        self.revision += 1;
243    }
244}

246/// Where items put back at the indexes `ats` (ascending) land in a list of
247/// `old_len`: each at its index, past the one before it, and no further than
248/// the end of the list as it grows.
249pub(crate) fn landing_places(ats: impl Iterator<Item = usize>, old_len: usize) -> Vec<usize> {
250    let mut lands: Vec<usize> = Vec::new();
251    for (k, at) in ats.enumerate() {
252        let past = lands.last().map_or(0, |&before| before + 1);
253        lands.push(at.max(past).min(old_len + k));
254    }
255    lands
256}

258/// Move the items appended to `items` after its first `old_len` to the
259/// places `lands` names (ascending, one per appended item, in their order),
260/// shifting the items after each place up. One walk back from the end puts
261/// each item it moves in its place with one swap, and allocates nothing the
262/// size of the list.
263///
264/// The appended items still to place fill a window just below the walk, in
265/// an order the walk shuffles: each old item it moves up swaps with the top
266/// of the window, which drops to the bottom. `at_slot[pos % n]` names the
267/// appended item at `pos` and `pos_of` the reverse; the window spans at most
268/// `n` places, so no two of them share a slot.
269pub(crate) fn merge_appended<T>(items: &mut [T], old_len: usize, lands: &[usize]) {
270    let n = lands.len();
271    debug_assert_eq!(items.len(), old_len + n);
272    let mut at_slot: Vec<usize> = vec![0; n];
273    let mut pos_of: Vec<usize> = (old_len..old_len + n).collect();
274    for (k, &pos) in pos_of.iter().enumerate() {
275        at_slot[pos % n] = k;
276    }
277    // The window is `old..write`: the items not placed yet.
278    let mut write = items.len();
279    let mut old = old_len;
280    for k in (0..n).rev() {
281        // Every old item that lands past item `k` moves up first.
282        while write > lands[k] + 1 {
283            write -= 1;
284            old -= 1;
285            let top = at_slot[write % n];
286            items.swap(old, write);
287            at_slot[old % n] = top;
288            pos_of[top] = old;
289        }
290        write -= 1;
291        let from = pos_of[k];
292        if from != write {
293            let top = at_slot[write % n];
294            items.swap(from, write);
295            at_slot[from % n] = top;
296            pos_of[top] = from;
297        }
298    }
