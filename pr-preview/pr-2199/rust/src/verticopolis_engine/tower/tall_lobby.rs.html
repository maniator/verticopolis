1//! Tall lobbies (#308): a lobby one, two or three stories high.
2//!
3//! A tall lobby is lobby units each covering the stories above their floor
4//! as well: the tower's structure index holds the lobby on every story it
5//! covers, so nothing else can be built there, the story above it stands on
6//! it, and like any lobby it stays unless Modern's bulldozer takes a sky
7//! hall's tile (`Tower::removal_reason`). Only its own floor counts
8//! as a lobby floor (`lobby_tiles`), though the distance to a lobby counts
9//! from every story the hall fills (`nearest_lobby_floor_distance`), so the
10//! floor over a tall hall is next to it. The stories it rises through are no
11//! stop: a shaft may pass through them but never stops or ends there, as in
12//! the original, where the shaft shows no floor number on them.
13//!
14//! Each lobby unit carries its own height (`Extras::lobby_stories`), set when
15//! it is laid and never changed after. How a tile's height is chosen is a
16//! mode rule, `GameMode::lobby_height_per_hall`:
17//!
18//! - Classic keeps one height per floor. The floor's first lobby takes the
19//!   height asked for while [`crate::sim::Simulation::with_lobby_height`]
20//!   runs, every later tile takes the same, and the stories it rises through
21//!   are claimed across the whole lot, as the `.TDT` header describes.
22//! - Modern keeps one height per hall (#2068), a hall being a row of lobby
23//!   tiles of one height. A tile laid against a hall takes that hall's
24//!   height (the hall on its left first, so a tile that closes a gap between
25//!   two heights joins the left one); a tile on its own takes the height
26//!   asked for. A tall hall claims only the stories over its own tiles, and
27//!   halls of different heights sit side by side, with a wall and a
28//!   revolving door between them that only the frontend draws.
29//!
30//! Which floors may rise at all is `GameMode::raises_lobby_on`.
31

36    /// How many stories `u` covers: a lobby rises to its own height, every
37    /// other unit to its kind's.
38    pub(crate) fn unit_stories(&self, u: &Unit) -> i64 {
39        if u.kind == Kind::Lobby {
40            u.extras().lobby_stories.map_or(1, i64::from)
41        } else {
42            u.kind.floors()
43        }
44    }

46    /// The height in stories of the tallest lobby hall on `floor`: 1 for a
47    /// one-story lobby and for a floor with no lobby. In Classic every hall
48    /// on a floor has this height.
49    pub fn lobby_height(&self, floor: i64) -> i64 {
50        if !self.floor_has_lobby(floor) {
51            return 1;
52        }
53        (2..=MAX_LOBBY_HEIGHT)
54            .rev()
55            .find(|&h| self.tall_lobby_tiles.contains_key(&(floor, h)))
56            .unwrap_or(1)
57    }

59    /// The lobby unit laid on `floor` itself at column `x`, if any: not one
60    /// rising through `floor` from below. One index lookup.
61    pub(crate) fn hall_at(&self, floor: i64, x: i64) -> Option<&Unit> {
62        let id = self.structure.get(&(floor, x))?;
63        let u = self.get_unit(*id)?;
64        (u.kind == Kind::Lobby && u.floor == floor).then_some(u)
65    }

67    /// The height in stories of the lobby hall at `(floor, x)`: 1 where no
68    /// lobby is laid on that floor at that column.
69    pub fn lobby_height_at(&self, floor: i64, x: i64) -> i64 {
70        self.hall_at(floor, x).map_or(1, |u| self.unit_stories(u))
71    }

73    /// The floor of the tall lobby that rises through some part of `story`,
74    /// when one does. The lobby's own floor is not one of the stories it
75    /// rises through.
76    pub fn tall_lobby_below(&self, story: i64) -> Option<i64> {
77        (1..MAX_LOBBY_HEIGHT)
78            .find(|&rise| self.lobby_height(story.saturating_sub(rise)) > rise)
79            .map(|rise| story - rise)
80    }

82    /// Whether some part of `story` is inside a tall lobby, above its floor.
83    pub fn is_tall_lobby_story(&self, story: i64) -> bool {
84        self.tall_lobby_below(story).is_some()
85    }

87    /// The tall lobby that claims `story` over the columns `x..x + w`, as
88    /// `(its floor, its height)`, or `None` where nothing claims it. In
89    /// Classic a tall lobby claims every story it rises through across the
90    /// lot; in Modern a hall claims only the tiles over its own (#2068),
91    /// which the structure index holds, so this is a lookup per column.
92    pub(crate) fn raised_lobby_at(&self, story: i64, x: i64, w: i64) -> Option<(i64, i64)> {
93        if !self.mode.lobby_height_per_hall() {
94            let floor = self.tall_lobby_below(story)?;
95            return Some((floor, self.lobby_height(floor)));
96        }
97        (x..x.saturating_add(w)).find_map(|col| {
98            let u = self.get_unit(*self.structure.get(&(story, col))?)?;
99            (u.kind == Kind::Lobby && u.floor < story).then(|| (u.floor, self.unit_stories(u)))
100        })
101    }

103    /// Every floor with a lobby taller than one story, as `(floor, height)`
104    /// with its tallest hall's height, by floor. Reads the tile counts, a few
105    /// entries at most.
106    pub fn tall_lobbies(&self) -> Vec<(i64, i64)> {
107        let mut out: Vec<(i64, i64)> = Vec::new();
108        for &(floor, height) in self.tall_lobby_tiles.keys() {
109            match out.last_mut() {
110                Some(last) if last.0 == floor => last.1 = last.1.max(height),
111                _ => out.push((floor, height)),
112            }
113        }
114        out
115    }

117    /// The height a lobby tile laid on `floor` now gets: the floor's own
118    /// height once it holds a lobby, so it never changes; otherwise the
119    /// height asked for, where the mode lets this floor's lobby rise. The
120    /// Classic rule, and the rule for a floor as a whole; a tile's own
121    /// height is [`Self::placement_lobby_height_at`].
122    pub(crate) fn placement_lobby_height(&self, floor: i64) -> i64 {
123        if self.floor_has_lobby(floor) {
124            self.lobby_height(floor)
125        } else if self.mode.raises_lobby_on(floor) {
126            self.requested_lobby_height.clamp(1, MAX_LOBBY_HEIGHT)
127        } else {
128            1
129        }
130    }

132    /// The height a lobby `w` tiles wide laid at `(floor, x)` gets. In
133    /// Classic, the floor's ([`Self::placement_lobby_height`]). In Modern,
134    /// the height of the hall it is laid against, the left one first, or
135    /// the height asked for when it stands on its own (#2068).
136    pub(crate) fn placement_lobby_height_at(&self, floor: i64, x: i64, w: i64) -> i64 {
137        if !self.mode.lobby_height_per_hall() {
138            return self.placement_lobby_height(floor);
139        }
140        if !self.mode.raises_lobby_on(floor) {
141            return 1;
142        }
143        let beside = |col: i64| self.hall_at(floor, col).map(|u| self.unit_stories(u));
144        beside(x - 1)
145            .or_else(|| beside(x.saturating_add(w)))
146            .unwrap_or_else(|| self.requested_lobby_height.clamp(1, MAX_LOBBY_HEIGHT))
147    }

149    /// Why a lobby `w` tiles wide at `(floor, x)` cannot rise to the height
150    /// it would be built at, or `None` when it can (or stays one story).
151    /// A Classic floor's first lobby needs every story it rises through
152    /// empty, since the whole story becomes part of the lobby; any other
153    /// tile needs only the tiles over its own.
154    pub(crate) fn tall_lobby_reason(&self, floor: i64, x: i64, w: i64) -> Option<String> {
155        if let Some(reason) = self.lobby_cannot_rise_reason(floor) {
156            return Some(reason);
157        }
158        let height = self.placement_lobby_height_at(floor, x, w);
159        if height <= 1 {
160            return None;
161        }
162        let whole_story = !self.mode.lobby_height_per_hall() && !self.floor_has_lobby(floor);
163        for story in floor + 1..floor + height {
164            if story > self.lot().top_floor() {
165                return Some(format!(
166                    "A {height}-story lobby needs {height} stories here, and the lot ends at floor {}.",
167                    self.lot().top_floor()
168                ));
169            }
170            let blocked = if whole_story {
171                self.story_has_structure(story)
172            } else {
173                !self.structure_span_free(story, x, w) || !self.room_span_free(story, x, w)
174            };
175            if blocked {
176                return Some(format!(
177                    "A {height}-story lobby rises through floor {story}. Clear floor {story} first."
178                ));
179            }
180        }
181        None
182    }

184    /// Why a floor's first lobby, asked to rise, cannot be laid here: the
185    /// mode keeps this floor's lobby one story. Refused rather than laid one
186    /// story high, since a Classic lobby is never removed and the player asked
187    /// for a taller one. `None` once the floor holds a lobby (every later
188    /// tile takes its height) or when the height asked is one story.
189    fn lobby_cannot_rise_reason(&self, floor: i64) -> Option<String> {
190        if self.requested_lobby_height <= 1
191            || self.floor_has_lobby(floor)
192            || self.mode.raises_lobby_on(floor)
193        {
194            return None;
195        }
196        let which = if self.mode.raises_sky_lobbies() {
197            "Only the ground lobby and sky lobbies can be taller than one story."
198        } else {
199            "In Classic, only the ground lobby can be taller than one story."
200        };
201        Some(format!(
202            "{which} Pick 1 story to build a lobby on floor {floor}."
203        ))
204    }

209        let (lobby, _) = self.raised_lobby_at(story, x, w)?;
210        Some(format!(
211            "Floor {story} is part of the tall lobby on floor {lobby}. Build above the lobby instead."

215    /// The tall-lobby refusal for a structural tile of `kind`, `w` wide at
216    /// `(floor, x)`: a lobby needs room for the stories it rises through
217    /// itself. A story a tall lobby claims is named earlier, with the
218    /// structure already standing (`structure_taken_reason`).
219    pub(super) fn tall_lobby_structure_reason(
220        &self,
221        kind: Kind,
222        floor: i64,
223        x: i64,
224        w: i64,

226        (kind == Kind::Lobby)
227            .then(|| self.tall_lobby_reason(floor, x, w))

232    /// `(floor, x)` now carries: unset for one story.
233    pub(crate) fn placement_lobby_stories(&self, floor: i64, x: i64, w: i64) -> Option<u8> {

238    /// Whether any tile of `story` holds structure. The build preview asks on
239    /// every pointer move, so it reads the revision-keyed structural tile
240    /// index (one lookup) rather than probing each column. That index lists
241    /// the structural units by their own floor, which is every structure a
242    /// story above a floor's first lobby can hold: a tall lobby's raised
243    /// story would make that floor a lobby story, refused before this runs.
244    fn story_has_structure(&self, story: i64) -> bool {
245        self.structural_tiles(story).is_some()
246    }

248    /// Why a shaft over the columns `x..x + w` with ends at `bottom` and
249    /// `top` cannot be, or `None`: no shaft stops inside a tall lobby, so
250    /// none ends there either.
251    pub(crate) fn tall_lobby_shaft_reason(
252        &self,
253        x: i64,
254        w: i64,
255        bottom: i64,
256        top: i64,
257    ) -> Option<String> {
258        let (inside, (floor, height)) = [bottom, top]
259            .into_iter()
260            .find_map(|end| Some((end, self.raised_lobby_at(end, x, w)?)))?;
261        Some(format!(
262            "Floor {inside} is inside the {height}-story lobby on floor {floor}. A shaft can pass through it but can't stop there."
263        ))
264    }

266    /// How many stories the lobby under a flight `w` wide at `(floor, x)`
267    /// rises: the floor's tallest in Classic, the tallest hall under the
268    /// flight's columns in Modern.
269    fn lobby_rise_under(&self, floor: i64, x: i64, w: i64) -> i64 {
270        if !self.mode.lobby_height_per_hall() {
271            return self.lobby_height(floor);
272        }
273        (x..x.saturating_add(w))
274            .map(|col| self.lobby_height_at(floor, col))
275            .max()
276            .unwrap_or(1)
277    }

279    /// The top a flight of stairs or an escalator at `x` from `bottom` to
280    /// `top` gets: a one-story flight out of a tall lobby climbs the whole
281    /// lobby, to the floor above it, as the original's tall flights do (its
282    /// two- and three-story stair types). Any other span stays as asked.
283    pub fn walkway_top(&self, kind: Kind, x: i64, bottom: i64, top: i64) -> i64 {
284        if !(kind.is_fixed_span() && top == bottom + 1) {
285            return top;
286        }
287        let height = self.lobby_rise_under(bottom, x, kind.facility().width);
288        if height > 1 {
289            bottom + height
290        } else {
291            top
292        }
293    }

295    /// The most floors a flight at `x` from `bottom` may span: a tall lobby
296    /// under it lets it climb the whole lobby.
297    pub(super) fn walkway_span_under(&self, kind: Kind, x: i64, bottom: i64) -> i64 {
298        self.lobby_rise_under(bottom, x, kind.facility().width)
299    }

301    /// The stories inside a tall lobby strictly between `t`'s ends, which
302    /// it passes without stopping.
303    pub(crate) fn tall_lobby_stories_in(&self, t: &Transport) -> Vec<i64> {
304        // Walks the tall lobbies, a few at most, never the shaft's span.
305        let tall = self.tall_lobbies().into_iter();
306        tall.flat_map(|(floor, height)| floor + 1..floor + height)
307            .filter(|&story| t.bottom < story && story < t.top)
308            .filter(|&story| self.raised_lobby_at(story, t.x, t.width).is_some())
309            .collect()
310    }

312    /// Adds every tall-lobby story shaft `id` passes to its skip list, so
313    /// it never stops there. Answers whether the list changed; a change is
314    /// noted for the open undo gesture, and the caller bumps the revision.
315    pub(crate) fn skip_tall_lobby_stories(&mut self, id: i64) -> bool {
316        let Some(i) = self.transport_index(id) else {
317            return false;
318        };
319        let skip = self.transports[i].skip_floors.as_deref().unwrap_or(&[]);
320        let missing: Vec<f64> = self
321            .tall_lobby_stories_in(&self.transports[i])
322            .into_iter()
323            .map(|story| story as f64)
324            .filter(|story| !skip.contains(story))
325            .collect();
326        if missing.is_empty() {
327            return false;
328        }
329        let Some(i) = self.transport_index_edit(id) else {
330            return false;
331        };
332        let t = &mut self.transports[i];
333        let mut skip = t.skip_floors.take().unwrap_or_default();
334        skip.extend(missing);
335        skip.sort_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
336        t.skip_floors = Some(skip);
337        true
338    }

340    /// After a tall lobby tile `w` wide is laid at `(floor, x)`, the shafts
341    /// already passing over it skip the stories it now rises through. In
342    /// Classic those shafts skipped them already. Answers whether a skip
343    /// list changed.
344    pub(crate) fn skip_over_new_hall(&mut self, floor: i64, x: i64, w: i64, height: i64) -> bool {
345        let passing: Vec<i64> = self
346            .transports
347            .iter()
348            .filter(|t| t.x < x + w && x < t.x + t.width)
349            .filter(|t| t.bottom < floor + height && t.top > floor + 1)
350            .map(|t| t.id)
351            .collect();
352        let mut changed = false;
353        for id in passing {
354            changed |= self.skip_tall_lobby_stories(id);
355        }
356        changed
357    }
