10    "This save was made by a newer version of Verticopolis. Update the game to load it.";

12/// Builds a game from save JSON (from [`decode_vctower`] or
13/// [`Simulation::serialize`]). An older save is upgraded first, and damaged
14/// or edited fields are corrected the way the web game's loader corrects
15/// them. Refuses a save from a newer version of the engine, and one it cannot
16/// read, with a sentence a player can read.
17pub fn deserialize(raw: &Value) -> Result<Simulation, String> {
18    deserialize_owned(raw.clone())
19}

21/// [`deserialize`] for a save that [`import_tdt`](crate::import_tdt) made from
22/// a 1994 `.TDT` file. It also keeps the basement stops of the original's
23/// express elevators, which a normal load would drop.
24#[doc(hidden)]
25pub fn deserialize_tdt(raw: &Value) -> Result<Simulation, String> {
26    let mut sim = deserialize(raw)?;
27    if sim.mode != GameMode::Classic {
28        return Ok(sim);
29    }
30    let Some(source) = raw.get("transports").and_then(Value::as_array) else {
31        return Ok(sim);
32    };
33    let mut changed = false;
34    for shaft in &mut sim.tower.transports {
35        if shaft.kind != Kind::ElevatorExpress || shaft.bottom >= 0 {
36            continue;
37        }
38        let original = source.iter().find(|t| {
39            t.get("kind").and_then(Value::as_str) == Some("elevatorExpress")
40                && t.get("x").and_then(Value::as_i64) == Some(shaft.x)
41                && t.get("bottom").and_then(Value::as_i64) == Some(shaft.bottom)
42                && t.get("top").and_then(Value::as_i64) == Some(shaft.top)
43        });
44        let Some(authored) = original
45            .and_then(|t| t.get("skipFloors"))
46            .and_then(Value::as_array)
47        else {
48            continue;
49        };
50        let saved_skips: Vec<f64> = authored
51            .iter()
52            .filter_map(Value::as_f64)
53            .filter(|v| v.is_finite())
54            .collect();
55        let Some(skips) = shaft.skip_floors.as_mut() else {
56            continue;
57        };
58        // Mark only shafts actually read from the original TDT. The marker
59        // remains on ordinary .vctower saves and is checked during reload.
60        shaft.preserve_basement_stops = true;
61        let before = skips.len();
62        // Only remove basement skips added by general load normalization.
63        // Authored underground skips and all above-ground rules survive.
64        skips.retain(|floor| *floor > 0.0 || saved_skips.contains(floor));
65        changed |= skips.len() != before;
66    }
67    if changed {
68        sim.tower.revision += 1;
69    }
70    Ok(sim)
71}

73/// [`deserialize`] for save JSON the caller no longer needs: it upgrades the
74/// save in place and frees it as it goes, so a large tower loads in less
75/// memory.
76pub fn deserialize_owned(raw: Value) -> Result<Simulation, String> {
77    // `migrateSave(null)` throws reading `version`; any other top-level value
78    // loads as the TypeScript loads it (#858).
79    if raw.is_null() {
80        return Err("not a tower save (the save is null)".into());
81    }
82    // A save from a build newer than this one may carry fields this loader
83    // would drop, and the next save would write the loss back. Refuse it so
84    // the stored copy stays whole until the game is updated (#1603).
85    if let Some(version) = raw.get("version").and_then(Value::as_f64) {
86        if version.is_finite() && version > SAVE_VERSION as f64 {
87            return Err(NEWER_SAVE_REFUSAL.into());
88        }
89    }
90    // The founder test reads the save as written, and `assemble` reads
91    // nothing else from it, so two keys stand in for it.
92    let written = founder_keys(&raw);
93    let mut data = migrate_save_owned(raw);
94    let modern = data.get("mode").and_then(Value::as_str) == Some("modern");
95    let listed = match data.get_mut("units") {
96        Some(Value::Array(units)) => std::mem::take(units),
97        _ => Vec::new(),
98    };
99    // The entries `unit_from_value` keeps, counted before any unit is built,
100    // so a save past the cap is refused first and the rest fill vectors of
101    // their exact size.
102    let kept = listed
103        .iter()
104        .filter(|u| kind_of(u).is_some())
105        .take(UNIT_CAP + 1)
106        .count();
107    if kept > UNIT_CAP {
108        return Err(format!(
109            "This save lists more than {UNIT_CAP} units, more than the whole lot can hold, so it cannot be a real tower."
110        ));
111    }
112    let mut units = Vec::with_capacity(kept + unit_list_headroom(kept));
113    let mut ids = Vec::with_capacity(kept);
114    for entry in listed {
115        if let Some((unit, id)) = unit_from_value(&entry, modern) {
116            units.push(unit);
117            ids.push(id);
118        }
119    }
120    debug_assert_eq!(units.len(), kept, "unit_from_value skipped a counted entry");
121    let mut sim = assemble(&data, units, ids, &written);
122    // The saved office head counts are the workers at their desks, so a
123    // bulldoze before the first step still sends them out.
124    crate::crowd::commuters::reconcile(&mut sim.crowd, &mut sim.tower);
125    Ok(sim)
126}

128/// The keys `detect_founder` reads off a save.
129pub(crate) fn founder_keys(raw: &Value) -> Value {
130    let mut keys = Map::new();
131    for key in ["founder", "appVersion"] {
132        if let Some(v) = raw.get(key) {
133            keys.insert(key.to_string(), v.clone());
134        }
135    }
136    Value::Object(keys)
137}

139/// One entry of a save's `units` array as `deserialize` reads it, with its
140/// saved id; `None` for an entry it skips (null, or a kind the catalog does
141/// not know), exactly when `kind_of` gives `None`, which the unit cap's
142/// count in `deserialize` relies on. It reads nothing outside the entry but
143/// the save's mode.
144pub(crate) fn unit_from_value(u: &Value, modern: bool) -> Option<(Unit, Option<i64>)> {
145    let kind = kind_of(u)?;
146    let cat = kind.facility().width;
147    // `drop_overlapping_units` indexes its tile grids by the floors and tiles
148    // these clamps allow, so a loosened clamp would trap there.
149    let floor =
150        (round(num(u.get("floor"), 1.0)) as i64).clamp(MIN_FLOOR, kind.highest_base_floor());
151    let x = (round(num(u.get("x"), 0.0)) as i64).clamp(0, LOT - 1);
152    let cap = if kind.is_structural() { cat } else { LOT - x };
153    let width = (round(num(u.get("width"), cat as f64)) as i64)
154        .min(cap)
155        .min(LOT - x)
156        .max(1);
157    let state = if kind.is_structural() {
158        UnitState::Empty
159    } else {
160        parse_state(u.get("state")).unwrap_or(UnitState::Empty)
161    };
162    let not_owned = matches!(
163        state,
164        UnitState::Empty | UnitState::Construction | UnitState::Gutted
165    );
166    let ever_occupied =
167        u.get("everOccupied") == Some(&Value::Bool(true)) && !(not_owned && !kind.is_hotel());
168    let sold_condo = kind == Kind::Condo && ever_occupied;
169    let keep_household = ever_occupied && kind.has_household();
170    let ladder_priced = !modern && classic_ladder(kind).is_some();
171    let mut rent: Option<f64> = match u.get("rent") {
172        None => None,
173        Some(r) if ladder_priced => Some(r.as_f64().unwrap_or(f64::NAN)),
174        Some(r) => Some(num(
175            Some(r),
176            rent_config(kind).map(|c| c.default).unwrap_or(0.0),
177        )),
178    };
179    if let Some(r) = rent {
180        if kind == Kind::Condo {
181            let ladder = if modern {
182                None
183            } else {
184                classic_ladder(Kind::Condo)
185            };
186            if sold_condo {
187                rent = Some(match ladder {
188                    Some(l) => js_max(l[0], js_min(l[3], r)),
189                    None => js_max(SOLD_CONDO_MIN_PRICE, js_min(SOLD_CONDO_MAX_PRICE, r)),
190                });
191            } else if ladder.is_none() {
192                let band = rent_config(Kind::Condo).unwrap();
193                rent = Some(js_max(band.min, js_min(band.max, r)));
194            }
195        }
196    }
197    let residents = if keep_household && modern {
198        match u.get("residents") {
199            None => None,
200            Some(r) => {
201                let v = if r.as_f64().is_some_and(|x| x.is_finite()) {
202                    r.as_f64().unwrap()
203                } else {
204                    3.0
205                };
206                Some((round(v) as i64).clamp(2, 5))
207            }
208        }
209    } else {
210        None
211    };
212    let film_policy = film_policy(u.get("filmPolicy"));
213    let subtype: Option<&'static str> = match (
214        u.get("subtype").and_then(Value::as_str),
215        kind.subtype_list(),
216    ) {
217        (Some(s), Some(list)) => list.iter().copied().find(|c| *c == s),
218        _ => None,
219    };
220    let retail = kind.subtype_list().is_some();
221    let retail_field = |key: &str| -> Option<f64> {
222        if !retail {
223            return None;
224        }
225        let v = u.get(key)?;
226        Some(num(Some(v), 0.0).max(0.0))
227    };
228    let no_rate =
229        rent_config(kind).is_some() && !modern && u.get("noRate") == Some(&Value::Bool(true));
230    let unit = Unit {
231        id: 0,
232        kind,
233        floor,
234        x,
235        width,
236        state,
237        satisfaction: if kind.is_structural() {
238            1.0
239        } else {
240            num(u.get("satisfaction"), 1.0).min(1.0).max(0.0)
241        },
242        occupants: if kind.counts_visitors() || kind.is_structural() {
243            0
244        } else {
245            num(u.get("occupants"), 0.0).clamp(0.0, MAX_SAFE_INTEGER) as i64
246        },
247        customers_in: None,
248        hotel_customers_in: None,
249        out_for_meal: None,
250        residents,
251        visitors_yest: if census_follows_the_day(kind) {
252            u.get("visitorsYest")
253                .map(|v| num(Some(v), 0.0).clamp(0.0, f64::from(u8::MAX)).floor() as u8)
254        } else {
255            None
256        },
257        ever_occupied,
258        pending_income: num(u.get("pendingIncome"), 0.0),
259        rent,
260        no_rate,
261        label: u
262            .get("label")
263            .and_then(Value::as_str)
264            .map(|s| s.to_string())
265            .unwrap_or_else(|| kind.facility().name.to_string()),
266        subtype,
267        extras: Extras {
268            vacate_reason: vacate_reason(u.get("vacateReason")),
269            vacate_at: u.get("vacateAt").map(|v| num(Some(v), 0.0)),
270            film_policy,
271            film_days: film_days(kind, u.get("filmDays")),
272            film_classic: cinema_flag(kind, u.get("filmClassic")),
273            // Only Modern shows a blockbuster (#1950); Classic reads a new film.
274            film_blockbuster: modern && cinema_flag(kind, u.get("filmBlockbuster")),
275            ticket_price: ticket_price(kind, modern, u.get("ticketPrice")),
276            second_screen: second_screen(kind, u.get("screen2")),
277            patronage_today: retail_field("patronageToday"),
278            patronage_yest: retail_field("patronageYest"),
279            profit_today: retail_field("profitToday"),
280            profit_yest: retail_field("profitYest"),
281            complete_at: u.get("completeAt").and_then(Value::as_f64),
282            complete_at_raw: match u.get("completeAt") {
283                Some(v) if !v.is_number() => Some(v.clone()),
284                _ => None,
285            },
286            dirty_days: coerce_dirty_days(state, kind, u.get("dirtyDays")),
287            // Set from the save's `lobbyHeights` once every unit is read.
288            lobby_stories: None,
289        }
290        .without_mirrored_screen()
291        .boxed(),
292    };
293    Some((unit, sane_id(u.get("id"))))
294}
