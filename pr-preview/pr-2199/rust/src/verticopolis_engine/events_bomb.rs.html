1//! Classic's bomb search: a declined ransom hides a bomb in a room, and
2//! Security's guards sweep the tower floor by floor until they find it or
3//! the deadline passes and it goes off (#349). Modern keeps the instant
4//! answer in `events.rs`.
5//!
6//! What the sources agree on (the project's SimTower reference, 2026-10-30):
7//! the bomb always goes off at 1:00 PM (the manual), each office sends six
8//! guards who each cover a floor (kiwizoid), guards walk the peripheral
9//! emergency stairs and never ride the elevators (the manual), a full-width
10//! tower is hard to search in time while a small one usually is (MentalMan,
11//! furdude2), and a missed bomb guts about five floors. The search pace is
12//! our estimate until a run in the original measures it; each number is one
13//! named constant so that run changes a number, not the shape.
14//!
15//! A guard sweeps one stretch of floor at a time (`events_stairs.rs`): the
16//! whole floor in a tower that fills it, one tower's part of it on a lot with
17//! several towers standing apart. Guards reach a stretch only by the
18//! emergency stairs, so a tower no floor joins to a Security office's own is
19//! never searched, and one joined only through a shared lobby costs the walk
20//! down to it and back up.
21

40/// Guards each Security office sends out.
41pub const GUARDS_PER_OFFICE: usize = 6;
42/// The hour of day the bomb goes off (1:00 PM).
43pub const BOMB_HOUR: f64 = 13.0;
44/// Game minutes a guard spends walking one tile of a floor's rooms.
45pub const SWEEP_MINUTES_PER_TILE: f64 = 0.5;
46/// Game minutes a guard spends on the emergency stairs per floor climbed.
47pub const STAIR_MINUTES_PER_FLOOR: f64 = 2.0;
48/// Floors above and below the bomb its blast guts (five floors in all).
49pub const BLAST_FLOORS: i64 = 2;
50
51/// The next 1:00 PM at or after `now`, as a minute count.
52fn deadline_after(now: f64) -> f64 {
53    let today = (now / 1440.0).floor() * 1440.0 + BOMB_HOUR * 60.0;
54    if today > now {
55        today
56    } else {
57        today + 1440.0
58    }
59}

61/// One guard on the search: the floor it is sweeping, or `None` once no
62/// stretch is left for it, the tile `x` that names the stretch on that floor,
63/// and the minute that sweep ends. `from` and `start` are where and when it
64/// set off, `route` the corners of its walk on the emergency stairs (the
65/// floor and stairs tile where it sets off, turns back or changes stairs,
66/// and steps off; empty for a guard from an older save, which climbed
67/// straight), and `arrive` the minute it reached the stretch, so a host can
68/// draw the walk on the stairs before the walk along the rooms.
69#[derive(Clone, Debug, PartialEq)]
70pub struct Guard {
71    pub floor: Option<i64>,
72    /// `None` for a guard from a save made before towers were searched one by
73    /// one: it sweeps its whole floor.
74    pub x: Option<i64>,
75    pub done_at: f64,
76    pub from: i64,
77    pub route: Vec<(i64, i64)>,
78    pub start: f64,
79    pub arrive: f64,
80}

82/// A stretch swept clean: its floor and a tile of it, or `None` for a whole
83/// floor (a save from before towers were searched one by one).
84pub type Swept = (i64, Option<i64>);

86/// A bomb waiting to be found. `floor` and `x` are its hiding place (the
87/// player never sees them), `deadline` the minute it goes off.
88#[derive(Clone, Debug, PartialEq)]
89pub struct BombSearch {
90    pub floor: i64,
91    pub x: f64,
92    pub deadline: f64,
93    pub guards: Vec<Guard>,
94    /// Stretches swept clean, in the order the guards finished them.
95    pub searched: Vec<Swept>,
96    /// Bumped on every change a viewer can see, so a host rereads the search
97    /// only when it moved.
98    pub seq: i64,
99}

101/// The stretch a guard sets off for: its floor and first tile, the width of
102/// its rooms, the floors of stairs on the way and the corners of that walk.
103struct Target {
104    floor: i64,
105    x: i64,
106    width: i64,
107    floors: i64,
108    route: Vec<(i64, i64)>,
109}

111/// The stretches already swept or taken by a guard, so each is swept once.
112/// Whole floors stand for the guards and sweeps of an older save.
113#[derive(Default)]
114struct Claims {
115    stretches: BTreeSet<usize>,
116    floors: BTreeSet<i64>,
117}
118
119impl Claims {
120    fn claim(&mut self, stairs: &Stairs, floor: i64, x: Option<i64>) {
121        match x {
122            None => {
123                self.floors.insert(floor);
124            }
125            Some(x) => {
126                if let Some(i) = stairs.at(floor, x) {
127                    self.stretches.insert(i);
128                }
129            }
130        }
131    }
132
133    fn holds(&self, i: usize, floor: i64) -> bool {
134        self.stretches.contains(&i) || self.floors.contains(&floor)
135    }
136
137    /// Everything swept so far and every stretch a guard is headed for.
138    fn of(stairs: &Stairs, b: &BombSearch) -> Claims {
139        let mut c = Claims::default();
140        for &(f, x) in &b.searched {
141            c.claim(stairs, f, x);
142        }
143        for g in &b.guards {
144            if let Some(f) = g.floor {
145                c.claim(stairs, f, g.x);
146            }
147        }
148        c
149    }
150}

152/// `h:mm AM` for a minute count, as the log writes a time of day.
153fn clock_label(minutes: f64) -> String {
154    let of_day = ((minutes % 1440.0) + 1440.0) % 1440.0;
155    let (h, m) = (
156        (of_day / 60.0).floor() as i64,
157        (of_day % 60.0).floor() as i64,
158    );
159    let half = if h < 12 { "AM" } else { "PM" };
160    let h12 = if h % 12 == 0 { 12 } else { h % 12 };
161    format!("{h12}:{m:02} {half}")
162}

165impl Simulation {
166    /// The answer to a declined ransom in both modes: hide the bomb in a random room
167    /// and send Security out. With no room to hide it in, the threat was
168    /// empty; with no Security, nobody searches and it simply waits for the
169    /// deadline.
170    pub(super) fn start_bomb_search(&mut self) {
171        let targets = self.flammable_units();
172        if targets.is_empty() {
173            self.emit(
174                "💣 The bomb threat was a hoax: there was nowhere to hide a bomb.",
175                LogKind::Good,
176            );
177            return;
178        }
179        let gi = *self.rng.pick(&targets);
180        let room = &self.tower.units[gi];
181        let (floor, x) = (room.floor, room.x as f64 + room.width as f64 / 2.0);
182        let now = self.clock.minutes;
183        let deadline = deadline_after(now);
184        let offices: Vec<(i64, i64)> = self
185            .tower
186            .units
187            .iter()
188            .filter(|u| u.kind == Kind::Security && u.is_operational())
189            .map(|u| (u.floor, u.x))
190            .collect();
191        self.events.bomb = Some(BombSearch {
192            floor,
193            x,
194            deadline,
195            guards: Vec::new(),
196            searched: Vec::new(),
197            seq: 1,
198        });
199        let mut guards = Vec::with_capacity(offices.len() * GUARDS_PER_OFFICE);
200        if !offices.is_empty() {
201            let stairs = Stairs::of(&self.tower);
202            let rooms = self.stretch_rooms(&stairs);
203            let mut claims = Claims::default();
204            for &(office, ox) in &offices {
205                guards.extend(self.send_guards(&stairs, &rooms, &mut claims, office, ox, now));
206            }
207        }
208        if let Some(b) = self.events.bomb.as_mut() {
209            b.guards = guards;
210        }
211        let when = clock_label(deadline);
212        if offices.is_empty() {
213            self.emit(&format!("💣 You refused the ransom. With no Security to search, the bomb goes off at {when}. Build Security!"), LogKind::Bad);
214        } else {
215            let n = offices.len() * GUARDS_PER_OFFICE;
216            self.emit(&format!("💣 You refused the ransom. {n} Security guards are searching the tower; the bomb goes off at {when}."), LogKind::Bad);
217        }
218    }

220    /// Whether the tower is on lockdown for a bomb: while any of Security's
221    /// guards is still out searching for it, in both modes. People and cars
222    /// hold still meanwhile and only the guards move. A bomb nobody searches
223    /// for (no Security, or every stretch the guards can reach swept) freezes
224    /// nothing while it waits for the deadline (owner, 2026-11-13).
225    pub fn bomb_lockdown(&self) -> bool {
226        self.events
227            .bomb
228            .as_ref()
229            .is_some_and(|b| b.guards.iter().any(|g| g.floor.is_some()))
230    }

232    /// A Security office at (`office`, `x`) just opened. In a mode where late
233    /// offices join, its guards set out from there now.
234    pub(crate) fn join_bomb_search(&mut self, office: i64, x: i64) {
235        let Some(b) = self.events.bomb.as_ref() else {
236            return;
237        };
238        if !self.mode.security_joins_bomb_search() {
239            return;
240        }
241        let now = self.clock.minutes;
242        let stairs = Stairs::of(&self.tower);
243        let rooms = self.stretch_rooms(&stairs);
244        let mut claims = Claims::of(&stairs, b);
245        let fresh = self.send_guards(&stairs, &rooms, &mut claims, office, x, now);
246        if let Some(b) = self.events.bomb.as_mut() {
247            b.guards.extend(fresh);
248            b.seq += 1;
249        }
250        self.emit(
251            &format!("💣 {GUARDS_PER_OFFICE} more Security guards joined the bomb search."),
252            LogKind::Good,
253        );
254    }

256    /// One office's guards setting out at minute `now` from its stretch, each
257    /// to the nearest stretch by the stairs that nobody has claimed. An
258    /// office on no stretch of floor (never placed by the game) sends guards
259    /// with nowhere to go.
260    fn send_guards(
261        &self,
262        stairs: &Stairs,
263        rooms: &BTreeMap<usize, (i64, i64)>,
264        claims: &mut Claims,
265        office: i64,
266        x: i64,
267        now: f64,
268    ) -> Vec<Guard> {
269        let start = stairs.at(office, x);
270        let walk = start.map(|s| stairs.walk_from(s));
271        (0..GUARDS_PER_OFFICE)
272            .map(|_| {
273                let next = start
274                    .zip(walk.as_ref())
275                    .and_then(|(s, w)| next_stretch(stairs, rooms, claims, s, w));
276                if let Some(t) = &next {
277                    claims.claim(stairs, t.floor, Some(t.x));
278                }
279                sweep(next, office, now)
280            })
281            .collect()
282    }

284    /// Whether guard `g` is sweeping the bomb's stretch.
285    fn sweeps_bomb(stairs: &Stairs, b: &BombSearch, g: &Guard) -> bool {
286        if g.floor != Some(b.floor) {
287            return false;
288        }
289        let Some(x) = g.x else {
290            return true;
291        };
292        let here = stairs.at(b.floor, x);
293        here.is_some() && here == stairs.at(b.floor, b.x.floor() as i64)
294    }

296    /// One step of the search: guards who finished a stretch move on (a big
297    /// step can finish several for one guard), then a guard who swept the
298    /// bomb's stretch by the deadline finds it, and otherwise at the deadline
299    /// it goes off. The stairs are mapped only when a sweep has ended.
300    pub fn advance_bomb_search(&mut self) {
301        let Some(b) = self.events.bomb.as_ref() else {
302            return;
303        };
304        // Nothing after the deadline can change the outcome.
305        let now = self.clock.minutes.min(b.deadline);
306        let due = b
307            .guards
308            .iter()
309            .any(|g| g.floor.is_some() && g.done_at <= now);
310        let mut found = false;
311        if due {
312            let stairs = Stairs::of(&self.tower);
313            self.move_finished_guards(now, &stairs);
314            if let Some(b) = self.events.bomb.as_ref() {
315                found = b
316                    .guards
317                    .iter()
318                    .any(|g| g.done_at <= now && Self::sweeps_bomb(&stairs, b, g));
319            }
320        }
321        let Some(b) = self.events.bomb.as_ref() else {
322            return;
323        };
324        let deadline = b.deadline;
325        if found {
326            let floor = b.floor;
327            self.events.bomb = None;
328            let cost = 2_000.0 + self.rng.int(0, 3_000) as f64;
329            self.money -= cost;
330            self.gameplay.push(GameplayEvent::BombFound);
331            let at = self.floor_label(floor);
332            self.emit(&format!("💣 Security found the bomb on {at} and defused it in time. The evacuation cost ${}.", with_thousands(cost)), LogKind::Good);
333        } else if self.clock.minutes >= deadline {
334            let (floor, x) = (b.floor, b.x);
335            self.events.bomb = None;
336            let fine = 15_000.0 + self.rng.int(0, 15_000) as f64;
337            let destroyed = self.blast(fine, Some((floor, x)));
338            self.emit(&format!("💣 Security ran out of time and the bomb went off on {}. {destroyed} room(s) across ~5 floors were gutted, plus a ${} fine.", self.floor_label(floor), with_thousands(fine)), LogKind::Bad);
339        }
340    }

342    /// Every guard whose sweep of a stretch without the bomb has ended by
343    /// `now` marks it clear and starts the next, from the minute it finished,
344    /// until each is busy past `now` or out of stretches. One walk of the
345    /// rooms.
346    fn move_finished_guards(&mut self, now: f64, stairs: &Stairs) {
347        let rooms = self.stretch_rooms(stairs);
348        let Some(b) = self.events.bomb.as_ref() else {
349            return;
350        };
351        let mut claims = Claims::of(stairs, b);
352        let mut moved = false;
353        loop {
354            let Some(b) = self.events.bomb.as_ref() else {
355                return;
356            };
357            // The earliest finish goes first (the lowest index on a tie), so
358            // one long step hands out stretches in the order short steps would.
359            let Some(i) = b
360                .guards
361                .iter()
362                .enumerate()
363                .filter(|(_, g)| {
364                    g.floor.is_some() && g.done_at <= now && !Self::sweeps_bomb(stairs, b, g)
365                })
366                .min_by(|(_, a), (_, b)| a.done_at.total_cmp(&b.done_at))
367                .map(|(i, _)| i)
368            else {
369                break;
370            };
371            let bomb_tile = b.x.floor() as i64;
372            let Some(b) = self.events.bomb.as_mut() else {
373                return;
374            };
375            let done = b.guards[i].clone();
376            let from = done.floor.unwrap_or(b.floor);
377            b.searched.push((from, done.x));
378            // A guard from an older save swept its whole floor: it leaves
379            // from the floor's first stretch with rooms.
380            let start = match done.x {
381                Some(x) => stairs.at(from, x),
382                None => rooms
383                    .keys()
384                    .copied()
385                    .find(|&s| stairs.stretch(s).floor == from)
386                    .or_else(|| stairs.at(from, bomb_tile)),
387            };
388            let next = start.and_then(|s| {
389                let walk = stairs.walk_from(s);
390                next_stretch(stairs, &rooms, &claims, s, &walk)
391            });
392            if let Some(t) = &next {
393                claims.claim(stairs, t.floor, Some(t.x));
394            }
395            b.guards[i] = sweep(next, from, done.done_at);
396            moved = true;
397        }
398        if moved {
399            if let Some(b) = self.events.bomb.as_mut() {
400                b.seq += 1;
401            }
402        }
403    }
404}

406/// The nearest stretch by the stairs from `start` (along `walk`) that has
407/// rooms and nobody has claimed; the upper one when two are as near, the
408/// left one on one floor. `None` when every stretch the stairs reach is
409/// covered.
410fn next_stretch(
411    stairs: &Stairs,
412    rooms: &BTreeMap<usize, (i64, i64)>,
413    claims: &Claims,
414    start: usize,
415    walk: &Walk,
416) -> Option<Target> {
417    rooms
418        .iter()
419        .filter_map(|(&i, &(left, right))| {
420            let s = stairs.stretch(i);
421            let floors = walk.floors[i]?;
422            (!claims.holds(i, s.floor)).then_some((i, s.floor, floors, right - left))
423        })
424        .min_by_key(|&(i, floor, floors, _)| (floors, -floor, i))
425        .map(|(i, floor, floors, width)| Target {
426            floor,
427            x: stairs.stretch(i).first,
428            width,
429            floors,
430            route: stairs.route(walk, start, i),
431        })
432}

434/// A guard leaving `from` at minute `start` for `next`: the stairs there,
435/// then the walk along the rooms.
436fn sweep(next: Option<Target>, from: i64, start: f64) -> Guard {
437    match next {
438        Some(t) => {
439            let arrive = start + t.floors as f64 * STAIR_MINUTES_PER_FLOOR;
440            Guard {
441                floor: Some(t.floor),
442                x: Some(t.x),
443                done_at: arrive + t.width as f64 * SWEEP_MINUTES_PER_TILE,
444                from,
445                route: t.route,
446                start,
447                arrive,
448            }
449        }
450        None => Guard {
451            floor: None,
452            x: None,
453            done_at: start,
454            from,
455            route: Vec::new(),
456            start,
457            arrive: start,
458        },
459    }
