162/// Bytes a built elevator slot appends after its header: the fixed block,
163/// one per-floor entry for every floor the shaft spans and a single car
164/// block. An inverted span is not a shaft the game can have written, so it
165/// is refused rather than sized.
166pub fn built_shaft_payload_size(bottom_floor: i64, top_floor: i64) -> Result<i64, String> {
167    let spanned_floors = top_floor - bottom_floor + 1;
168    if spanned_floors < 1 {
169        return Err(format!(
170            "elevator span must be a whole number of floors, at least one (bottom {bottom_floor}, top {top_floor})"
171        ));
172    }
173    Ok(TDT_ELEVATOR_BUILT_FIXED
174        + spanned_floors * TDT_ELEVATOR_PER_FLOOR_SIZE
175        + TDT_ELEVATOR_CAR_BLOCK_SIZE)
176}

178/// One built shaft's payload size the way the 1994 game sizes it, by kind:
179/// standard and service by spanned floors, express by the floors it stops at
180/// (clamped up to one entry).
181pub fn built_shaft_payload_size_for(
182    type_id: i64,
183    bottom_floor: i64,
184    top_floor: i64,
185    serviced_count: i64,
186) -> Result<i64, String> {
187    let spanned = built_shaft_payload_size(bottom_floor, top_floor)?;
188    if type_id != TDT_ELEVATOR_TYPE_EXPRESS {
189        return Ok(spanned);
190    }
191    built_shaft_payload_size(1, serviced_count.max(1))
192}

194/// The floors an express shaft carries a payload entry for: the ones from
195/// `bottom_floor` to `top_floor` (file numbering) that are at or below the
196/// first floor, or a multiple of 15 (the sky lobbies). All eight express
197/// shafts in the nine original saves size to this count.
198pub fn express_eligible_floors(bottom_floor: i64, top_floor: i64) -> i64 {
199    (bottom_floor..=top_floor)
200        .filter(|raw| {
201            let floor = raw - TDT_FLOOR_OFFSET;
202            floor <= 1 || floor % 15 == 0
203        })
204        .count() as i64
205}
206
207/// One express shaft's payload size: an entry for each eligible floor,
208/// and at least one.
209pub fn express_payload_size(bottom_floor: i64, top_floor: i64) -> Result<i64, String> {
210    built_shaft_payload_size(bottom_floor, top_floor)?;
211    built_shaft_payload_size(1, express_eligible_floors(bottom_floor, top_floor).max(1))
212}

225/// The stair record type for a flight of stairs (or an escalator) climbing
226/// `stories`: 0 or 1 for one story, 2 or 3 for two, 4 or 5 for three, the
227/// inverse of [`tdt_stair_stories`]. The multi-story types are the flights
228/// out of a tall ground lobby (#308).
229pub fn tdt_stair_type(stairs: bool, stories: i64) -> i64 {
230    i64::from(stairs) + 2 * (stories.clamp(1, 3) - 1)

233/// `viewWordsFromView`: a saved camera view (center in our grid units) to
234/// the header's view-scroll words. Non-finite members fall back to the New
235/// Tower default; finite values clamp into the scrollable range, and the
236/// (0, 0) sentinel is never emitted for a real view.
237pub fn view_words_from_view(tile: f64, floor: f64) -> (i64, i64) {
238    if !tile.is_finite() || !floor.is_finite() {
239        return (TDT_DEFAULT_VIEW_X, TDT_DEFAULT_VIEW_Y);
240    }
241    let f_tdt = floor + TDT_FLOOR_OFFSET as f64;
242    let center_x = tile * TDT_TILE_PX;
243    let center_y = (TDT_FLOOR_COUNT as f64 - 1.0 - f_tdt) * TDT_FLOOR_PX + TDT_FLOOR_PX / 2.0;
244    let x = round(center_x - TDT_VIEW_W / 2.0)
245        .min(TDT_WORLD_W - TDT_VIEW_W)
246        .max(0.0);
247    let y = round(center_y - TDT_VIEW_H / 2.0)
248        .min(TDT_WORLD_H - TDT_VIEW_H)
249        .max(0.0);
250    let (x, y) = (x as i64, y as i64);
251    if x == 0 && y == 0 {
252        (0, 1)
253    } else {
254        (x, y)
255    }
256}

258/// `viewFromViewWords`: the inverse, header words to a camera center in our
259/// grid units; (0, 0) is "no saved view" and maps to `None`. Values stay
260/// fractional and unclamped, as the TypeScript leaves them.
261pub fn view_from_view_words(x: i64, y: i64) -> Option<(f64, f64)> {
262    if x == 0 && y == 0 {
263        return None;
264    }
265    let tile = (x as f64 + TDT_VIEW_W / 2.0) / TDT_TILE_PX;
266    let center_y = y as f64 + TDT_VIEW_H / 2.0;
267    let f_tdt = TDT_FLOOR_COUNT as f64 - 1.0 - (center_y - TDT_FLOOR_PX / 2.0) / TDT_FLOOR_PX;
268    Some((tile, f_tdt - TDT_FLOOR_OFFSET as f64))
269}

271/// `parseTdtBinary`: walk a `.TDT` byte stream into the raw model. The
272/// header and floor map are load-bearing; everything after is walked
273/// tolerantly, downgrading misfits to warnings.
274pub fn parse_tdt_binary(bytes: &[u8]) -> Result<TdtTower, LegacyImportError> {
275    if bytes.len() > TDT_MAX_FILE_BYTES {
276        return Err(LegacyImportError(
277            "This file is too large to be a SimTower save.".into(),
278        ));
279    }
280    if bytes.len() < TDT_HEADER_SIZE {
281        return Err(LegacyImportError(
282            "This file is too small to be a SimTower save.".into(),
283        ));
284    }
285    let mut r = ByteReader::new(bytes);
286    r.enter_block("header");
287    let version = r.u16()?;
288    if version != TDT_MAGIC {
289        return Err(LegacyImportError(
290            "This file doesn't look like a SimTower save.".into(),
291        ));
292    }
293    let level = r.u16()?;
294    let balance = r.i32()?;
295    r.skip(12)?; // otherIncome, constructionCosts, lastQuarterMoney
296    let frame_time = r.u16()?;
297    let current_day = r.i32()?;
298    r.skip(TDT_HEADER_LOBBY_HEIGHT_AT - r.offset())?;
299    let lobby_height = r.u16()?;
300    r.skip(0x26 - r.offset())?;
301    let view_x = r.u16()?;
302    let view_y = r.u16()?;
303    r.skip(TDT_HEADER_SIZE - r.offset())?;
304    let header = TdtHeader {
305        version,
306        level,
307        balance,
308        frame_time,
309        current_day,
310        lobby_height,
311        view_x,
312        view_y,
313    };
314
315    let mut floors = Vec::with_capacity(TDT_FLOOR_COUNT as usize);
316    for index in 0..TDT_FLOOR_COUNT {
317        r.enter_block(&format!("floor map (floor record {index})"));
318        let tenant_count = r.u16()?;
319        let left_edge = r.u16()?;
320        let right_edge = r.u16()?;
321        if tenant_count > TDT_MAX_TENANTS_PER_FLOOR {
322            return Err(LegacyImportError(format!(
323                "This SimTower save is corrupt: one floor claims {} rooms.",
324                with_thousands(tenant_count as f64)
325            )));
326        }
327        let floor_bytes =
328            tenant_count as usize * TDT_TENANT_RECORD_SIZE + TDT_FLOOR_INDEX_ENTRIES * 2;
329        if r.remaining() < floor_bytes {
330            return Err(LegacyImportError(
331                "This SimTower save is cut short. The file ends in the middle of its floor map."
332                    .into(),
333            ));
334        }
335        let mut tenants = Vec::with_capacity(tenant_count as usize);
336        for _ in 0..tenant_count {
337            let left = r.u16()?;
338            let right = r.u16()?;
339            let type_id = r.i8()?;
340            let status_at = r.offset();
341            let status = r.u8()?;
342            let variant = r.u8()?;
343            let row_high = r.u8()?;
344            r.skip(7)?;
345            let draw = r.u8()?;
346            let rent_rate = r.u8()?;
347            let subtype = r.u8()?;
348            tenants.push(TdtTenant {
349                left,
350                right,
351                type_id,
352                status,
353                status_at,
354                variant,
355                row_high,
356                draw,
357                rent_rate,
358                subtype,
359            });
360        }
361        let mut index_map = Vec::with_capacity(TDT_FLOOR_INDEX_ENTRIES);
362        for _ in 0..TDT_FLOOR_INDEX_ENTRIES {
363            index_map.push(r.u16()? as u16);
364        }
365        floors.push(TdtFloor {
366            index,
367            left_edge,
368            right_edge,
369            tenants,
370            index_map,
371        });
372    }
373
374    let tail = walk_tolerant_tail(&mut r);
375    Ok(TdtTower {
376        header,
377        floors,
378        tail,
379    })
380}

386    #[test]
387    fn an_express_shaft_has_an_entry_for_each_floor_at_or_below_the_first_and_each_sky_lobby() {
388        // Floors 0 to 60: the ground, the first and the lobbies at 15, 30, 45, 60.
389        let raw = |floor: i64| floor + TDT_FLOOR_OFFSET;
390        assert_eq!(express_eligible_floors(raw(0), raw(60)), 6);
391        // Floors -9 to 60 add the nine basements and the ground below the first.
392        assert_eq!(express_eligible_floors(raw(-9), raw(60)), 15);
393        assert_eq!(express_eligible_floors(raw(2), raw(14)), 0);
394        assert_eq!(
395            express_payload_size(raw(2), raw(14)),
396            Ok(3140 + 324 + 348),
397            "an entry is kept for a shaft with no eligible floor"
398        );
399        assert_eq!(
400            express_payload_size(raw(0), raw(60)),
401            Ok(3140 + 6 * 324 + 348)
402        );
403        assert!(express_payload_size(raw(20), raw(15)).is_err());
404    }

406    #[test]
407    fn payload_size_is_car_independent_and_refuses_inverted_spans() {
408        assert_eq!(built_shaft_payload_size(10, 10), Ok(3140 + 324 + 348));
409        assert_eq!(built_shaft_payload_size(10, 100), Ok(3140 + 91 * 324 + 348));
410        assert!(built_shaft_payload_size(20, 15).is_err());
411        assert_eq!(
412            built_shaft_payload_size_for(0, 10, 100, 8),
413            Ok(3140 + 8 * 324 + 348)
414        );
415        assert_eq!(
416            built_shaft_payload_size_for(0, 10, 100, 0),
417            Ok(3140 + 324 + 348)
418        );
419        assert!(built_shaft_payload_size_for(0, 20, 15, 3).is_err());
420    }
