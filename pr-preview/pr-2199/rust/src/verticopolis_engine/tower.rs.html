29/// Whether the `w` tiles from `x` on one story have floor on both sides,
30/// so they are not at either end of the story and taking them out in
31/// Classic would leave a gap in its middle, or widen one a loaded save
32/// already has. `tile_at` answers the kind of the floor or lobby tile at a
33/// tile of that story, if one stands there. The bulldozer
34/// (`Tower::removal_reason`) asks it, and the undo's sale advice
35/// (`Ground::sells_from_an_end`) asks its two sides, so they agree.
36pub(crate) fn span_has_floor_on_both_sides(
37    x: i64,
38    w: i64,
39    tile_at: impl Fn(i64) -> Option<Kind>,
40) -> bool {
41    floor_left_of(x, &tile_at) && floor_right_of(x + w, &tile_at)
42}
43
44/// Whether the story has floor on the left of the tile at `x`, by Classic's
45/// floor end rule: plain floor stands anywhere left of it, or the tiles
46/// touching it on the left, unbroken by a gap, hold a lobby tile (owner
47/// ruling, 2026-11-13, on #2062). Classic never removes a lobby tile, so counting
48/// one beyond a gap would trap the plain floor between it and the far end
49/// for good; a lobby the tile touches still counts, so a floor tile right
50/// beside a lobby is not an end. Plain floor anywhere counts, so the tiles
51/// beside a gap a saved tower has are not ends and the gap is not cut wider
52/// one tile at a time. A plain tile in the touching run counts on its own,
53/// so the touching run counts exactly when the tile right beside holds floor
54/// or lobby.
55pub(crate) fn floor_left_of(x: i64, tile_at: impl Fn(i64) -> Option<Kind>) -> bool {
56    touches_floor_or_lobby(x - 1, &tile_at) || plain_floor_on(0..x, &tile_at)
57}
58
59/// Whether the story has floor on the right of the tiles that end before
60/// `end`, by the rule of [`floor_left_of`].
61pub(crate) fn floor_right_of(end: i64, tile_at: impl Fn(i64) -> Option<Kind>) -> bool {
62    touches_floor_or_lobby(end, &tile_at) || plain_floor_on(end..LOT_WIDTH, &tile_at)
63}
64
65fn touches_floor_or_lobby(beside: i64, tile_at: impl Fn(i64) -> Option<Kind>) -> bool {
66    (0..LOT_WIDTH).contains(&beside) && tile_at(beside).is_some()
67}
68
69fn plain_floor_on(side: std::ops::Range<i64>, tile_at: impl Fn(i64) -> Option<Kind>) -> bool {
70    side.into_iter().any(|tx| tile_at(tx) == Some(Kind::Floor))
71}

123/// The room a loaded game's unit list keeps past its units. A list with no
124/// spare room moves to a new block on the first placement while it still holds
125/// the old one, which on the large perf tower took the engine's WASM memory
126/// from 24.6 to 52.4 MiB for one floor tile (#1259), and a WASM memory keeps
127/// its peak for the session. An eighth, at least 64, holds the placements most
128/// sessions make; past it `Tower::add_unit` grows the list by a quarter.
129pub(crate) fn unit_list_headroom(units: usize) -> usize {
130    (units / 8).max(64)
131}

267    /// Why structure `kind` cannot go over `w` tiles at `(floor, x)` for
268    /// what already stands there, or `None`. A tall lobby's raised story is
269    /// named first (#2068), since its tiles are the lobby's own structure;
270    /// then built structure a lobby cannot be laid over, then rooms on the
271    /// plain floor a lobby would take over.
272    fn structure_taken_reason(&self, kind: Kind, floor: i64, x: i64, w: i64) -> Option<String> {
273        if let Some(r) = self.tall_lobby_story_reason(floor, x, w) {
274            return Some(r);
275        }
276        if self.structure_span_free(floor, x, w) {
277            return None;
278        }
279        if kind != Kind::Lobby || !self.span_upgradeable_to_lobby(floor, x, w) {
280            return Some("Structure already here.".into());
281        }
282        (!self.room_span_free(floor, x, w))
283            .then(|| "Lobbies are transit-only. Clear the rooms here first.".into())
284    }

298    pub(crate) fn register(&mut self, idx: usize) {
299        let u = self.units[idx].clone();
300        self.by_id.insert(u.id, idx);
301        let hgt = self.unit_stories(&u);
302        let structural = u.kind.is_structural();
303        for fl in 0..hgt {
304            for i in 0..u.width {
305                let key = (u.floor + fl, u.x + i);
306                if structural {
307                    self.structure.insert(key, u.id);
308                    self.struct_kind.insert(key, u.kind);
309                } else {
310                    self.rooms.insert(key, u.id);
311                }
312            }
313            if !structural {
314                *self.room_tiles.entry(u.floor + fl).or_insert(0) += u.width;
315            }
316        }
317        if u.kind == Kind::Lobby {
318            *self.lobby_tiles.entry(u.floor).or_insert(0) += u.width;
319            if hgt > 1 {
320                *self.tall_lobby_tiles.entry((u.floor, hgt)).or_insert(0) += u.width;
321            }
322        }
323        if u.kind == Kind::Metro {
324            self.memo.get_mut().metro_changes += 1;
325        }
326    }
327
328    pub(crate) fn unregister(&mut self, u: &Unit) {
329        self.by_id.remove(&u.id);
330        let hgt = self.unit_stories(u);
331        let structural = u.kind.is_structural();
332        for fl in 0..hgt {
333            for i in 0..u.width {
334                let key = (u.floor + fl, u.x + i);
335                if structural {
336                    self.structure.remove(&key);
337                    self.struct_kind.remove(&key);
338                } else {
339                    self.rooms.remove(&key);
340                }
341            }
342            if !structural {
343                Self::dec_count(&mut self.room_tiles, u.floor + fl, u.width);
344            }
345        }
346        if u.kind == Kind::Lobby {
347            if let Some(c) = self.lobby_tiles.get_mut(&u.floor) {
348                *c -= u.width;
349                if *c <= 0 {
350                    self.lobby_tiles.remove(&u.floor);
351                }
352            }
353            if hgt > 1 {
354                let key = (u.floor, hgt);
355                if let Some(c) = self.tall_lobby_tiles.get_mut(&key) {
356                    *c -= u.width;
357                    if *c <= 0 {
358                        self.tall_lobby_tiles.remove(&key);
359                    }
360                }
361            }
362        }
363        if u.kind == Kind::Metro {
364            self.memo.get_mut().metro_changes += 1;
365        }
366    }
367
368    fn dec_count(map: &mut HashMap<i64, i64>, floor: i64, by: i64) {
369        if let Some(c) = map.get_mut(&floor) {
370            *c -= by;
371            if *c <= 0 {
372                map.remove(&floor);
373            }
374        }
375    }

379    pub fn cap_reason(&self, kind: Kind) -> Option<String> {
380        if let Some(single) = build_cap_in(self.mode, kind) {
381            if self.count_kind(kind) >= single {
382                let f = kind.facility();
383                return Some(format!(
384                    "Only {single} {}{} allowed per tower.",
385                    f.name,
386                    if single == 1 { "" } else { "s" }
387                ));
388            }
389        }
390        for pool in pools_for(self.mode) {
391            if !pool.kinds.contains(&kind) {
392                continue;
393            }
394            let total: i64 = pool.kinds.iter().map(|&k| self.count_kind(k)).sum();
395            if total >= pool.cap {
396                return Some(format!(
397                    "Only {} {} allowed per tower.",
398                    pool.cap, pool.label
399                ));
400            }
401        }
402        None
403    }

407    /// `reindex()`: rebuild every per-tile index after a bulk unit assignment.
408    pub fn reindex(&mut self) {
409        self.structure.clear();
410        self.struct_kind.clear();
411        self.rooms.clear();
412        self.by_id.clear();
413        self.lobby_tiles.clear();
414        self.tall_lobby_tiles.clear();
415        self.room_tiles.clear();
416        for i in 0..self.units.len() {
417            self.register(i);
418        }
419        self.reindex_transports();
420        *self.memo.borrow_mut() = Default::default();
421        self.revision += 1;
422    }
423
424    /// `coerceExpressStops()`: re-assert the lobby-only express rule on load.
425    pub fn coerce_express_stops(&mut self) {
426        let mut changed = false;
427        for i in 0..self.transports.len() {
428            if self.transports[i].kind != Kind::ElevatorExpress {
429                continue;
430            }
431            let (bottom, top) = (self.transports[i].bottom, self.transports[i].top);
432            let mut skip: Vec<f64> = self.transports[i].skip_floors.clone().unwrap_or_default();
433            // Never infer PC TDT provenance from skip-list signs. An authentic
434            // file can skip none, some, or all underground floors.
435            let keep_basement_stops = self.mode.keeps_preserved_basement_stops()
436                && bottom < 0
437                && self.transports[i].preserve_basement_stops;
438            for fl in bottom + 1..top {
439                if keep_basement_stops && fl <= 0 {
440                    continue;
441                }
442                if !self.floor_has_lobby(fl) && !skip.contains(&(fl as f64)) {
443                    skip.push(fl as f64);
444                }
445            }
446            skip.retain(|&f| f != bottom as f64 && f != top as f64);
447            let mut next: Vec<f64> = Vec::new();
448            for f in skip {
449                if !next.contains(&f) {
450                    next.push(f);
451                }
452            }
453            next.sort_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
454            let before = self.transports[i].skip_floors.clone().unwrap_or_default();
455            if next != before {
456                self.transports[i].skip_floors = Some(next);
457                changed = true;
458            }
459        }
460        if changed {
461            self.revision += 1;
462        }
463    }
