1//! The catalog table: one `fac!` row per facility, in `Kind` order.
2
3use super::{Category, Facility, Kind};
4
5macro_rules! fac {
6    ($kind:ident, $key:literal, $cat:ident, $name:literal, $w:literal, $floors:expr, $cost:literal, $star:literal, $pop:literal, $att:expr, mo=$mo:literal, t=$t:literal, so=$so:literal, b=$b:literal) => {
7        Facility {
8            kind: Kind::$kind,
9            key: $key,
10            category: Category::$cat,
11            name: $name,
12            width: $w,
13            floors: $floors,
14            cost: $cost as f64,
15            min_star: $star,
16            population: $pop,
17            attendance: $att,
18            modern_only: $mo,
19            transport: $t,
20            staff_only: $so,
21            basement: $b,
22        }
23    };
24}

26mod rest;

28const fn join<const N: usize>(head: &[Facility], tail: &[Facility]) -> [Facility; N] {
29    assert!(head.len() + tail.len() == N);
30    let mut out = [head[0]; N];
31    let mut i = 0;
32    while i < head.len() {
33        out[i] = head[i];
34        i += 1;
35    }
36    let mut j = 0;
37    while j < tail.len() {
38        out[head.len() + j] = tail[j];
39        j += 1;
40    }
41    out
42}
