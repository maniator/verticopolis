1//! Port of `crowd/segmentGraph.ts`: reachability probes.
2//!
3//! `segReachable` asks whether a passenger route runs from any lobby
4//! segment to a target, and the TypeScript answers with one route search
5//! per lobby segment for every room it checks (#849). The demand, income
6//! and move-in passes ask it for each room on a split floor every hour, so
7//! here the answer comes from a set of every segment reachable from a
8//! lobby, built once per tower revision (#1143). The routing graph is
9//! memoized on the same key, so the set answers exactly as fresh searches
10//! over that graph would.
11

19/// `segReachable`: whether `passengerPath` finds a route to `target` from
20/// any lobby segment. A route search succeeds exactly when its target is in
21/// the closure of its origin under the router's moves, so the union of
22/// those closures over every lobby segment answers for all of them. In
23/// Modern that closure is the plain graph's, which `connected_set` already
24/// holds; the Classic router walks a state graph with a walkway budget, so
25/// its closure has its own memo.
26fn seg_reachable(crowd: &mut Crowd, tower: &Tower, target: i64) -> bool {
27    if tower.mode.walkway_willingness_applies() {
28        walk_budget_reach(crowd, tower).contains(&target)
29    } else {
30        connected_set(crowd, tower).contains(&target)
31    }
32}
33
34/// Every segment the walk-budget router (`bfs_route_walk_budget`) reaches
35/// from some lobby segment: one search from all of them at once, with the
36/// router's own step, so a state is reached here exactly when one of the
37/// single-origin searches would reach it.
38fn walk_budget_reach<'a>(crowd: &'a mut Crowd, tower: &Tower) -> &'a HashSet<i64> {
39    let fresh = !matches!(&crowd.lobby_reach, Some((rev, _)) if *rev == tower.revision);
40    if fresh {
41        let origins = tower.lobby_segs();
42        let reach = closure_under_walk_budget(adjacency(crowd, tower), &origins);
43        crowd.lobby_reach = Some((tower.revision, reach));
44    }
45    &crowd.lobby_reach.as_ref().unwrap().1
46}

48fn closure_under_walk_budget(adj: &AdjGraph, origins: &[i64]) -> HashSet<i64> {
49    let mut seen: HashSet<WalkState> = HashSet::new();
50    let mut segs: HashSet<i64> = HashSet::new();
51    let mut frontier: Vec<WalkState> = Vec::new();
52    for &s in origins {
53        let origin: WalkState = (s, 0, None);
54        if seen.insert(origin) {
55            segs.insert(s);
56            frontier.push(origin);
57        }
58    }
59    while let Some(s) = frontier.pop() {
60        let Some(edges) = adj.get(&s.0) else { continue };
61        for edge in edges {
62            let Some(ns) = walk_budget_step(s, edge) else {
63                continue;
64            };
65            if seen.insert(ns) {
66                segs.insert(ns.0);
67                frontier.push(ns);
68            }
69        }
70    }
71    segs
72}

74pub fn position_reachable(crowd: &mut Crowd, tower: &Tower, floor: i64, x: i64) -> bool {
75    let s = tower.seg_at(floor, Some(x as f64));
76    seg_reachable(crowd, tower, s)
77}

79pub fn floor_reachable_from_lobby(crowd: &mut Crowd, tower: &Tower, floor: i64) -> bool {
80    if floor == 1 {
81        return true;
82    }
83    for (start, _) in tower.segments_of(floor) {
84        if seg_reachable(crowd, tower, seg_id(floor, start)) {
85            return true;
86        }
87    }
88    false
89}

91fn connected_set<'a>(crowd: &'a mut Crowd, tower: &Tower) -> &'a HashSet<i64> {
92    let fresh = !matches!(&crowd.seg_served, Some((rev, _)) if *rev == tower.revision);
93    if fresh {
94        let mut seen: HashSet<i64> = HashSet::new();
95        let mut frontier: Vec<i64> = Vec::new();
96        for s1 in tower.lobby_segs() {
97            if seen.insert(s1) {
98                frontier.push(s1);
99            }
100        }
101        {
102            let adj = adjacency(crowd, tower);
103            while let Some(s) = frontier.pop() {
104                if let Some(edges) = adj.get(&s) {
105                    for e in edges {
106                        if seen.insert(e.f) {
107                            frontier.push(e.f);
108                        }
109                    }
110                }
111            }
112        }
113        crowd.seg_served = Some((tower.revision, seen));
114    }
115    &crowd.seg_served.as_ref().unwrap().1
116}

118pub fn segment_connected(crowd: &mut Crowd, tower: &Tower, floor: i64, x: i64) -> bool {
119    let s = tower.seg_at(floor, Some(x as f64));
120    connected_set(crowd, tower).contains(&s)
121}

132    /// `segReachable` as it was before the memo: one route search per
133    /// lobby segment.
134    fn searched(crowd: &mut Crowd, tower: &Tower, target: i64) -> bool {
135        for s1 in tower.lobby_segs() {
136            if passenger_path(crowd, tower, s1, target).is_some() {
137                return true;
138            }
139        }
140        false
141    }

143    /// The segment ids to check: every run on every floor, the ids a
144    /// position off every run carries (one per tile on a floor with a run,
145    /// or with `sample` only those past each lot edge and next to each run;
146    /// past the lot edges alone on an empty floor), the off-run sentinel,
147    /// and every node of the routing graph (a landing over no tile carries
148    /// an id no run starts at). Sorted, so the order does not depend on the
149    /// graph's hash map.
150    fn every_segment(crowd: &mut Crowd, tower: &Tower, sample: bool) -> Vec<i64> {
151        let mut segs = vec![NO_SEGMENT];
152        for floor in MIN_FLOOR - 1..=MAX_FLOOR + 1 {
153            let runs = tower.segments_of(floor);
154            segs.extend(runs.iter().map(|r| seg_id(floor, r.0)));
155            let xs: Vec<i64> = if sample || runs.is_empty() {
156                let edges = runs.iter().flat_map(|r| [r.0 - 1, r.1 + 1]);
157                [-1, 0, LOT_WIDTH].into_iter().chain(edges).collect()
158            } else {
159                (-1..=LOT_WIDTH).collect()
160            };
161            segs.extend(xs.into_iter().map(|x| tower.seg_at(floor, Some(x as f64))));
162        }
163        for (&from, edges) in adjacency(crowd, tower) {
164            segs.push(from);
165            segs.extend(edges.iter().map(|e| e.f));
166        }
167        segs.sort();
168        segs.dedup();
169        segs
170    }

172    /// The memo answers every segment as the per-lobby searches do, and
173    /// returns the targets it found reachable.
174    fn assert_matches_searches(sim: &mut Simulation) -> HashSet<i64> {
175        check_segments(sim, false)
176    }
177
178    fn check_segments(sim: &mut Simulation, sample: bool) -> HashSet<i64> {
179        let Simulation {
180            ref mut crowd,
181            ref tower,
182            ..
183        } = *sim;
184        let mut reached = HashSet::new();
185        for target in every_segment(crowd, tower, sample) {
186            let memo = seg_reachable(crowd, tower, target);
187            assert_eq!(memo, searched(crowd, tower, target), "segment {target}");
188            if memo {
189                reached.insert(target);
190            }
191        }
192        // The answers came from this revision's memo for the mode, and
193        // Modern never builds the walk-budget set.
194        let current = |memo: &Option<(i64, HashSet<i64>)>| matches!(memo, Some((rev, _)) if *rev == tower.revision);
195        if tower.mode.walkway_willingness_applies() {
196            assert!(current(&crowd.lobby_reach));
197        } else {
198            assert!(current(&crowd.seg_served) && crowd.lobby_reach.is_none());
199        }
200        reached
201    }

203    /// A lobby on floor 1 and floor tiles on floors 2 to `top`, all at x 100
204    /// to 199.
205    fn bare_tower(mode: GameMode, top: i64) -> Simulation {
206        let mut sim = Simulation::new_game(5, mode);
207        sim.money = 1e12;
208        for x in 100..200 {
209            assert!(sim.tower.place(Kind::Lobby, 1, x).ok);
210        }
211        for floor in 2..=top {
212            for x in 100..200 {
213                assert!(sim.tower.place(Kind::Floor, floor, x).ok);
214            }
215        }
216        sim
217    }

219    /// `bare_tower` with one flight of each given kind per floor, climbing
220    /// from the lobby. The floor above the last flight has no way in.
221    fn walkway_tower(mode: GameMode, flights: &[Kind]) -> Simulation {
222        let mut sim = bare_tower(mode, flights.len() as i64 + 2);
223        for (i, &kind) in flights.iter().enumerate() {
224            let (x, bottom) = (100 + 12 * i as i64, 1 + i as i64);
225            assert!(sim.tower.place_transport(kind, x, bottom, bottom + 1).ok);
226        }
227        sim
228    }

303    #[test]
304    fn every_lobby_run_is_an_origin_with_its_own_walk() {
305        // Two lobby runs, x 100 to 139 and 160 to 199, with floors 2 to 4
306        // split the same way and floors 5 to 9 whole. One run climbs four
307        // flights of stairs to floor 5; the other takes an elevator to floor
308        // 4 and one flight to floor 5. Stairs from floor 5 climb to 9.
309        // Arriving with one flight walked, the elevator side's walkers get to
310        // floor 8; the stair side's, with four, get no higher than floor 5.
311        // The mirrored layout swaps the sides, so either run can be the one
312        // a search reaches floor 5 from first.
313        for mode in [GameMode::Classic, GameMode::Modern] {
314            for mirrored in [false, true] {
315                // The x of something `width` wide, flipped about the middle
316                // of x 100 to 199 in the mirrored layout.
317                let at = |x: i64, width: i64| if mirrored { 300 - x - width } else { x };
318                let mut sim = bare_tower(mode, 9);
319                for floor in 1..=4 {
320                    for x in 140..160 {
321                        let tile = sim.tower.unit_at(floor, x).map(|u| u.id).unwrap();
322                        assert!(sim.tower.remove_unit(tile).is_some());
323                    }
324                }
325                let stairs = [(100, 1), (110, 2), (120, 3), (130, 4), (160, 4)];
326                let more = [(140, 5), (152, 6), (164, 7), (176, 8)];
327                for (x, bottom) in stairs.into_iter().chain(more) {
328                    let x = at(x, 8);
329                    let placed = sim
330                        .tower
331                        .place_transport(Kind::Stairs, x, bottom, bottom + 1);
332                    assert!(placed.ok, "stairs at {x}, floor {bottom}");
333                }
334                let lift = sim
335                    .tower
336                    .place_transport(Kind::ElevatorStandard, at(184, 4), 1, 4);
337                assert!(lift.ok);
338                assert_eq!(sim.tower.lobby_segs().len(), 2);
339                let reached = assert_matches_searches(&mut sim);
340                let stair_side = floor_seg(&sim, 1, at(120, 1));
341                let lift_side = floor_seg(&sim, 1, at(180, 1));
342                let (top, roof) = (floor_seg(&sim, 8, 150), floor_seg(&sim, 9, 150));
343                if mode.walkway_willingness_applies() {
344                    assert!(reached.contains(&top) && !reached.contains(&roof));
345                    // Floor 8 is in reach from the elevator side only.
346                    let Simulation {
347                        ref mut crowd,
348                        ref tower,
349                        ..
350                    } = sim;
351                    assert!(passenger_path(crowd, tower, stair_side, top).is_none());
352                    assert!(passenger_path(crowd, tower, lift_side, top).is_some());
353                } else {
354                    assert!(reached.contains(&roof));
355                }
356            }
357        }
358    }

360    #[test]
361    fn the_memo_follows_each_layout_edit() {
362        for mode in [GameMode::Classic, GameMode::Modern] {
363            let mut sim = stair_tower(mode);
364            assert_matches_searches(&mut sim);
365            // An elevator resets the walk run: floors 6 and 7 come in reach.
366            let lift = sim
367                .tower
368                .place_transport(Kind::ElevatorStandard, 170, 1, 7)
369                .unit_id
370                .unwrap();
371            let reached = assert_matches_searches(&mut sim);
372            assert!(reached.contains(&floor_seg(&sim, 6, 150)));
373            assert!(reached.contains(&floor_seg(&sim, 7, 150)));
374            // A tile pulled from floor 3 splits it: the stair landing keeps
375            // the left part, the elevator the right.
376            let gap = sim.tower.unit_at(3, 160).map(|u| u.id).unwrap();
377            assert!(sim.tower.remove_unit(gap).is_some());
378            let reached = assert_matches_searches(&mut sim);
379            assert!(reached.contains(&floor_seg(&sim, 3, 110)));
380            assert!(reached.contains(&floor_seg(&sim, 3, 170)));
381            // An elevator that skips floor 3 leaves its right part with no
382            // way in, until it stops there again.
383            assert!(sim.tower.set_stop(lift, 3, false));
384            let reached = assert_matches_searches(&mut sim);
385            assert!(!reached.contains(&floor_seg(&sim, 3, 170)));
386            assert!(sim.tower.set_stop(lift, 3, true));
387            let reached = assert_matches_searches(&mut sim);
388            assert!(reached.contains(&floor_seg(&sim, 3, 170)));
389            // Cut back to floor 5, the elevator leaves floor 7 with no way
390            // in, while floor 6 is one flight above it.
391            assert!(sim.tower.resize_transport(lift, 1, 5).ok);
392            let reached = assert_matches_searches(&mut sim);
393            assert!(reached.contains(&floor_seg(&sim, 6, 150)));
394            assert!(!reached.contains(&floor_seg(&sim, 7, 150)));
395            // Without the elevator the right part of floor 3 has no way in.
396            assert!(sim.tower.remove_transport(lift).is_some());
397            let reached = assert_matches_searches(&mut sim);
398            assert!(!reached.contains(&floor_seg(&sim, 3, 170)), "{mode:?}");
399            // A lobby split leaves the stairs on the left-hand run, and the
400            // right-hand run, with no transport, an origin of its own.
401            let lobby_gap = sim.tower.unit_at(1, 140).map(|u| u.id).unwrap();
402            assert!(sim.tower.remove_unit(lobby_gap).is_some());
403            assert_eq!(sim.tower.lobby_segs().len(), 2);
404            let reached = assert_matches_searches(&mut sim);
405            assert!(reached.contains(&floor_seg(&sim, 1, 170)));
406            assert!(reached.contains(&floor_seg(&sim, 2, 150)));
407            // Filling the gap joins the runs again.
408            assert!(sim.tower.place(Kind::Lobby, 1, 140).ok);
409            assert_eq!(sim.tower.lobby_segs().len(), 1);
410            let reached = assert_matches_searches(&mut sim);
411            assert!(reached.contains(&floor_seg(&sim, 1, 170)));
412        }
413    }

427    fn the_fixture_towers_answer_as_the_searches_do() {
428        let dir = concat!(env!("CARGO_MANIFEST_DIR"), "/../conformance/fixtures/");
429        let mut modes = Vec::new();
430        for name in ["split-tower-4", "split-tower", "sixseven_2", "towerone_6"] {
431            let path = format!("{dir}{name}.vctower");
432            let text = std::fs::read_to_string(&path).unwrap_or_else(|e| panic!("{path}: {e}"));
433            let raw = crate::load::decode_vctower(&text).expect("decodes");

436            // searches walk the whole graph for each unreachable one, so the
437            // fixtures check a sample of those; the small towers check every
438            // tile of every floor with a run.
439            let reached = check_segments(&mut sim, true);
440            let upstairs = reached.iter().filter(|&&s| floor_of_seg(s) > 1).count();
441            assert!(upstairs > 10, "{name}: {upstairs} segments above the lobby");
442            modes.push(sim.tower.mode);
443        }
444        assert!(modes.contains(&GameMode::Classic) && modes.contains(&GameMode::Modern));
