12/// The graph over the transports' landing segments, and those segments in
13/// the order they were first listed, for a walk that must not follow the
14/// map's order.
15fn build_adjacency<'a>(
16    tower: &Tower,
17    transports: impl Iterator<Item = &'a Transport>,
18) -> (AdjGraph, Vec<i64>) {
19    let mut adj: AdjGraph = HashMap::new();
20    let mut listed: Vec<i64> = Vec::new();
21    for t in transports {
22        let stops = tower.stops_of(t);
23        let walk_kind = if t.kind.is_walkway() {
24            Some(t.kind)
25        } else {
26            None
27        };
28        let seg_lists: Vec<Vec<i64>> = stops.iter().map(|&fl| tower.landing_segs(t, fl)).collect();
29        for ai in 0..stops.len() {
30            for &a_seg in &seg_lists[ai] {
31                let list = adj.entry(a_seg).or_insert_with(|| {
32                    listed.push(a_seg);
33                    Vec::new()
34                });
35                for bi in 0..stops.len() {
36                    if bi == ai {
37                        continue;
38                    }
39                    for &b_seg in &seg_lists[bi] {
40                        list.push(Edge {
41                            f: b_seg,
42                            shaft: t.id,
43                            walk_kind,
44                        });
45                    }
46                }
47            }
48        }
49    }
50    (adj, listed)
51}

53pub fn adjacency<'a>(crowd: &'a mut Crowd, tower: &Tower) -> &'a AdjGraph {
54    let fresh = !matches!(&crowd.adj, Some((rev, _)) if *rev == tower.revision);
55    if fresh {
56        let (adj, _) = build_adjacency(
57            tower,
58            tower
59                .transports
60                .iter()
61                .filter(|t| !t.kind.is_staff_only_transport()),
62        );
63        crowd.adj = Some((tower.revision, adj));
64    }
65    &crowd.adj.as_ref().unwrap().1
66}

68/// The staff graph, and for each segment on it the part of the graph it lies
69/// in (see `graph_parts`).
70pub fn staff_adjacency<'a>(crowd: &'a mut Crowd, tower: &Tower) -> (&'a AdjGraph, &'a StaffParts) {
71    let fresh = !matches!(&crowd.staff_adj, Some((rev, _, _)) if *rev == tower.revision);
72    if fresh {
73        let mut service_first: Vec<&Transport> = tower.transports.iter().collect();
74        service_first.sort_by_key(|t| !t.kind.is_staff_only_transport());
75        let (adj, listed) = build_adjacency(
76            tower,
77            service_first
78                .into_iter()
79                .filter(|t| t.kind.is_staff_transport()),
80        );
81        let parts = graph_parts(&adj, &listed);
82        crowd.staff_adj = Some((tower.revision, adj, parts));
83    }
84    let (_, adj, parts) = crowd.staff_adj.as_ref().unwrap();
85    (adj, parts)
86}

88/// Maps each segment that has an edge to a label for its part of the graph:
89/// the segments joined by edges taken in either direction, so a part holds
90/// every segment a search from any of its members could reach. A part is
91/// labeled with its first segment in `listed` order, the order the graph was
92/// built in, so the labels never follow a map's order; only their equality
93/// is read. `listed` holds every key of `adj`, as `build_adjacency` returns it.
94fn graph_parts(adj: &AdjGraph, listed: &[i64]) -> StaffParts {
95    let mut near: FastMap<i64, Vec<i64>> = FastMap::default();
96    for &a in listed {
97        for e in &adj[&a] {
98            near.get_or_insert_with(a, Vec::new).push(e.f);
99            near.get_or_insert_with(e.f, Vec::new).push(a);
100        }
101    }
102    let mut part: StaffParts = FastMap::default();
103    for &seg in listed {
104        if part.get(&seg).is_some() || near.get(&seg).is_none() {
105            continue;
106        }
107        part.insert(seg, seg);
108        let mut stack = vec![seg];
109        while let Some(at) = stack.pop() {
110            for &next in &near[&at] {
111                if part.get(&next).is_none() {
112                    part.insert(next, seg);
113                    stack.push(next);
114                }
115            }
116        }
117    }
118    part
119}

121/// Whether `a` and `b` differ and lie in different parts of the staff graph
122/// (or either lies in none). When it holds no route joins them, and on the
123/// tower's graphs, whose edges come in pairs, it holds whenever none does.
124fn parts_split(parts: &StaffParts, a: i64, b: i64) -> bool {
125    let pa = parts.get(&a);
126    a != b && (pa.is_none() || pa != parts.get(&b))
127}

129/// `passengerPath`: the per-mode router on segment ids.
130pub fn passenger_path(
131    crowd: &mut Crowd,
132    tower: &Tower,
133    from_seg: i64,
134    to_seg: i64,
135) -> Option<Route> {
136    let walk_budget = tower.mode.walkway_willingness_applies();
137    let adj = adjacency(crowd, tower);
138    if walk_budget {
139        bfs_route_walk_budget(adj, from_seg, to_seg)
140    } else {
141        bfs_route(adj, from_seg, to_seg)
142    }
143}

151/// The JavaScript gives a position that sits over a structural gap its own
152/// fractional segment id, which no graph node carries; here both such
153/// positions map to one sentinel (`NO_SEGMENT`). The ids are equal in
154/// JavaScript only when the two positions are the same, so when both ends
155/// are off-run, decide the route the way the fractional ids would: the same
156/// position routes to itself with no shafts, a different one has no route.
157/// `None` means the ids are not both off-run and the graph decides.
158fn off_run_pair(
159    a: i64,
160    b: i64,
161    from_floor: i64,
162    from_x: Option<f64>,
163    to_floor: i64,
164    to_x: Option<f64>,
165) -> Option<Option<Route>> {
166    if a != NO_SEGMENT || b != NO_SEGMENT {
167        return None;
168    }
169    // The fractional ids themselves decide, as `from === to` does: a position
170    // past one floor's lot edge carries the same id as one past the neighbor's.
171    let id = |floor: i64, x: Option<f64>| {
172        (floor - MIN_FLOOR) as f64 * LOT_WIDTH as f64 + x.unwrap_or(0.0)
173    };
174    let (a, b) = (id(from_floor, from_x), id(to_floor, to_x));
175    // `floorOfSeg` of that id.
176    let floor = (a / LOT_WIDTH as f64).floor() as i64 + MIN_FLOOR;
177    Some((a == b).then(|| Route {
178        floors: vec![floor],
179        shafts: vec![],
180    }))
181}

183pub fn route(
184    crowd: &mut Crowd,
185    tower: &Tower,
186    from_floor: i64,
187    from_x: Option<f64>,
188    to_floor: i64,
189    to_x: Option<f64>,
190) -> Option<Route> {
191    let a = tower.seg_at(from_floor, from_x);
192    let b = tower.seg_at(to_floor, to_x);
193    if let Some(decided) = off_run_pair(a, b, from_floor, from_x, to_floor, to_x) {
194        return decided.map(|r| balance_shafts(crowd, tower, r));
195    }
196    let mut r = passenger_path(crowd, tower, a, b)?;
197    to_real_floors(&mut r);
198    Some(balance_shafts(crowd, tower, r))
199}

201pub fn reachable(
202    crowd: &mut Crowd,
203    tower: &Tower,
204    from_floor: i64,
205    from_x: Option<f64>,
206    to_floor: i64,
207    to_x: Option<f64>,
208) -> bool {
209    let a = tower.seg_at(from_floor, from_x);
210    let b = tower.seg_at(to_floor, to_x);
211    if let Some(decided) = off_run_pair(a, b, from_floor, from_x, to_floor, to_x) {
212        return decided.is_some();
213    }
214    passenger_path(crowd, tower, a, b).is_some()
215}

217pub fn staff_route(
218    crowd: &mut Crowd,
219    tower: &Tower,
220    from_floor: i64,
221    from_x: Option<f64>,
222    to_floor: i64,
223    to_x: Option<f64>,
224) -> Option<Route> {
225    let a = tower.seg_at(from_floor, from_x);
226    let b = tower.seg_at(to_floor, to_x);
227    if let Some(decided) = off_run_pair(a, b, from_floor, from_x, to_floor, to_x) {
228        return decided.map(|r| balance_shafts(crowd, tower, r));
229    }
230    let (adj, parts) = staff_adjacency(crowd, tower);
231    // A search between two parts of the staff graph visits all of the
232    // first one before it gives up, and dispatch keeps asking for such
233    // routes: a floor-level staff link can leave a room's segment off the
234    // crew's network (#1143 item 6). The parts answer that without a search.
235    if parts_split(parts, a, b) {
236        return None;
237    }
238    let mut r = bfs_route(adj, a, b)?;
239    to_real_floors(&mut r);
240    Some(balance_shafts(crowd, tower, r))
241}

243/// Fewest-edges BFS; the first-listed edge wins a level tie.
244pub fn bfs_route(adj: &AdjGraph, from: i64, to: i64) -> Option<Route> {
245    if from == to {
246        return Some(Route {
247            floors: vec![from],
248            shafts: vec![],
249        });
250    }
251    let mut prev: FastMap<i64, (i64, i64)> = FastMap::default();
252    let mut seen: FastSet<i64> = FastSet::default();
253    seen.insert(from);
254    let mut frontier = vec![from];
255    while !frontier.is_empty() {
256        let mut next = Vec::new();
257        for &f in &frontier {
258            let Some(edges) = adj.get(&f) else { continue };
259            for edge in edges {
260                if !seen.insert(edge.f) {
261                    continue;
262                }
263                prev.insert(edge.f, (f, edge.shaft));
264                if edge.f == to {
265                    let mut floors = vec![to];
266                    let mut shafts = Vec::new();
267                    let mut cur = to;
268                    while cur != from {
269                        let (pf, ps) = prev[&cur];
270                        floors.push(pf);
271                        shafts.push(ps);
272                        cur = pf;
273                    }
274                    floors.reverse();
275                    shafts.reverse();
276                    return Some(Route { floors, shafts });
277                }
278                next.push(edge.f);
279            }
280        }
281        frontier = next;
282    }
283    None
284}

286/// A state of the Classic walk-budget search: (floor, walkRun, runCap);
287/// `None` is the NO_CAP sentinel (Infinity).
288pub(super) type WalkState = (i64, i64, Option<i64>);
289
290/// One edge of the walk-budget search: the state it leads to, or `None`
291/// when the edge is a walkway the budget refuses.
292#[inline]
293pub(super) fn walk_budget_step(s: WalkState, edge: &Edge) -> Option<WalkState> {
294    let Some(wk) = edge.walk_kind else {
295        return Some((edge.f, 0, None));
296    };
297    let w = wk.walkway_willingness()?;
298    let cap = match s.2 {
299        Some(c) => c.min(w),
300        None => w,
301    };
302    if s.1 + 1 > cap {
303        return None;
304    }
305    Some((edge.f, s.1 + 1, Some(cap)))
306}

308/// The Classic router with the contiguous-walk budget.
309pub fn bfs_route_walk_budget(adj: &AdjGraph, from: i64, to: i64) -> Option<Route> {
310    if from == to {
311        return Some(Route {
312            floors: vec![from],
313            shafts: vec![],
314        });
315    }
316    type St = WalkState;
317    let origin: St = (from, 0, None);
318    let mut prev: FastMap<St, (St, i64, i64)> = FastMap::default();
319    let mut seen: FastSet<St> = FastSet::default();
320    seen.insert(origin);
321    let mut frontier: Vec<St> = vec![origin];
322    while !frontier.is_empty() {
323        let mut next = Vec::new();
324        for &s in &frontier {
325            let Some(edges) = adj.get(&s.0) else { continue };
326            for edge in edges {
327                let Some(ns) = walk_budget_step(s, edge) else {
328                    continue;
329                };
330                if !seen.insert(ns) {
331                    continue;
332                }
333                prev.insert(ns, (s, s.0, edge.shaft));
334                if edge.f == to {
335                    let mut floors = vec![to];
336                    let mut shafts = Vec::new();
337                    let mut cur = ns;
338                    while cur != origin {
339                        let (pk, pf, ps) = prev[&cur];
340                        floors.push(pf);
341                        shafts.push(ps);
342                        cur = pk;
343                    }
344                    floors.reverse();
345                    shafts.reverse();
346                    return Some(Route { floors, shafts });
347                }
348                next.push(ns);
349            }
350        }
351        frontier = next;
352    }
353    None
354}

356/// `elevatorCalls`.
357pub fn elevator_calls(crowd: &Crowd, tower: &Tower) -> ElevatorCalls {
358    let mut calls = ElevatorCalls::default();
359    let mut bump = |shaft: i64, floor: i64| {
360        *calls
361            .hall
362            .entry(shaft)
363            .or_default()
364            .entry(floor)
365            .or_insert(0.0) += 1.0;
366    };
367    for p in &crowd.people {
368        let Some(sid) = p.shaft_id else { continue };
369        if p.state == PState::Waiting {
370            bump(sid, p.floor);
371        } else if p.state == PState::Riding && p.car_index.is_some() {
372            let Some(&dest) = p.floors.get(p.leg + 1) else {
373                continue;
374            };
375            calls
376                .cab
377                .entry(sid)
378                .or_default()
379                .entry(p.car_index.unwrap())
380                .or_default()
381                .insert(dest);
382        } else if p.staff && p.state == PState::ToShaft {
383            if let Some(shaft) = tower.get_transport(sid) {
384                if shaft.kind.is_staff_only_transport() {
385                    bump(sid, p.floor);
386                }
387            }
388        }
389    }
390    calls
391}

400    use crate::clock::GameMode;

404    /// Two positions over structural gaps carry fractional ids in the
405    /// JavaScript: equal only for the same position, which routes to itself
406    /// with no shafts; a different one has no route and is not reachable.
407    #[test]
408    fn off_run_positions_route_as_their_fractional_ids_would() {
409        let mut sim = Simulation::new_game(3, GameMode::Classic);
410        sim.money = 1e9;
411        for x in 170..200 {
412            assert!(sim.build(Kind::Lobby, 1, x).ok);
413        }
414        let Simulation {
415            ref mut crowd,
416            ref tower,
417            ..
418        } = sim;
419        // Over the gap right of the lobby run, off the integer grid.
420        let same = route(crowd, tower, 1, Some(210.5), 1, Some(210.5)).expect("itself");
421        assert_eq!(same.floors, vec![1]);
422        assert!(same.shafts.is_empty());
423        assert!(route(crowd, tower, 1, Some(210.5), 1, Some(220.5)).is_none());
424        assert!(route(crowd, tower, 1, Some(210.5), 2, Some(210.5)).is_none());
425        assert!(reachable(crowd, tower, 1, Some(210.5), 1, Some(210.5)));
426        assert!(!reachable(crowd, tower, 1, Some(210.5), 1, Some(220.5)));
427        // An off-run start and an on-run end: no route, as the graph has no
428        // node for the start.
429        assert!(route(crowd, tower, 1, Some(210.5), 1, Some(180.0)).is_none());
430        // Past the lot's left edge the fractional id belongs to the floor
431        // below, as `floorOfSeg` reads it.
432        let edge = route(crowd, tower, 1, Some(-0.5), 1, Some(-0.5)).expect("itself");
433        assert_eq!(edge.floors, vec![0]);
434        // The same id spelled from the floor below: equal ids, one route.
435        let twin = route(crowd, tower, 1, Some(-0.5), 0, Some(374.5)).expect("same id");
436        assert_eq!(twin.floors, vec![0]);
437    }
