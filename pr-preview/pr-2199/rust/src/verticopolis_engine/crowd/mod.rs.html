47/// `dwellSecondsRange`.
48pub fn dwell_seconds_range(kind: Option<Kind>) -> (i64, i64) {
49    let m = match kind {
50        Some(Kind::Cinema) => Some((90, 120)),
51        Some(Kind::PartyHall) => Some((60, 120)),
52        Some(Kind::WeddingHall) => Some((120, 180)),
53        Some(Kind::AquaticCenter) => Some((60, 120)),
54        _ => None,
55    };
56    match m {
57        None => (EAT_SECONDS_MIN, EAT_SECONDS_MAX),
58        Some((a, b)) => (a * 2, b * 2),
59    }
60}

147    /// The `Person` shape with only the fields that are set.
148    pub fn to_json(&self) -> Value {
149        let mut m = Map::new();
150        m.insert("id".into(), json!(self.id));
151        m.insert("seed".into(), json!(self.seed));
152        m.insert("state".into(), json!(self.state.as_str()));
153        m.insert("floor".into(), json!(self.floor));
154        m.insert("fy".into(), json!(self.fy));
155        m.insert("x".into(), json!(self.x));
156        m.insert("floors".into(), json!(self.floors));
157        m.insert("originFloor".into(), json!(self.origin_floor));
158        m.insert("shafts".into(), json!(self.shafts));
159        m.insert("leg".into(), json!(self.leg));
160        // `shaftId` and `carIndex` are `number | null` in the TypeScript, so
161        // an absent value is a literal null; every other optional is omitted.
162        m.insert("shaftId".into(), json!(self.shaft_id));
163        m.insert("carIndex".into(), json!(self.car_index));
164        m.insert("destX".into(), json!(self.dest_x));
165        m.insert("wait".into(), json!(self.wait));
166        m.insert("tripWait".into(), json!(self.trip_wait));
167        m.insert("age".into(), json!(self.age));
168        m.insert("linger".into(), json!(self.linger));
169        if self.staff {
170            m.insert("staff".into(), json!(true));
171        }
172        if let Some(v) = self.clean_unit_id {
173            m.insert("cleanUnitId".into(), json!(v));
174        }
175        if let Some(v) = self.meal_venue_id {
176            m.insert("mealVenueId".into(), json!(v));
177        }
178        if let Some(v) = self.venue_unit_id {
179            m.insert("venueUnitId".into(), json!(v));
180        }
181        if self.counted_hotel_guest {
182            m.insert("countedHotelGuest".into(), json!(true));
183        }
184        if let Some(v) = self.origin_unit_id {
185            m.insert("originUnitId".into(), json!(v));
186        }
187        if let Some(v) = self.dwell_seconds_left {
188            m.insert("dwellSecondsLeft".into(), json!(v));
189        }
190        if self.returning {
191            m.insert("returning".into(), json!(true));
192        }
193        if let Some(v) = self.routine {
194            m.insert("routine".into(), json!(v));
195        }
196        if let Some(v) = self.linger_for {
197            m.insert("lingerFor".into(), json!(v));
198        }
199        Value::Object(m)
200    }

357    pub fn record_commute(&mut self, origin_floor: i64, wait_seconds: f64) {
358        self.commute_revision = self.commute_revision.wrapping_add(1);
359        let prev = self
360            .commute_wait_by_floor
361            .get(&origin_floor)
362            .copied()
363            .unwrap_or(wait_seconds);
364        self.commute_wait_by_floor.insert(
365            origin_floor,
366            prev + (wait_seconds - prev) * COMMUTE_STRESS_ALPHA,
367        );
368    }

370    /// Replace the commute-stress averages with `[[floor, seconds], ...]`
371    /// rows, the shape [`Crowd::commute_stress_json`] writes: a host seeds
372    /// them at attach, since no save carries them (#1083). The revision is
373    /// left alone. Refuses rows that are not whole floors with finite
374    /// seconds, or that name a floor twice, and then changes nothing.
375    pub fn seed_commute_stress(&mut self, rows: &Value) -> Result<(), String> {
376        let bad =
377            || "commuteStress must be [[floor, seconds], ...] with each floor once".to_string();
378        let mut seeded = HashMap::new();
379        for row in rows.as_array().ok_or_else(bad)? {
380            let pair = row.as_array().filter(|p| p.len() == 2).ok_or_else(bad)?;
381            let floor = pair[0]
382                .as_f64()
383                .filter(|f| f.fract() == 0.0 && f.abs() < 9007199254740992.0)
384                .ok_or_else(bad)?;
385            let seconds = pair[1].as_f64().filter(|s| s.is_finite()).ok_or_else(bad)?;
386            if seeded.insert(floor as i64, seconds).is_some() {
387                return Err(bad());
388            }
389        }
390        self.commute_wait_by_floor = seeded;
391        Ok(())
392    }

394    /// The commute-stress averages as `[[floor, seconds], ...]`, by floor.
395    pub fn commute_stress_json(&self) -> Value {
396        let mut rows: Vec<(i64, f64)> = self
397            .commute_wait_by_floor
398            .iter()
399            .map(|(&f, &w)| (f, w))
400            .collect();
401        rows.sort_by_key(|r| r.0);
402        json!(rows)
403    }
