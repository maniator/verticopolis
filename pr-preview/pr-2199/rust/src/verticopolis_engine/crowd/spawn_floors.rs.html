1//! The floors and rooms a spawn step picks from: `spawnFloors` in
2//! `crowd/spawn.ts`, rebuilt on each spawn step that has room to spawn
3//! (#1143 item 4, A4 in #1158). The TypeScript also builds them when the
4//! crowd is full (T2 in #1170).
5//!
6//! Each list keeps the order in which the walk over the rooms first meets a
7//! floor, as the TypeScript's `Set`s do. The lists dedupe with one bit per
8//! floor, and the rooms of each floor come from a tower index built once
9//! per revision, so a build hashes nothing.
10

19/// A bit per floor slot, in two words: a `u128` shift is a library call on
20/// wasm32.
21#[derive(Default)]
22struct FloorBits([u64; 2]);
23
24impl FloorBits {
25    /// Sets slot `i`'s bit; whether it was clear.
26    #[inline]
27    fn insert(&mut self, i: usize) -> bool {
28        let (word, bit) = (i >> 6, 1u64 << (i & 63));
29        let new = self.0[word] & bit == 0;
30        self.0[word] |= bit;
31        new
32    }
33}
34
35/// Floors in the order first added, each once.
36#[derive(Default)]
37struct FloorList {
38    floors: Vec<i64>,
39    seen: FloorBits,
40}
41
42impl FloorList {
43    /// Adds `floor`; whether it was new.
44    #[inline]
45    fn insert(&mut self, floor: i64) -> bool {
46        let new = match floor_slot(floor) {
47            Some(i) => self.seen.insert(i),
48            None => !self.floors.contains(&floor),
49        };
50        if new {
51            self.floors.push(floor);
52        }
53        new
54    }
55}

57/// The floors with an open venue, per venue kind. A kind with none has no

61impl VenueFloors {

63        self.0.iter().find(|(k, _)| k == kind).map(|(_, f)| f)

67pub struct SpawnFloors {
68    pub leased_offices: Vec<i64>,
69    pub staffed_offices: Vec<i64>,
70    pub homes: Vec<i64>,
71    pub condo_floors: Vec<i64>,
72    pub household_floors: Vec<i64>,
73    pub hotel_floors: Vec<i64>,
74    pub staff_floors: Vec<(Kind, i64)>,
75    pub venues_by_kind: VenueFloors,
76    /// The mean of the open theaters' ticket price pick multipliers, or
77    /// `None` where every one charges Normal and the crowd draws them as it
78    /// always has (#1950).
79    pub cinema_picks: Option<f64>,
80    /// Where `cinema_picks` is set, the mean pick multiplier of each floor's
81    /// open theaters by floor slot (1 on a floor without one), so a visit
82    /// weighs a floor in O(1) without walking its rooms. Empty otherwise.
83    cinema_floor_picks: Vec<f64>,
84    units_by_floor: Arc<FloorRooms>,
85    /// Operational metro station ids.
86    pub metro_stations: Vec<i64>,
87}

89impl SpawnFloors {
90    /// The mean ticket price pick multiplier of the open theaters on cinema
91    /// floor `floor`, 1 where none charges other than Normal.
92    pub fn cinema_floor_pick(&self, floor: i64) -> f64 {
93        floor_slot(floor)
94            .and_then(|i| self.cinema_floor_picks.get(i))
95            .copied()
96            .unwrap_or(1.0)
97    }
98
99    /// Room unit ids on `floor`, in unit order.
100    pub fn units_on(&self, floor: i64) -> &[i64] {
101        self.units_by_floor.on(floor)
102    }
103}

105/// Adds `floor` to `kind`'s list in `lists`, which holds one list per kind.
106fn add_floor(lists: &mut Vec<(Kind, FloorList)>, kind: Kind, floor: i64) {
107    match lists.iter_mut().find(|(k, _)| *k == kind) {
108        Some((_, list)) => {
109            list.insert(floor);
110        }
111        None => {
112            let mut list = FloorList::default();
113            list.insert(floor);
114            lists.push((kind, list));
115        }
116    }
117}

119/// Adds `(kind, floor)` to the staff floors unless it is there already: a
120/// bit per floor for each kind, where the TypeScript keeps a `seenStaff`
121/// `Set`, and a search of the list for a floor past the slots.
122fn add_staff_floor(
123    floors: &mut Vec<(Kind, i64)>,
124    seen: &mut Vec<(Kind, FloorBits)>,
125    kind: Kind,
126    floor: i64,
127) {
128    let new = match floor_slot(floor) {
129        Some(i) => {
130            let at = match seen.iter().position(|(k, _)| *k == kind) {
131                Some(at) => at,
132                None => {
133                    seen.push((kind, FloorBits::default()));
134                    seen.len() - 1
135                }
136            };
137            seen[at].1.insert(i)
138        }
139        None => !floors.contains(&(kind, floor)),
140    };
141    if new {
142        floors.push((kind, floor));
143    }
144}

146pub fn spawn_floors(tower: &Tower, clock: &Clock) -> SpawnFloors {
147    let hour = clock.hour();
148    let weekend = clock.is_weekend();
149    let mut leased = FloorList::default();
150    let mut staffed = FloorList::default();
151    let mut homes = FloorList::default();
152    let mut metro_stations: Vec<i64> = Vec::new();
153    let mut condo_floors = FloorList::default();
154    let mut household_floors = FloorList::default();
155    let mut hotel_floors = FloorList::default();
156    let mut staff_floors: Vec<(Kind, i64)> = Vec::new();
157    let mut staff_seen: Vec<(Kind, FloorBits)> = Vec::new();
158    let mut venues_by_kind: Vec<(Kind, FloorList)> = Vec::new();
159    let (mut cinemas, mut cinema_pick_sum, mut cinema_priced) = (0_u32, 0.0, false);
160    // Each cinema floor's pick sum and open theater count.
161    let mut cinema_floors: Vec<(i64, f64, u32)> = Vec::new();
162    for u in tower.room_units() {
163        if u.kind.is_staff_kind() {
164            if u.is_operational() {
165                add_staff_floor(&mut staff_floors, &mut staff_seen, u.kind, u.floor);
166            }
167            continue;
168        }
169        if u.kind == Kind::WeddingHall {
170            if u.is_operational() {
171                add_floor(&mut venues_by_kind, u.kind, u.floor);
172            }
173            continue;
174        }
175        if u.kind == Kind::Metro {
176            if u.is_operational() {
177                metro_stations.push(u.id);
178            }
179            continue;
180        }
181        if !(u.is_tenanted() || u.state == UnitState::Asleep) {
182            continue;
183        }
184        if u.kind == Kind::Office {
185            if !weekend {
186                leased.insert(u.floor);
187            }
188            if u.occupants > 0 {
189                staffed.insert(u.floor);
190            }
191        } else if u.kind == Kind::Condo || u.kind.is_rental() {
192            homes.insert(u.floor);
193            if u.kind != Kind::RentalStudio || u.visible_occupants() > 0 {
194                condo_floors.insert(u.floor);
195            }
196            if u.kind.has_household() {
197                household_floors.insert(u.floor);
198            }
199        } else if u.kind.is_hotel() {
200            homes.insert(u.floor);
201            hotel_floors.insert(u.floor);
202        } else if u.kind.is_ambient_venue() && u.kind.is_open_at(hour) {
203            add_floor(&mut venues_by_kind, u.kind, u.floor);
204        } else if u.kind.attendance_cap().is_some() && u.kind.is_open_at(hour) {
205            add_floor(&mut venues_by_kind, u.kind, u.floor);
206            if u.kind == Kind::Cinema {
207                let pick = ticket_pick_mult(tower.mode, u);
208                cinemas += 1;
209                cinema_pick_sum += pick;
210                cinema_priced |= charged(tower.mode, u).is_some();
211                match cinema_floors.iter_mut().find(|(f, _, _)| *f == u.floor) {
212                    Some((_, sum, n)) => {
213                        *sum += pick;
214                        *n += 1;
215                    }
216                    None => cinema_floors.push((u.floor, pick, 1)),
217                }
218            }
219        }
220    }
221    SpawnFloors {
222        leased_offices: leased.floors,
223        staffed_offices: staffed.floors,
224        homes: homes.floors,
225        condo_floors: condo_floors.floors,
226        household_floors: household_floors.floors,
227        hotel_floors: hotel_floors.floors,
228        staff_floors,
229        venues_by_kind: VenueFloors(
230            venues_by_kind
231                .into_iter()
232                .map(|(k, list)| (k, list.floors))
233                .collect(),
234        ),
235        cinema_picks: cinema_priced.then(|| cinema_pick_sum / f64::from(cinemas)),
236        cinema_floor_picks: if cinema_priced {
237            let mut picks = vec![1.0; FLOOR_SLOTS];
238            for (f, sum, n) in cinema_floors {
239                if let Some(i) = floor_slot(f) {
240                    picks[i] = sum / f64::from(n);
241                }
242            }
243            picks
244        } else {
245            Vec::new()
246        },
247        units_by_floor: tower.floor_rooms(),
248        metro_stations,
249    }
250}

355    /// The lists answer as before; which of the nine outputs had anything in.
356    fn assert_as_before(tower: &Tower, clock: &Clock) -> [bool; 9] {
357        let at = format!("day {} hour {}", clock.day(), clock.hour());
358        let now = spawn_floors(tower, clock);
359        let was = before(tower, clock);
360        let lists = [
361            &now.leased_offices,
362            &now.staffed_offices,
363            &now.homes,
364            &now.condo_floors,
365            &now.household_floors,
366            &now.hotel_floors,
367        ];
368        for (i, (now, was)) in lists.iter().zip(&was.lists).enumerate() {
369            assert_eq!(*now, was, "{at}: list {i}");
370        }
371        assert_eq!(now.staff_floors, was.staff_floors, "{at}");
372        assert_eq!(now.metro_stations, was.metro_stations, "{at}");
373        assert_eq!(now.venues_by_kind.0.len(), was.venues_by_kind.len(), "{at}");
374        for (kind, floors) in &was.venues_by_kind {
375            assert_eq!(now.venues_by_kind.get(kind), Some(floors), "{at}: {kind:?}");
376        }
377        for floor in MIN_FLOOR - 1..=MAX_FLOOR + 1 {
378            let was = was.units_by_floor.get(&floor).map_or(&[][..], |v| v);
379            assert_eq!(now.units_on(floor), was, "{at}: floor {floor}");
380        }
381        let mut filled = [false; 9];
382        for (i, list) in lists.iter().enumerate() {
383            filled[i] = !list.is_empty();
384        }
385        filled[6] = !now.staff_floors.is_empty();
386        filled[7] = !now.metro_stations.is_empty();
387        filled[8] = !now.venues_by_kind.0.is_empty();
388        filled
389    }

392    fn the_fixture_towers_list_the_floors_as_before() {
393        let dir = concat!(env!("CARGO_MANIFEST_DIR"), "/../conformance/fixtures/");
394        let mut filled = [false; 9];
395        let mut seen = |now: [bool; 9]| {
396            for (seen, now) in filled.iter_mut().zip(now) {
397                *seen |= now;
398            }
399        };
400        for name in [
401            "split-tower-4",
402            "sixseven_2",
403            "towerone_6",
404            "sixseven-december",
405        ] {
406            let path = format!("{dir}{name}.vctower");
407            let text = std::fs::read_to_string(&path).unwrap_or_else(|e| panic!("{path}: {e}"));
408            let raw = crate::load::decode_vctower(&text).expect("decodes");
409            let mut sim = crate::load::deserialize(&raw).expect("loads");
410            // Every hour of the tower's week, weekend included, on the
411            // rooms as loaded.
412            let calendar = sim.clock.calendar;
413            let week_start = (sim.clock.day() - sim.clock.day_of_week()) * 1440;
414            let mut weekend = [false; 2];
415            for hour in 0..calendar.week_days * 24 {
416                let clock = Clock::new((week_start + hour * 60) as f64, calendar);
417                weekend[clock.is_weekend() as usize] = true;
418                seen(assert_as_before(&sim.tower, &clock));
419            }
420            assert_eq!(weekend, [true, true], "{name}");
421            // And on the rooms as a day of play (12 ticks of 120 game
422            // minutes) changes them.
423            for _ in 0..12 {
424                sim.tick(120.0);
425                seen(assert_as_before(&sim.tower, &sim.clock));
426            }
427        }
428        // Every output had floors at some point, so none was compared only
429        // empty against empty.
430        assert_eq!(filled, [true; 9]);
431    }

433    #[test]
434    fn a_floor_past_the_slots_takes_the_slow_path() {
435        assert_eq!(floor_slot(MIN_FLOOR), Some(0));
436        assert_eq!(
437            floor_slot(MIN_FLOOR + FLOOR_SLOTS as i64 - 1),
438            Some(FLOOR_SLOTS - 1)
439        );
440        for floor in [
441            MIN_FLOOR - 1,
442            MIN_FLOOR + FLOOR_SLOTS as i64,
443            i64::MIN,
444            i64::MAX,
445            200,
446            -50,
447        ] {
448            assert_eq!(floor_slot(floor), None, "{floor}");
449        }
450        let mut list = FloorList::default();
451        let added: Vec<bool> = [5, 200, 5, -50, 200, 7, -50]
452            .into_iter()
453            .map(|floor| list.insert(floor))
454            .collect();
455        assert_eq!(added, [true, true, false, true, false, true, false]);
456        assert_eq!(list.floors, [5, 200, -50, 7]);
457        // Staff floors dedupe per kind, past the slots too.
458        let (mut floors, mut seen) = (Vec::new(), Vec::new());
459        for (kind, floor) in [
460            (Kind::Security, 5),
461            (Kind::Medical, 5),
462            (Kind::Security, 200),
463            (Kind::Security, 5),
464            (Kind::Medical, 200),
465            (Kind::Security, 200),
466            (Kind::Medical, 5),
467        ] {
468            add_staff_floor(&mut floors, &mut seen, kind, floor);
469        }
470        assert_eq!(
471            floors,
472            [
473                (Kind::Security, 5),
474                (Kind::Medical, 5),
475                (Kind::Security, 200),
476                (Kind::Medical, 200),
477            ]
478        );
