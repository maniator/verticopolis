1//! The office commuters in a save: each office's desk (the workers inside,
2//! on the way in and booked to leave) and the arrivals and departures booked
3//! for the rest of the hour, so a reload carries on the commute where it
4//! stood instead of booking it twice. The workers on the way are crowd people
5//! and travel with the rest of the crowd (`save.rs`).
6//!
7//! `desksStale` marks a save made after an edit the desks were not squared
8//! with yet (workers stranded, a venue's crew to call in), so the load
9//! leaves that to its first step, as the game that saved would have. `venueStaff` marks a
10//! save from a game whose venue staff commute (`CrowdMarks`).
11//!
12//! Both lists are written only while they hold something, so a tower whose
13//! offices count by the clock (every Classic tower, and every tower while the
14//! real commutes are switched off) saves neither. A save without them loads
15//! as before: the first hourly pass takes the head counts as the workers
16//! inside (`commuters::reconcile`).
17

33/// The most desks a save restores, one per office, staffed venue or home:
34/// far above the rooms any tower holds, it only bounds what a hand-edited
35/// save can make the loader keep.
36const MAX_SAVED_DESKS: usize = 10_000;
37
38/// A desk as the save writes it: `[officeId, floor, x, width, inside,
39/// coming, going]`.
40fn desk_json(id: i64, d: &Desk) -> Value {
41    json!([id, d.floor, d.x, d.width, d.inside, d.coming, d.going])
42}
43
44/// A booking's row: `[at, office, inbound]`, with the station it is held
45/// for as a fourth entry only when it has one, so a save without held
46/// arrivals is written as before.
47fn shift_json(s: &Shift) -> Value {
48    match s.station {
49        Some(sid) => json!([s.at, s.workplace_id, s.inbound, sid]),
50        None => json!([s.at, s.workplace_id, s.inbound]),
51    }
52}

54fn desk_from(v: &Value) -> Option<(i64, Desk)> {
55    let row = v.as_array().filter(|r| r.len() == 7)?;
56    let n: Vec<i64> = row
57        .iter()
58        .map(|n| n.as_f64().and_then(whole))
59        .collect::<Option<_>>()?;
60    // Head counts are never negative, and an office is at least a tile wide.
61    if n[3] < 1 || n[4..].iter().any(|&c| c < 0) {
62        return None;
63    }
64    // No more workers than an office holds, which no venue's staff pass
65    // (`no_venue_keeps_more_staff_than_an_office_holds`): each one inside may be sent out
66    // as a person on the first hourly pass (`commuters::reconcile`), so a
67    // forged count must not make that pass endless (#2230).
68    let most = Kind::Office.facility().population;
69    let desk = Desk {
70        floor: n[1],
71        x: n[2],
72        width: n[3],
73        inside: n[4].min(most),
74        coming: n[5].min(most),
75        going: n[6].min(most),
76    };
77    Some((n[0], desk))
78}
79
80fn shift_from(v: &Value) -> Option<Shift> {
81    let row = v.as_array().filter(|r| r.len() == 3 || r.len() == 4)?;
82    let station = match row.get(3) {
83        Some(sid) => Some(sid.as_f64().and_then(whole)?),
84        None => None,
85    };
86    Some(Shift {
87        at: row[0].as_f64().filter(|n| n.is_finite())?,
88        workplace_id: row[1].as_f64().and_then(whole)?,
89        inbound: row[2].as_bool()?,
90        station,
91    })
92}

94/// The desks as the save streams them: one small row at a time, so a tower
95/// with thousands of offices builds no list of them beside the save's text.
96struct DeskRows<'a>(&'a BTreeMap<i64, Desk>);
97
98impl Serialize for DeskRows<'_> {
99    fn serialize<S: Serializer>(&self, s: S) -> Result<S::Ok, S::Error> {
100        let mut seq = s.serialize_seq(Some(self.0.len()))?;
101        for (&id, d) in self.0 {
102            seq.serialize_element(&desk_json(id, d))?;
103        }
104        seq.end()
105    }
106}

108impl Crowd {
109    /// Writes the desks and the booked commutes into the crowd's save map
110    /// as [`Crowd::commuters_save`] adds them, each only when it holds
111    /// something.
112    pub(crate) fn commuters_write<M: SerializeMap>(
113        &self,
114        m: &mut M,
115        marks: CrowdMarks,
116    ) -> Result<(), M::Error> {
117        for key in marks.keys() {
118            m.serialize_entry(key, &true)?;
119        }
120        if !self.desks.is_empty() {
121            m.serialize_entry("desks", &DeskRows(&self.desks))?;
122        }
123        if !self.shifts.is_empty() {
124            m.serialize_entry("shifts", &Each(&self.shifts, shift_json))?;
125        }
126        Ok(())
127    }
128
129    /// Adds the desks and the booked commutes to the crowd's save object,
130    /// each only when it holds something.
131    pub(crate) fn commuters_save(&self, m: &mut Map<String, Value>, marks: CrowdMarks) {
132        for key in marks.keys() {
133            m.insert(key.into(), json!(true));
134        }
135        if !self.desks.is_empty() {
136            let rows = self.desks.iter().map(|(&id, d)| desk_json(id, d));
137            m.insert("desks".into(), Value::Array(rows.collect()));
138        }
139        if !self.shifts.is_empty() {
140            let rows = self.shifts.iter().map(shift_json);
141            m.insert("shifts".into(), Value::Array(rows.collect()));
142        }
143    }
144
145    /// The desks and the bookings a save's crowd object carries. A row that
146    /// does not read is skipped, and a second desk for one office keeps the
147    /// first.
148    pub(crate) fn commuters_restore(&mut self, o: &Map<String, Value>) {
149        if let Some(rows) = o.get("desks").and_then(Value::as_array) {
150            for (id, desk) in rows.iter().filter_map(desk_from) {
151                if self.desks.len() == MAX_SAVED_DESKS {
152                    break;
153                }
154                self.desks.entry(id).or_insert(desk);
155            }
156        }
157        if let Some(rows) = o.get("shifts").and_then(Value::as_array) {
158            let rows = rows.iter().take(MAX_SAVED_WALK_INS);
159            self.shifts.extend(rows.filter_map(shift_from));
160            self.sort_shifts();
161        }
162    }
163
164    /// Bounds the workers a save's desks can send out at once. The first
165    /// pass after a load (`commuters::reconcile`) walks the workers of an
166    /// office that no longer stands out of the building, one person each. A
167    /// save taken just after a bulldoze holds a few such desks; a forged one
168    /// could hold thousands. Desks of offices that are gone are kept in id
169    /// order while their workers fit beside the saved people under the
170    /// people a save may hold, and the rest are dropped.
171    pub(crate) fn bound_gone_desks(&mut self, tower: &Tower) {
172        let mut room = MAX_SAVED_PEOPLE.saturating_sub(self.people.len()) as i64;
173        self.desks.retain(|&id, desk| {
174            let stands = tower
175                .get_unit(id)
176                .is_some_and(|u| Workplace::of(u, tower.mode).is_some());
177            if stands {
178                return true;
179            }
180            let fits = desk.inside <= room;
181            if fits {
182                room -= desk.inside;
183            }
184            fits
185        });
186    }
187}

189/// What a save marks beside the crowd's desks.
190#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
191pub(crate) struct CrowdMarks {
192    /// `desksStale`: the next step squares the desks with an edit.
193    pub desks_stale: bool,
194    /// `venueStaff`: the game that saved had its venue staff commute, so a
195    /// venue without a desk had nobody in. A save without it is from before
196    /// they did, and its open venues start with their staff at work.
197    pub venue_staff: bool,
198}
199
200impl CrowdMarks {
201    /// The marks a save of `crowd` in `tower` at `hour` carries. A crowd
202    /// no step has squared yet is not marked `venueStaff`: its first step
203    /// has the open venues' staff at work, as a load without the mark does.
204    pub(crate) fn of(crowd: &Crowd, tower: &Tower, hour: i64) -> CrowdMarks {
205        let squared_once = crowd.desk_revision.is_some();
206        CrowdMarks {
207            desks_stale: desks_stale(crowd, tower, hour),
208            venue_staff: squared_once && tower.mode.venue_staff_commute_in_person(),
209        }
210    }
211
212    /// The marks set, by their save keys, in the order a save writes them.
213    fn keys(self) -> impl Iterator<Item = &'static str> {
214        let marks = [
215            ("desksStale", self.desks_stale),
216            ("venueStaff", self.venue_staff),
217        ];
218        marks.into_iter().filter(|m| m.1).map(|m| m.0)
219    }
220
221    /// The marks a save's crowd object carries.
222    pub(crate) fn read(crowd: Option<&Value>) -> CrowdMarks {
223        let set = |key: &str| crowd.and_then(|c| c.get(key)) == Some(&Value::Bool(true));
224        CrowdMarks {
225            desks_stale: set("desksStale"),
226            venue_staff: set("venueStaff"),
227        }
228    }
229}

231/// The first squaring of a fresh crowd: each office's head count becomes
232/// its workers inside, and, with `venues_at` an hour (a save from before
233/// venue staff commuted), each venue staffed then with no desk starts with
234/// its crew at work.
235pub(super) fn adopt_head_counts(crowd: &mut Crowd, tower: &Tower, venues_at: Option<i64>) {
236    for u in tower.room_units() {
237        let inside = match Workplace::of(u, tower.mode) {
238            Some(Workplace::Office | Workplace::Home | Workplace::Hotel) => u.occupants,
239            Some(Workplace::Venue)
240                if !crowd.desks.contains_key(&u.id)
241                    && u.is_operational()
242                    && venues_at.is_some_and(|hour| u.kind.staffed_at(hour)) =>
243            {
244                u.kind.staff_size()
245            }
246            _ => 0,
247        };
248        if inside > 0 {
249            let desk = crowd.desks.entry(u.id).or_default();
250            (desk.floor, desk.x, desk.width) = (u.floor, u.x, u.width);
251            desk.inside = inside;
252        }
253    }
254}

256/// Squares a loaded tower's desks as `reconcile` does, without sending
257/// anyone out: a save marks (`desksStale`) the workers a step would still
258/// send out, and they go at the first step, as in the game that saved. A
259/// save that marks `venueStaff` keeps a venue without a desk empty: its
260/// crew was not in yet. A save without it (one from before venue staff
261/// commuted, or one no step had squared) has its open venues' staff at
262/// work, and its first step squares the desks as a fresh crowd's does.
263pub fn restore_desks(crowd: &mut Crowd, tower: &mut Tower, hour: i64, marks: CrowdMarks) {
264    if !tower.mode.office_workers_commute_in_person() {
265        return;
266    }
267    let venues_at = (!marks.venue_staff).then_some(hour);
268    adopt_head_counts(crowd, tower, venues_at);
269    square(crowd, tower, false);
270    if !marks.venue_staff {
271        crowd.desk_revision = None;
272    } else if marks.desks_stale {
273        crowd.desk_revision = Some(UNSQUARED);
274    }
275}

277/// Whether the next step's squaring with an edit (`start_commutes`) does
278/// anything: sends out the workers a workplace taken away strands, or calls
279/// in the crew of a venue built while it should be open.
280pub fn desks_stale(crowd: &Crowd, tower: &Tower, hour: i64) -> bool {
281    let squared = crowd.desk_revision == Some(tower.revision);
282    let strands = || (crowd.desks.iter()).any(|(&id, d)| stranded(tower, id, d).1 > 0);
283    tower.mode.office_workers_commute_in_person()
284        && !squared
285        && (strands() || venue_bookings_due(crowd, tower, hour))
