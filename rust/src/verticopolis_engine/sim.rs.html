1//! One game: [`Simulation`] holds a whole tower, its people, its money and
2//! its clock, and every player command and time step goes through it.
3//!
4//! ```
5//! use verticopolis_engine::clock::GameMode;
6//! use verticopolis_engine::facilities::Kind;
7//! use verticopolis_engine::sim::Simulation;
8//!
9//! let mut sim = Simulation::new_game(42, GameMode::Classic);
10//! for x in 100..124 {
11//!     sim.build(Kind::Lobby, 1, x);
12//! }
13//! sim.tick(60.0);
14//! assert!(sim.money < verticopolis_engine::sim::STARTING_MONEY);
15//! ```

45/// The newest save format version this engine writes. It loads every older
46/// version back to the oldest one [`crate::identity::identity`] names.
47///
48/// Version 8 adds one key, `lobbyHeights` (#308), so a save is stamped with
49/// the oldest version that reads it whole ([`save_version`]): 8 when it
50/// holds a tall lobby, which a build from before tall lobbies refuses rather
51/// than drop, and [`PLAIN_SAVE_VERSION`] otherwise, which every build from
52/// version 7 on still loads.
53pub const SAVE_VERSION: i64 = 9;
54/// The version a save with no tall lobby is written at: the format before
55/// `lobbyHeights`, which it does not need.
56pub const PLAIN_SAVE_VERSION: i64 = 7;
57
58/// The version a save is stamped with: [`SAVE_VERSION`] when it holds a tall
59/// lobby, [`PLAIN_SAVE_VERSION`] when it does not.
60pub fn save_version(has_tall_lobby: bool) -> i64 {
61    if has_tall_lobby {
62        SAVE_VERSION
63    } else {
64        PLAIN_SAVE_VERSION
65    }
66}

160/// One game of Verticopolis.
161///
162/// Start one with [`Simulation::new_game`], or load one from a save with
163/// [`crate::load::deserialize`]. Advance it with [`Simulation::tick`] and
164/// change it only through its commands ([`build`](Simulation::build),
165/// [`add_car`](Simulation::add_car), [`adjust_rent`](Simulation::adjust_rent)
166/// and the rest), which apply the rules, charge the money and refuse what a
167/// player could not do. The public fields are for reading; writing one skips
168/// those rules.
169///
170/// A renderer reads the tower through [`crate::view`] or [`crate::frame`],
171/// and an analytics or achievements layer drains [`Simulation::gameplay`].
172pub struct Simulation {

184    /// web game shows these towers a founder badge.

313        };
314        Simulation {

316            mode,

321                true
322            },
323            founder: false,
324            clock: Clock::new(0.0, resolve_calendar(mode, modern_calendar)),
325            crowd: Crowd::new(seed),
326            events: EventSystem::new(seed),
327            ledger: Ledger::new(),
328            tower: {
329                let mut t = Tower::new();
330                t.allows_escalator_on_office_floors = mode.escalators_on_office_floors();
331                t.mode = mode;
332                t
333            },
334            money: STARTING_MONEY,
335            star: 1,
336            last_quarter_money: 0.0,
337            evaluated_tower: Some(false),
338            vip_visit_day: -1.0,
339            vip_favorable: false,
340            vip_visits: 0,
341            last_vip_nag_day: -100,
342            treasures_found: 0.0,
343            extermination_due_day: None,
344            excavated: Vec::new(),
345            blockbusters: Vec::new(),
346            milestones: Vec::new(),
347            log: Vec::new(),
348            log_seq: 0,
349            constructing: IndexSet::new(),
350            elevators: ElevatorDispatch::new(),
351            telemetry: ElevatorTelemetry::default(),

388    #[doc(hidden)]
389    /// `floorReachable(floor)`: memoized per tower revision, no rng.
390    pub fn floor_reachable(&mut self, floor: i64) -> bool {
391        if floor == 1 {
392            return true;
393        }
394        if self.reach_memo.0 != self.tower.revision {
395            self.reach_memo = (self.tower.revision, HashMap::new());
396        }
397        if let Some(&hit) = self.reach_memo.1.get(&floor) {
398            return hit;
399        }
400        let hit = self.crowd.floor_reachable(&self.tower, floor);
401        self.reach_memo.1.insert(floor, hit);
402        hit
403    }

405    #[doc(hidden)]
406    /// `positionReachable(floor, x)`.
407    pub fn position_reachable(&mut self, floor: i64, x: i64) -> bool {
408        if floor == 1 {
409            return true;
410        }
411        if self.tower.segments_of(floor).len() <= 1 {
412            return self.floor_reachable(floor);
413        }
414        self.crowd.position_reachable(&self.tower, floor, x)
415    }

417    #[doc(hidden)]
418    /// `unitReachable` (sim/demand.ts).
419    pub fn unit_reachable(&mut self, floor: i64, x: i64) -> bool {
420        self.position_reachable(floor, x)
421    }

423    /// Starts a new game: an empty lot, [`STARTING_MONEY`], 07:00 on day 0.
424    ///
425    /// The same `seed` and the same commands give the same game. A Modern
426    /// game gets the real-world calendar and automatic floor bridging; use
427    /// [`Simulation::new_game_with`] to choose them.
428    ///
429    /// ```
430    /// use verticopolis_engine::clock::GameMode;
431    /// use verticopolis_engine::sim::Simulation;
432    ///
433    /// let sim = Simulation::new_game(7, GameMode::Modern);
434    /// assert_eq!(sim.star, 1);
435    /// assert_eq!(sim.population(), 0);
436    /// ```
437    pub fn new_game(seed: u32, mode: GameMode) -> Simulation {
438        Simulation::new_game_with(seed, mode, CalendarKind::RealWorld, false)
439    }
440
441    /// Starts a new game with a choice of calendar and bridging. Both apply
442    /// to a Modern game only: a Classic game always runs the canon calendar
443    /// with bridging on, whatever these say.
444    ///
445    /// ```
446    /// use verticopolis_engine::clock::{CalendarKind, GameMode};
447    /// use verticopolis_engine::sim::Simulation;
448    ///
449    /// let sim = Simulation::new_game_with(7, GameMode::Modern, CalendarKind::Canon, true);
450    /// assert!(!sim.auto_bridge);
451    /// ```
452    pub fn new_game_with(
453        seed: u32,
454        mode: GameMode,
455        modern_calendar: CalendarKind,
456        start_unbridged: bool,
457    ) -> Simulation {
458        let mut sim = Simulation::new(seed, mode, modern_calendar, start_unbridged);
459        sim.emit(
460            "Welcome! Lay a lobby on the ground line to open your tower.",
461            LogKind::Info,
462        );
463        sim.gameplay
464            .push(GameplayEvent::TowerFounded { mode: sim.mode });
465        sim
466    }

468    #[doc(hidden)]
469    /// `sim.emit(text, kind)`: append to the ring, no dedupe.
470    pub fn emit(&mut self, text: &str, kind: LogKind) {
471        self.log.push(LogEntry {
472            minute: self.clock.minutes,
473            text: text.to_string(),
474            kind,
475        });
476        self.log_seq += 1;
477        if self.log.len() > LOG_RING_CAP {
478            self.log.remove(0);
479        }
480    }
