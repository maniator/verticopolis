1//! The ground a step leaves (`sim_undo_check.rs`): the tower's floor and
2//! lobby tiles, less those of the units the step takes out, with those of
3//! the units it puts back. A room, a shaft or the metro station the step
4//! puts back must stand where building it could, a floor or lobby tile
5//! where its story and the station let it, and what the step leaves
6//! standing must keep standing. The lobby rule, that taking out a
7//! ground-floor lobby tile leaves nothing laid since hanging, has its own
8//! file (`sim_undo_lobby.rs`).
9use crate::crowd::fasthash::{FastMap, FastSet};
10use crate::facilities::{is_sky_lobby_floor, Kind, LOT_WIDTH, MIN_FLOOR};
11use crate::tower::{floor_left_of, floor_right_of, Tower, Transport, Unit};
12use crate::undo::{Step, UnitChange};

14/// The floor the lobby concourse runs along.
15const GROUND_FLOOR: i64 = 1;

17#[path = "sim_undo_lobby.rs"]

20pub(crate) struct Ground<'a> {
21    tower: &'a Tower,
22    leaving: &'a FastSet<i64>,
23    /// The floor and lobby tiles the step puts back, each with the id and
24    /// kind of the unit it puts back there.
25    laid: FastMap<(i64, i64), (i64, Kind)>,
26    /// The floor and lobby tiles the step takes out and does not lay again,
27    /// as (x, floor) in order, so a shaft finds those in its column with a
28    /// search.
29    taken: Vec<(i64, i64)>,
30    /// Where on the ground floor the step takes out a lobby tile and does
31    /// not lay a tile again.
32    lobbies_taken: Vec<i64>,
33    /// The lobby tiles the step adds to each floor, less those it takes out.
34    lobbies_added: FastMap<i64, i64>,
35    /// The room tiles the step takes out on each row a room it takes out
36    /// stands on.
37    rooms_leaving: FastMap<i64, i64>,
38    /// The row the metro stations stand on once the step applies: that of a
39    /// station the step leaves standing, else that of one it puts back.
40    station: Option<i64>,
41    /// The ground-floor and basement tiles the step lays that reach no tile
42    /// there that stays (`adrift_low_tiles`, #1843).
43    adrift: FastSet<(i64, i64)>,
44}

46impl<'a> Ground<'a> {
47    pub(crate) fn of(tower: &'a Tower, step: &Step, leaving: &'a FastSet<i64>) -> Self {
48        let put_back: Vec<&Unit> = step
49            .units
50            .iter()
51            .filter_map(|img| match &img.change {
52                UnitChange::PutBack(u) => Some(u),
53                _ => None,
54            })
55            .collect();
56        let mut laid = FastMap::default();
57        for u in put_back.iter().filter(|u| u.kind.is_structural()) {
58            for tile in tiles_of(u) {
59                laid.insert(tile, (u.id, u.kind));
60            }
61        }
62        let mut lobbies_added = FastMap::default();
63        for u in put_back.iter().filter(|u| u.kind == Kind::Lobby) {
64            *lobbies_added.get_or_insert_with(u.floor, || 0) += u.width;
65        }
66        // One walk over the units the step takes out.
67        let (mut taken, mut lobbies_taken) = (Vec::new(), Vec::new());
68        let mut rooms_leaving = FastMap::default();
69        let taken_out = step
70            .units
71            .iter()
72            .filter(|img| matches!(img.change, UnitChange::TakeOut(_)))
73            .filter_map(|img| tower.get_unit(img.id));
74        for u in taken_out {
75            if u.kind.is_structural() {
76                for (floor, x) in tiles_of(u).filter(|tile| !laid.contains_key(tile)) {
77                    taken.push((x, floor));
78                    if u.kind == Kind::Lobby && floor == GROUND_FLOOR {
79                        lobbies_taken.push(x);
80                    }
81                }
82                if u.kind == Kind::Lobby {
83                    *lobbies_added.get_or_insert_with(u.floor, || 0) -= u.width;
84                }
85            } else {
86                for row in u.floor..u.floor + u.kind.floors() {
87                    *rooms_leaving.get_or_insert_with(row, || 0) += u.width;
88                }
89            }
90        }
91        taken.sort_unstable();
92        let station = tower
93            .metro_stations()
94            .find(|u| !leaving.contains(&u.id))
95            .or_else(|| put_back.iter().copied().find(|u| u.kind == Kind::Metro))
96            .map(|u| u.floor);
97        let mut ground = Ground {
98            tower,
99            leaving,
100            laid,
101            taken,
102            lobbies_taken,
103            lobbies_added,
104            rooms_leaving,
105            station,
106            adrift: FastSet::default(),
107        };
108        // The ground-floor and basement tiles the step lays, in step order.
109        let low = put_back
110            .iter()
111            .filter(|u| u.kind.is_structural() && u.floor <= GROUND_FLOOR)
112            .flat_map(|u| tiles_of(u))
113            .collect();
114        ground.adrift = ground.adrift_low_tiles(low);
115        ground
116    }

118    /// The ground-floor and basement tiles the step lays that reach no tile
119    /// there that stays, through tiles beside, above or below each other that
120    /// it lays too. A tile there hangs from one beside, above or below it, as
121    /// its build asked (`Tower::is_supported`), so a tile put back after its
122    /// neighbors went comes back touching nothing (#1843). Where no
123    /// ground-floor tile stays at all, the tiles the step lays on the ground
124    /// floor stand on their own, as a tower's first lobby does, and what they
125    /// reach is held. `low` holds the ground-floor and basement tiles the step
126    /// lays. One walk over them, plus one row of the lot.
127    fn adrift_low_tiles(&self, low: Vec<(i64, i64)>) -> FastSet<(i64, i64)> {
128        if low.is_empty() {
129            return FastSet::default();
130        }
131        let laid_low = |tile: &(i64, i64)| tile.0 <= GROUND_FLOOR && self.laid.contains_key(tile);
132        let stays_low = |(floor, x): (i64, i64)| {
133            floor <= GROUND_FLOOR && !self.laid.contains_key(&(floor, x)) && self.stays(floor, x)
134        };
135        let first = !(0..LOT_WIDTH).any(|x| stays_low((GROUND_FLOOR, x)));
136        let mut met: Vec<(i64, i64)> = low
137            .iter()
138            .copied()
139            .filter(|&tile| {
140                (first && tile.0 == GROUND_FLOOR) || lobby::beside(tile).into_iter().any(stays_low)
141            })
142            .collect();
143        let mut held: FastSet<(i64, i64)> = met.iter().copied().collect();
144        let mut next = 0;
145        while let Some(&tile) = met.get(next) {
146            next += 1;
147            for near in lobby::beside(tile) {
148                if laid_low(&near) && held.insert(near) {
149                    met.push(near);
150                }
151            }
152        }
153        low.into_iter()
154            .filter(|tile| !held.contains(tile))
155            .collect()
156    }

158    fn holds(&self, floor: i64, x: i64) -> bool {
159        self.laid.contains_key(&(floor, x)) || self.stays(floor, x)
160    }

162    /// The id and kind of the unit whose floor or lobby tile holds at
163    /// `tile`, as (floor, x), once the step applies: the one the step puts
164    /// back there, or the tower's that stays. Unit ids only grow, and a unit
165    /// an undo puts back keeps its own, so the id says when the tile was
166    /// built.
167    fn built_by(&self, tile: (i64, i64)) -> Option<(i64, Kind)> {
168        if let Some(&laid) = self.laid.get(&tile) {
169            return Some(laid);
170        }
171        let id = *self.tower.structure.get(&tile)?;
172        let kind = self.tower.struct_kind.get(&tile).copied()?;
173        (!self.leaving.contains(&id)).then_some((id, kind))
174    }

176    /// Whether the tower's floor or lobby tile at (`floor`, `x`) stays,
177    /// which is all a sale of what the step takes out leaves there.
178    fn stays(&self, floor: i64, x: i64) -> bool {
179        self.tower
180            .structure
181            .get(&(floor, x))
182            .is_some_and(|id| !self.leaving.contains(id))
183    }

185    /// Whether the step lays one of the tiles of floor or lobby tile `u`
186    /// again.
187    pub(crate) fn lays_again(&self, u: &Unit) -> bool {
188        tiles_of(u).any(|tile| self.laid.contains_key(&tile))
189    }

191    /// Whether `floor` has a lobby tile once the step applies.
192    fn has_lobby(&self, floor: i64) -> bool {
193        let added = self.lobbies_added.get(&floor).copied().unwrap_or(0);
194        self.tower.lobby_tile_count(floor) + added > 0
195    }

197    /// Whether the step leaves a room tile on `row`.
198    fn rooms_stay_on(&self, row: i64) -> bool {
199        let on_row = self.tower.room_tiles.get(&row).copied().unwrap_or(0);
200        on_row > self.rooms_leaving.get(&row).copied().unwrap_or(0)
201    }

203    /// Whether unit `u` stands where the build rules let it. A floor or lobby
204    /// tile stands on the metro station's row or above it, and above the ground
205    /// floor rests on the story below, as `Tower::is_supported` asks; a lobby
206    /// tile goes back on a sky lobby floor only where no room stays on that
207    /// floor (`Tower::can_place`). A floor tile may go back on a sky lobby
208    /// floor beside a room, or beside a lobby the lobby brush left there,
209    /// though `Tower::can_place` refuses a new one beside a lobby. A tile on
210    /// the ground floor or below must reach a tile there that stays, through
211    /// the tiles beside, above or below it (`adrift_low_tiles`, #1843). A room has
212    /// floor under every tile, as its build laid (`Tower::ensure_floor_under`),
213    /// sits on no sky lobby floor that holds a lobby (so never on a lobby tile,
214    /// which goes only on those floors and the ground floor, where no room
215    /// goes), and stands above the station's row (`metro_reason`). A wedding
216    /// hall's upper story stands above the top slab. The station itself has
217    /// nothing below it and no room on its row.
218    pub(crate) fn under_room(&self, u: &Unit) -> bool {
219        if u.kind.is_structural() {
220            let rests = if u.floor <= GROUND_FLOOR {
221                tiles_of(u).all(|tile| !self.adrift.contains(&tile))
222            } else {
223                (u.x..u.x + u.width).all(|x| self.holds(u.floor - 1, x))
224            };
225            let concourse = u.kind == Kind::Lobby && is_sky_lobby_floor(u.floor);
226            return rests
227                && !(concourse && self.rooms_stay_on(u.floor))
228                && self.station.is_none_or(|station| u.floor >= station);
229        }
230        let stories = self.tower.lot().slab_stories(u.kind, u.floor);
231        let floored = (u.floor..u.floor + stories).all(|f| {
232            !(is_sky_lobby_floor(f) && self.has_lobby(f))
233                && (u.x..u.x + u.width).all(|x| self.holds(f, x))
234        });
235        let site = if u.kind == Kind::Metro {
236            self.station == Some(u.floor) && self.clear_for_station(u.floor)
237        } else {
238            self.station.is_none_or(|station| u.floor > station)
239        };
240        floored && site
241    }
242
243    /// Whether nothing the step leaves stands below `row`, and no room it
244    /// leaves stands on that row but the other metro stations, so a station
245    /// can stand there beside them. A unit the step puts back below the row,
246    /// or a room it puts back on it, is refused on its own.
247    fn clear_for_station(&self, row: i64) -> bool {
248        let staying_stations: i64 = self
249            .tower
250            .metro_stations()
251            .filter(|u| u.floor == row && !self.leaving.contains(&u.id))
252            .map(|u| u.width)
253            .sum();
254        let on_row = self.tower.room_tiles.get(&row).copied().unwrap_or(0);
255        let leaving = self.rooms_leaving.get(&row).copied().unwrap_or(0);
256        if on_row - leaving > staying_stations {
257            return false;
258        }
259        // Every room stands on floor, so the floor tiles say where the
260        // units are.
261        let lowest = self.tower.floor_bounds().0.max(MIN_FLOOR);
262        (lowest..row).all(|floor| (0..LOT_WIDTH).all(|x| !self.holds(floor, x)))
263    }

265    /// Whether shaft `t` has floor or lobby in its column on every story it
266    /// spans, as `Tower::validate_transport` asks.
267    pub(crate) fn under_shaft(&self, t: &Transport) -> bool {
268        (t.bottom..=t.top).all(|f| (t.x..t.x + t.width).any(|x| self.holds(f, x)))
269    }

271    /// Whether shaft `t`, which stood on the tower's tiles, still has floor
272    /// or lobby in its column on each story where the step takes a tile
273    /// from under it.
274    pub(crate) fn still_under_shaft(&self, t: &Transport) -> bool {
275        self.under_shaft_where(t, |floor, x| self.holds(floor, x))
276    }

278    /// Whether shaft `t` would still have that floor once the player sold
279    /// what the step takes out: a sale lays no tile.
280    pub(crate) fn still_under_shaft_once_sold(&self, t: &Transport) -> bool {
281        self.under_shaft_where(t, |floor, x| self.stays(floor, x))
282    }

284    /// Whether shaft `t` has a tile that `holds` in its column on each
285    /// story where the step takes a tile from under it. Each column of the
286    /// shaft is a run of `taken`, found by a search.
287    fn under_shaft_where(&self, t: &Transport, holds: impl Fn(i64, i64) -> bool) -> bool {
288        let columns = t.x..t.x + t.width;
289        columns.clone().all(|x| {
290            let from = self.taken.partition_point(|&tile| tile < (x, t.bottom));
291            let to = self.taken.partition_point(|&tile| tile <= (x, t.top));
292            // A shaft whose span runs backward has no tiles under it.
293            self.taken
294                .get(from..to)
295                .unwrap_or_default()
296                .iter()
297                .all(|&(_, f)| columns.clone().any(|x| holds(f, x)))
298        })
299    }

301    /// Whether the player could sell floor tile `u`, with the other tiles
302    /// the step takes out, by the floor rule a sale keeps to
303    /// (`Tower::removal_reason`): in a mode that keeps floors whole
304    /// ([`GameMode::removes_mid_floor_tiles`](crate::clock::GameMode::removes_mid_floor_tiles)),
305    /// the sales take each run of leaving tiles from one of its ends, by
306    /// [`floor_left_of`] and [`floor_right_of`] over the tiles that stay, as
307    /// `Tower::floor_on_both_sides` asks of a sale. So a gap a save already
308    /// has is no end, and a lobby a run touches holds it in. A run still
309    /// standing is plain floor on one side of every other run on its story,
310    /// so the sales start at the story's two ends and work inward: the run
311    /// of `u` comes out once every run from one end of the story through it
312    /// can come out toward that end. Only the run's first tile asks, for the
313    /// whole run, and the others answer yes, since the step is sellable only
314    /// when every tile is.
315    pub(crate) fn sells_from_an_end(&self, u: &Unit) -> bool {
316        if self.tower.mode.removes_mid_floor_tiles() || self.leaves(u.floor, u.x - 1) {
317            return true;
318        }
319        // The run ends at the first tile past `u` that does not leave.
320        let run_end = (u.x + u.width..LOT_WIDTH)
321            .find(|&x| !self.leaves(u.floor, x))
322            .unwrap_or(LOT_WIDTH);
323        self.comes_out_leftward(u.floor, u.x) || self.comes_out_rightward(u.floor, run_end)
324    }
325
326    /// Whether every run of leaving tiles on `floor`, from the lot's left
327    /// edge through the run that starts at `start`, can come out from its
328    /// left end: none has floor on its left by [`floor_left_of`]. No plain
329    /// floor that stays is left of the run at `start`, so none is left of an
330    /// earlier run, and an earlier run has floor on its left only where a
331    /// tile that stays touches it.
332    fn comes_out_leftward(&self, floor: i64, start: i64) -> bool {
333        !floor_left_of(start, |x| self.kind_staying(floor, x))
334            && !(0..start).any(|x| self.stays(floor, x) && self.leaves(floor, x + 1))
335    }
336
337    /// Whether every run of leaving tiles on `floor`, from the run that
338    /// ends before `end` through the lot's right edge, can come out from its
339    /// right end, by [`floor_right_of`] and the reasoning of
340    /// `comes_out_leftward`.
341    fn comes_out_rightward(&self, floor: i64, end: i64) -> bool {
342        !floor_right_of(end, |x| self.kind_staying(floor, x))
343            && !(end..LOT_WIDTH).any(|x| self.stays(floor, x) && self.leaves(floor, x - 1))
344    }

346    /// Whether the tower's floor or lobby tile at (`floor`, `x`) is one the
347    /// step takes out.
348    fn leaves(&self, floor: i64, x: i64) -> bool {
349        self.tower
350            .structure
351            .get(&(floor, x))
352            .is_some_and(|id| self.leaving.contains(id))
353    }

355    /// The kind of the tower's floor or lobby tile at (`floor`, `x`) when it
356    /// stays, which is all a sale of what the step takes out leaves there.
357    fn kind_staying(&self, floor: i64, x: i64) -> Option<Kind> {
358        self.tower
359            .structure_kind_at(floor, x)
360            .filter(|_| self.stays(floor, x))
361    }

363    /// Whether taking out floor or lobby tile `u` leaves everything it held
364    /// up standing, by the rules a sale keeps to (`Tower::removal_reason`):
365    /// no room stays on its tiles, and from the ground floor up no story
366    /// stays above it, unless the step lays the tile again. The story above
367    /// a tall lobby tile is the one over its hall, since the stories it
368    /// rises through are its own, and a tile the step lays again in its
369    /// place holds only its own story, so it never holds that one up.
370    pub(crate) fn leaves_nothing_hanging_from(&self, u: &Unit) -> bool {
371        let above = self.tower.unit_stories(u);
372        tiles_of(u).all(|(f, x)| {
373            let room_stays = self
374                .tower
375                .rooms
376                .get(&(f, x))
377                .is_some_and(|id| !self.leaving.contains(id));
378            (above == 1 && self.laid.contains_key(&(f, x)))
379                || (!room_stays && !(f >= GROUND_FLOOR && self.holds(f + above, x)))
380        })
381    }
382}

384/// The tiles floor or lobby unit `u` covers.
385fn tiles_of(u: &Unit) -> impl Iterator<Item = (i64, i64)> + '_ {
386    (u.x..u.x + u.width).map(move |x| (u.floor, x))
