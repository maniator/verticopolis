31/// Result of a successful parse: the save, the binary walk's warnings and the
32/// tally the report is built from.
33#[derive(Clone, Debug)]
34pub struct ParsedLegacyTower {
35    pub save: Value,
36    pub warnings: Vec<String>,
37    pub counts: ImportCounts,
38    pub decode_stats: DecodeStats,
39    pub decoded: bool,
40    pub header_notes: Vec<String>,
41    pub tower: TdtTower,
42}

44/// A number as JavaScript would serialize it: an integer when whole.
45pub fn jnum(x: f64) -> Value {
46    if x.fract() == 0.0 && x.abs() < 9007199254740992.0 {
47        json!(x as i64)
48    } else {
49        json!(x)
50    }
51}

53/// `parseTDT`: an original SimTower `.TDT` buffer into our save schema.
54pub fn parse_tdt(bytes: &[u8], filename: &str) -> Result<ParsedLegacyTower, LegacyImportError> {
55    let tdt = parse_tdt_binary(bytes)?;
56
57    let mut header_notes: Vec<String> = vec![];
58    let star = tdt.header.level.clamp(1, 6);
59    if tdt.header.level < 1 || tdt.header.level > 6 {
60        header_notes.push(format!(
61            "The save's star rating ({}) was out of range and was clamped.",
62            tdt.header.level
63        ));
64    }
65    let money = tdt.header.balance * 100;
66    let day = tdt.header.current_day.clamp(0, MAX_IMPORT_DAY);
67    if tdt.header.current_day < 0 {
68        header_notes.push(
69            "The save's day counter was negative (a known quirk of the format) and was reset to day 1."
70                .into(),
71        );
72    } else if tdt.header.current_day > MAX_IMPORT_DAY {
73        header_notes.push(
74            "The save's day counter was impossibly far in the future and was clamped.".into(),
75        );
76    }
77    let frame = tdt.header.frame_time.clamp(0, 2599);
78    if tdt.header.frame_time > 2599 {
79        header_notes.push(
80            "The save's clock was out of range and was reset to the end of the night.".into(),
81        );
82    }
83    let minutes = day * 1440 + minute_of_day_for_frame(frame as f64) as i64;
84
85    let mut counts = ImportCounts::default();
86    let width = LOT_WIDTH as usize;
87    // Paved tiles per floor, in ascending floor order (the TypeScript map is
88    // filled in floor-index order, so insertion order and floor order agree).
89    let mut paved: BTreeMap<i64, Vec<u8>> = BTreeMap::new();
90    let pave_range = |paved: &mut BTreeMap<i64, Vec<u8>>, floor: i64, left: i64, right: i64| {
91        let lo = left.clamp(0, LOT_WIDTH) as usize;
92        let hi = right.clamp(0, LOT_WIDTH) as usize;
93        if hi <= lo {
94            return;
95        }
96        let row = paved.entry(floor).or_insert_with(|| vec![0; width]);
97        for cell in &mut row[lo..hi] {
98            *cell = 1;
99        }
100    };
101    let mut room_claimed: HashMap<i64, Vec<u8>> = HashMap::new();
102    let claim_room = |claimed: &mut HashMap<i64, Vec<u8>>,
103                      kind: Kind,
104                      floor: i64,
105                      left: i64,
106                      right: i64|
107     -> bool {
108        let stories = kind.floors();
109        for f in floor..floor + stories {
110            let row = claimed.entry(f).or_insert_with(|| vec![0; width]);
111            if row[left as usize..right as usize].iter().any(|&c| c != 0) {
112                return false;
113            }
114        }
115        for f in floor..floor + stories {
116            let row = claimed.entry(f).or_insert_with(|| vec![0; width]);
117            for cell in &mut row[left as usize..right as usize] {
118                *cell = 1;
119            }
120        }
121        true
122    };
123    // Clamp a tenant's extents onto the lot; `None` (counted) for a degenerate
124    // or fully off-lot extent.
125    let clamp_extent = |counts: &mut ImportCounts, t: &TdtTenant| -> Option<(i64, i64)> {
126        let x = t.left;
127        let mut right = t.right;
128        if right <= x || x >= LOT_WIDTH {
129            counts.off_lot += 1;
130            return None;
131        }
132        if right > LOT_WIDTH {
133            right = LOT_WIDTH;
134            counts.clamped += 1;
135        }
136        Some((x, right))
137    };
138
139    let mut next_id = 1;
140    let mut units: Vec<OutUnit> = vec![];
141    let mut part_records: Vec<PartRecord> = vec![];
142    let push_unit =
143        |units: &mut Vec<OutUnit>, counts: &mut ImportCounts, next_id: &mut i64, unit: OutUnit| {
144            counts.rooms += 1;
145            if unit.width != unit.kind.facility().width {
146                counts.width_mismatch += 1;
147            }
148            if unit.state == "construction" {
149                counts.construction += 1;
150            }
151            *next_id += 1;
152            units.push(unit);
153        };
154
155    for fl in &tdt.floors {
156        let ours = fl.index - TDT_FLOOR_OFFSET;
157        if ours > MAX_FLOOR {
158            if !fl.tenants.is_empty() || fl.right_edge > fl.left_edge {
159                counts.dropped_floors += 1;
160            }
161            continue;
162        }
163        pave_range(&mut paved, ours, fl.left_edge, fl.right_edge);
164        for (at, t) in fl.tenants.iter().enumerate() {
165            let under_construction = t.type_id < 0;
166            let type_id = t.type_id.abs();
167            if type_id == TDT_FLOOR || type_id == TDT_LOBBY {
168                pave_range(&mut paved, ours, t.left, t.right);
169                continue;
170            }
171            if type_id == TDT_BURNED {
172                pave_range(&mut paved, ours, t.left, t.right);
173                counts.burned += 1;
174                continue;
175            }
176            if type_id == TDT_METRO_TUNNEL {
177                pave_range(&mut paved, ours, t.left, t.right);
178                continue;
179            }
180            if let Some(part_kind) = part_family(type_id) {
181                let Some((x, right)) = clamp_extent(&mut counts, t) else {
182                    continue;
183                };
184                pave_range(&mut paved, ours, x, right);
185                part_records.push(PartRecord {
186                    kind: part_kind,
187                    type_id,
188                    floor: ours,
189                    left: x,
190                    right,
191                    construction: under_construction,
192                });
193                continue;
194            }
195            let Some(kind) = tenant_kind(type_id) else {
196                counts.unknown += 1;
197                continue;
198            };
199            let Some((x, right)) = clamp_extent(&mut counts, t) else {
200                continue;
201            };
202            if misplaced_on_floor(kind, ours) {
203                counts.misplaced += 1;
204                continue;
205            }
206            if !claim_room(&mut room_claimed, kind, ours, x, right) {
207                counts.overlapping += 1;
208                continue;
209            }
210            pave_range(&mut paved, ours, x, right);
211
212            let mut state = if under_construction {
213                "construction"
214            } else {
215                "empty"
216            };
217            let mut ever_occupied = false;
218            let mut occupants = 0;
219            if !under_construction && (kind == Kind::Office || kind == Kind::Condo) && t.status != 0
220            {
221                state = "occupied";
222                ever_occupied = true;
223                occupants = kind.facility().population;
224            } else if !under_construction && kind.is_hotel() && t.status != 0 {
225                if t.status & HOTEL_INFESTED_FLAG != 0 {
226                    state = "dirty";
227                    ever_occupied = true;
228                    counts.infested += 1;
229                    counts.hotel_dirty += 1;
230                } else if t.status & HOTEL_DIRTY_FLAG != 0 {
231                    state = "dirty";
232                    ever_occupied = true;
233                    counts.hotel_dirty += 1;
234                } else {
235                    ever_occupied = true;
236                    // The finance block counts the guests of every checked-in or
237                    // sleeping room at full complement, and our engine keeps one
238                    // in-room guest state, so both arrive as guests in their room.
239                    match t.status & HOTEL_STAY_STATE_MASK {
240                        HOTEL_STAY_CHECKED_IN | HOTEL_ASLEEP_FLAG => {
241                            state = "asleep";
242                            occupants = kind.facility().population;
243                            counts.hotel_asleep += 1;
244                        }
245                        _ => counts.hotel_booked += 1,
246                    }
247                }
248            } else if !under_construction
249                && matches!(kind, Kind::FastFood | Kind::Restaurant | Kind::Shop)
250            {
251                state = "occupied";
252                ever_occupied = true;
253            }
254            let rent = rent_from_class(kind, t.rent_rate).or_else(|| {
255                if t.rent_rate != 4 && is_priced(kind) {
256                    rent_from_class(kind, 2)
257                } else {
258                    None
259                }
260            });
261            if rent.is_some() {
262                counts.rents_applied += 1;
263            }
264            let no_rate = t.rent_rate == 4 && is_priced(kind);
265            let subtype = retail_subtype(&tdt, fl, at, t, kind);
266            let unit = OutUnit {
267                id: next_id,
268                kind,
269                floor: ours,
270                x,
271                width: right - x,
272                state,
273                occupants,
274                ever_occupied,
275                rent,
276                no_rate,
277                subtype,
278                complete_at: under_construction.then(|| minutes + kind.build_minutes() as i64),
279            };
280            push_unit(&mut units, &mut counts, &mut next_id, unit);
281
282            if kind == Kind::Office {
283                counts.offices += 1;
284                if state == "occupied" {
285                    counts.occupied_offices += 1;
286                }
287            } else if kind == Kind::Condo {
288                counts.condos += 1;
289                if state == "occupied" {
290                    counts.sold_condos += 1;
291                }
292            } else if kind.is_hotel() {
293                counts.hotel_rooms += 1;
294            } else if matches!(kind, Kind::FastFood | Kind::Restaurant | Kind::Shop) {
295                counts.venues += 1;
296            } else {
297                counts.services += 1;
298                if kind == Kind::Parking {
299                    counts.parking_stalls += 1;
300                }
301            }
302            if type_id == 4 {
303                counts.twin_rooms += 1;
304            }
305            if type_id == 17 {
306                counts.secom += 1;
307            }
308        }
309    }
310
311    close_parking_gaps(&mut units);
312    if let Some(pops) = tdt.tail.finance_population {
313        units = seed_from_finance(units, &pops, &mut counts);
314    }
315
316    for m in merge_parts(&part_records) {
317        // The cathedral stands on its bottom part and rises above floor 100, so
318        // every kind keys off its lowest floor.
319        let floor = m.floor;
320        if floor + m.kind.floors() - 1 > m.kind.highest_story() {
321            counts.off_lot += 1;
322            continue;
323        }
324        if misplaced_on_floor(m.kind, floor) {
325            counts.misplaced += 1;
326            continue;
327        }
328        if !claim_room(&mut room_claimed, m.kind, floor, m.left, m.right) {
329            counts.overlapping += 1;
330            continue;
331        }
332        let unit = OutUnit {
333            id: next_id,
334            kind: m.kind,
335            floor,
336            x: m.left,
337            width: m.right - m.left,
338            state: match (m.construction, counts_set_audience(m.kind)) {
339                (true, _) => "construction",
340                (false, true) => "occupied", // an open hall or theatre (#1500)
341                (false, false) => "empty",
342            },
343            occupants: 0,
344            ever_occupied: false,
345            rent: None,
346            no_rate: false,
347            subtype: None,
348            complete_at: m
349                .construction
350                .then(|| minutes + m.kind.build_minutes() as i64),
351        };
352        push_unit(&mut units, &mut counts, &mut next_id, unit);
353        if m.kind == Kind::Cinema || m.kind == Kind::PartyHall {
354            counts.venues += 1;
355        } else {
356            counts.services += 1;
357        }
358        if m.kind == Kind::WeddingHall {
359            counts.cathedral += 1;
360        }
361    }
362
363    // The paving pass: width-1 tiles, lobby on lobby floors, none on the
364    // stories a tall ground lobby rises through.
365    let lobby_height = stories::raise_ground_lobby(&mut paved, &tdt, &units);
366    let mut unit_values: Vec<Value> = units.iter().map(OutUnit::to_json).collect();
367    seed::seed_visitors(&mut unit_values, &tdt);
368    seed::seed_halls_and_theatres(&mut unit_values, &tdt);
369    let mut built_extents: BuiltExtents = BTreeMap::new();
370    for (&floor, row) in &paved {
371        let kind = if is_lobby_floor(floor) {
372            Kind::Lobby
373        } else {
374            Kind::Floor
375        };
376        let mut left = -1;
377        let mut right = -1;
378        for (x_tile, &cell) in row.iter().enumerate() {
379            if cell == 0 {
380                continue;
381            }
382            if left == -1 {
383                left = x_tile as i64;
384            }
385            right = x_tile as i64 + 1;
386            unit_values.push(json!({
387                "id": next_id,
388                "kind": kind.as_str(),
389                "floor": floor,
390                "x": x_tile,
391                "width": 1,
392                "state": "empty",
393                "satisfaction": 1,
394                "occupants": 0,
395                "everOccupied": false,
396                "pendingIncome": 0,
397                "label": kind.facility().name,
398            }));
399            next_id += 1;
400        }
401        if left != -1 {
402            built_extents.insert(floor, (left, right));
403        }
404    }
405
406    let excavated = stories::excavated_tiles(&paved);
407
408    let decoded = tdt.tail.elevators.is_some();
409    let mut decode_stats = DecodeStats::default();
410    let mut warnings = tdt.tail.warnings.clone();
411    let transports: Vec<OutTransport> = match &tdt.tail.elevators {
412        Some(elevators) => {
413            let d = transports_from_decoded(
414                elevators,
415                tdt.tail.stairs.as_deref().unwrap_or(&[]),
416                next_id,
417                lobby_height,
418            );
419            decode_stats = DecodeStats {
420                dropped_shafts: d.dropped_shafts,
421                adjusted_shafts: d.adjusted_shafts,
422                dropped_flights: d.dropped_flights,
423            };
424            d.transports
425        }
426        None => {
427            let rebuilt = fallback_transports(&units, &built_extents, next_id);
428            if rebuilt.left_out > 0 {
429                warnings.push(format!(
430                    "{} rebuilt elevator shaft{} had no free column and {} left out.",
431                    rebuilt.left_out,
432                    if rebuilt.left_out == 1 { "" } else { "s" },
433                    if rebuilt.left_out == 1 { "was" } else { "were" },
434                ));
435            }
436            rebuilt.transports
437        }
438    };
439    next_id += transports.len() as i64;
440
441    let tower_name = tower_name_from_filename(filename);
442    let has_wedding_hall = units.iter().any(|u| u.kind == Kind::WeddingHall);
443    let vip_visit_day = if has_wedding_hall && star < 6 {
444        minutes.div_euclid(1440) + 3
445    } else {
446        -1
447    };
448    let mut save = Map::new();
449    save.insert("version".into(), json!(save_version(lobby_height > 1)));
450    save.insert("seed".into(), json!(hash_seed(bytes)));
451    save.insert("money".into(), json!(money));
452    save.insert("star".into(), json!(star));
453    save.insert("minutes".into(), json!(minutes));
454    save.insert("mode".into(), json!("classic"));
455    save.insert("units".into(), Value::Array(unit_values));
456    save.insert(
457        "transports".into(),
458        Value::Array(transports.iter().map(OutTransport::to_json).collect()),
459    );
460    save.insert("nextId".into(), json!(next_id));
461    save.insert("towerName".into(), json!(tower_name));
462    save.insert("builtWeddingHall".into(), json!(has_wedding_hall));
463    if lobby_height > 1 {
464        save.insert("lobbyHeights".into(), json!([[1, lobby_height]]));
465    }
466    save.insert("evaluatedTower".into(), json!(star >= 6));
467    save.insert("vipVisitDay".into(), json!(vip_visit_day));
468    save.insert("vipFavorable".into(), json!(star >= 4));
469    save.insert("excavated".into(), json!(excavated));
470    if let Some((tile, floor)) = view_from_view_words(tdt.header.view_x, tdt.header.view_y) {
471        save.insert(
472            "view".into(),
473            json!({ "tile": jnum(tile), "floor": jnum(floor) }),
474        );
475    }
476
477    Ok(ParsedLegacyTower {
478        save: Value::Object(save),
479        warnings,
480        counts,
481        decode_stats,
482        decoded,
483        header_notes,
484        tower: tdt,
485    })
486}
