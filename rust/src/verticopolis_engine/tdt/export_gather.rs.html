1//! The gather pass of the `.TDT` export (`tdtExportGather.ts`): walk the
2//! serialized tower into per-floor tenant records, paving spans, header
3//! aggregate counts and a people census, without writing any bytes.
4//!
5//! The input is the save as JSON, which is any JavaScript value the
6//! TypeScript exporter would accept, so each field is read the way its
7//! TypeScript site reads it. There are three readers, and they are
8//! deliberately different:
9//!
10//! - `js_finite` is `Number.isFinite(v)`, which never coerces: a string, a
11//!   boolean or `null` is simply not a finite number. It serves every site
12//!   that guards with `Number.isFinite` (`money`, `lastQuarterMoney`, `star`,
13//!   the view's `tile` and `floor`, a transport's `x`, `bottom`, `top`,
14//!   `cars` and `carPositions`, a household's `residents`, and the extent of
15//!   a room that writes no record) and the `skipFloors` `Set`, whose
16//!   SameValueZero membership only a number can satisfy.
17//! - `js_num` is `Number(v)`, for the sites that do arithmetic on the raw
18//!   value and so coerce it: a unit's `occupants` (`Math.max(occupants, 1)`),
19//!   its `rent` (`rent - anchor` in `classFromRent`), and the clock's
20//!   `minutes` (`minutes % 1440`, read in `encoder.rs`).
21//! - `js_num_raw_site` serves the three unit fields the TypeScript gather
22//!   keeps raw: a unit's `floor`, `x` and `width`. It is the reading those
23//!   sites had before round 3 of the #881 review, kept on purpose; see its
24//!   own doc for where it parts from `Number()` (#884).
25//!
26//! A NaN in the live save crosses the WASM surface as JSON `null`, so `null`
27//! is the shape these sites see most: `Number.isFinite(null)` is false, while
28//! `Number(null)` is 0, which is why `js_finite` and the coercing readers
29//! cannot be swapped. Floors are keyed as doubles (`FloorKey`), the way the
30//! TypeScript `Map`s key a numeric floor. A floor that is not a number (a
31//! forged save only; the loader keeps floors finite) keys the TypeScript
32//! `Map` by its raw value and is coerced here by `js_num_raw_site`, so those
33//! saves write different bytes (#884).

68pub fn gather_tower(save: &Value) -> Result<GatheredTower, LegacyExportError> {
69    let Some(raw_units) = save.get("units").and_then(Value::as_array) else {
70        return Err(LegacyExportError("save.units is not an array".into()));
71    };
72    gather_units(raw_units, ground_lobby_height(save))
73}

75/// [`gather_tower`] over the save's unit entries as they come, each read on
76/// its own and dropped before the next, so the caller need not hold the
77/// `units` array whole (#1431). `lobby_height` is the ground lobby's height
78/// ([`ground_lobby_height`]): its lobby tiles rise through that many stories,
79/// which the file paves as lobby above them.
80pub fn gather_units<U: Borrow<Value>>(
81    raw_units: impl IntoIterator<Item = U>,
82    lobby_height: i64,
83) -> Result<GatheredTower, LegacyExportError> {
84    let mut tenants_by_tdt: IndexMap<FloorKey, Vec<OutTenant>> = IndexMap::new();
85    let mut extents: IndexMap<FloorKey, (f64, f64)> = IndexMap::new();
86    let mut covered_tiles: IndexMap<FloorKey, Vec<bool>> = IndexMap::new();
87    let mut retail_rows: Vec<(f64, i64)> = vec![];
88    let widen =
89        |extents: &mut IndexMap<FloorKey, (f64, f64)>, floor: f64, left: f64, right: f64| {
90            match extents.get_mut(&FloorKey::of(floor)) {
91                None => {
92                    extents.insert(FloorKey::of(floor), (left, right));
93                }
94                Some(e) => {
95                    e.0 = js_min(e.0, left);
96                    e.1 = js_max(e.1, right);
97                }
98            }
99        };
100    let push_tenant =
101        |tenants: &mut IndexMap<FloorKey, Vec<OutTenant>>, floor: f64, t: OutTenant| {
102            tenants
103                .entry(FloorKey::of(floor + TDT_FLOOR_OFFSET as f64))
104                .or_default()
105                .extend(split_theatre(t));
106        };
107
108    let mut counts = ExportCounts::default();
109    let mut header = ExportHeaderCounts::default();
110    let mut has_ground_lobby = false;
111    // The stories above the ground floor a tall ground lobby rises through.
112    let raised = 2..1 + lobby_height.clamp(1, MAX_LOBBY_HEIGHT);
113    let mut people_pop = 0.0;
114    let mut class_pop = [0.0; POPULATION_CLASSES];
115    let add_residents =
116        |people_pop: &mut f64, class_pop: &mut [f64; POPULATION_CLASSES], u: &GatheredRoom| {
117            let r = resident_count(u);
118            if r.is_finite() {
119                *people_pop += r;
120                if let Some(class) = population_class(u.kind) {
121                    class_pop[class] += r;
122                }
123            }
124        };
125
126    let mut rooms: Vec<GatheredRoom> = vec![];
127    for raw in raw_units {
128        let mut u = norm(raw.borrow())?;
129        if u.kind == Kind::Floor || u.kind == Kind::Lobby {
130            if u.kind == Kind::Lobby && u.floor == 1.0 && u.state != "fire" && u.state != "gutted" {
131                has_ground_lobby = true;
132            }
133            widen(&mut extents, u.floor, u.x, u.x + u.width);
134            if u.kind == Kind::Lobby && u.floor == 1.0 {
135                for story in raised.clone() {
136                    widen(&mut extents, story as f64, u.x, u.x + u.width);
137                }
138            }
139            continue;
140        }
141        let burned = u.state == "fire" || u.state == "gutted";
142        let stories = u.kind.floors() as f64;
143        let fits = fits_tdt_rows(u.kind, u.floor, stories);
144        let mappable = kind_tenant(u.kind).is_some() || part_stack(u.kind).is_some();
145        u.skip_reason = if !fits {
146            Some(SkipReason::OutOfRange)
147        } else if burned {
148            Some(SkipReason::Burned)
149        } else if !mappable {
150            Some(SkipReason::Unmappable)
151        } else {
152            None
153        };
154        u.emitted = !burned && fits && mappable;
155        let clamp = |v: f64| js_max(0.0, js_min(LOT_WIDTH as f64, v));
156        let mut left = 0.0;
157        let mut right = 0.0;
158        if u.emitted {
159            left = u.x;
160            right = u.x + u.width;
161        } else if u.coords_finite {
162            left = clamp(u.x);
163            right = clamp(u.x + u.width);
164        }
165        let emitted = u.emitted;
166        let (floor, kind) = (u.floor, u.kind);
167        rooms.push(u);
168        if !emitted && right <= left {
169            continue;
170        }
171        let mut fl = floor;
172        while fl < floor + kind.floors() as f64 {
173            widen(&mut extents, fl, left, right);
174            // A floor so large that `fl + 1` is `fl` would loop forever (as
175            // the TypeScript does); stop after the one story it can name.
176            if fl + 1.0 == fl {
177                break;
178            }
179            fl += 1.0;
180        }
181    }
182
183    for idx in 0..rooms.len() {
184        let u = &rooms[idx];
185        if !u.emitted {
186            match u.skip_reason {
187                Some(SkipReason::OutOfRange) => counts.out_of_range += 1,
188                Some(SkipReason::Burned) => counts.burned_out += 1,
189                _ => {}
190            }
191            continue;
192        }
193        let construction = u.state == "construction";
194        let tenanted = u.state == "occupied" || u.state == "moving_in" || u.state == "vacating";
195        let mut status: i64 = 0;
196        if !construction {
197            if (u.kind == Kind::Office || u.kind == Kind::Condo) && tenanted {
198                status = 1;
199                counts.occupied += 1;
200                add_residents(&mut people_pop, &mut class_pop, u);
201            } else if u.kind.is_commercial() && u.kind.facility().population > 0 {
202                if is_present(&u.state) {
203                    add_residents(&mut people_pop, &mut class_pop, u);
204                }
205            } else if u.kind.is_hotel() {
206                // The file's infested bit (`HOTEL_INFESTED_FLAG`) is not our
207                // infested state: the original clears it from a vacant or dirty
208                // room after about a minute of Fast Mode (PC run, #1629), so an infested room
209                // goes out as the dirty room the original can work with.
210                if u.state == "dirty" || u.state == "infested" {
211                    status = HOTEL_DIRTY_FLAG;
212                } else if u.state == "asleep" {
213                    let guests = js_min(js_max(u.occupants, 1.0), HOTEL_OCCUPANT_MASK as f64);
214                    status = HOTEL_ASLEEP_FLAG | super::byte_writer::to_int32(guests);
215                } else if tenanted || u.ever_occupied {
216                    // A room that has been rented and sits clean and empty.
217                    // The original's own saves write it as stay state 3; a
218                    // bare 1 (no stay state) is read as a sleeping guest and
219                    // rewritten to 0x10 when the game saves.
220                    status = HOTEL_STAY_VACANT;
221                }
222                if status != 0 {
223                    counts.hotel_states += 1;
224                    add_residents(&mut people_pop, &mut class_pop, u);
225                    // The finance block counts the guests in the rooms, so a
226                    // dirty or booked room, which the census counts for the
227                    // people block, holds none there.
228                    if let Some(class) = population_class(u.kind) {
229                        class_pop[class] -= resident_count(u);
230                        if u.state == "asleep" {
231                            class_pop[class] +=
232                                js_min(js_max(u.occupants, 1.0), HOTEL_OCCUPANT_MASK as f64);
233                        }
234                    }
235                }
236            }
237            if (u.kind == Kind::Office || u.kind == Kind::Condo) && !tenanted && u.ever_occupied {
238                counts.vacancy_history_lost += 1;
239            }
240        } else {
241            counts.construction += 1;
242        }
243        let rent_defined = u.rent.is_some();
244        let rent_num = u.rent.as_ref().map(|v| js_num(Some(v)));
245        let rent_class = if u.no_rate && is_priced(u.kind) {
246            4
247        } else {
248            class_from_rent(u.kind, if rent_defined { rent_num } else { None })
249        };
250        if let Some(band) = rent_config(u.kind) {
251            if !u.no_rate {
252                let effective = match &u.rent {
253                    None | Some(Value::Null) => band.default,
254                    Some(v) => js_num(Some(v)),
255                };
256                let back = rent_from_class(u.kind, rent_class);
257                if back.is_some_and(|b| b != effective) {
258                    counts.rents_snapped += 1;
259                }
260            }
261        }
262        if u.label
263            .as_deref()
264            .is_some_and(|l| !l.is_empty() && l != u.kind.facility().name)
265        {
266            counts.names_dropped += 1;
267        }
268        if u.kind == Kind::Parking {
269            counts.parking_stalls += 1;
270        }
271        match u.kind {
272            Kind::Recycling => header.recycling += 1,
273            Kind::Shop | Kind::Restaurant | Kind::FastFood => header.commercial += 1,
274            Kind::Security => header.security += 1,
275            Kind::Medical => header.medical += 1,
276            Kind::PartyHall | Kind::Cinema => header.hall_cinema += 1,
277            _ => {}
278        }
279
280        if let Some(stack) = part_stack(u.kind) {
281            if u.kind == Kind::WeddingHall {
282                // The cathedral stands on floor 100 with its bottom part and rises
283                // four floors above it, the top part last.
284                for (i, &part) in stack.iter().rev().enumerate() {
285                    let fl = u.floor + i as f64;
286                    widen(&mut extents, fl, u.x, u.x + u.width);
287                    push_tenant(
288                        &mut tenants_by_tdt,
289                        fl,
290                        OutTenant {
291                            left: u.x,
292                            right: u.x + u.width,
293                            type_id: if construction { -part } else { part },
294                            status: 0,
295                            rent_class,
296                            retail_row: None,
297                            draw: 0,
298                            venue: None,
299                        },
300                    );
301                }
302            } else {
303                for (i, &part) in stack.iter().enumerate() {
304                    push_tenant(
305                        &mut tenants_by_tdt,
306                        u.floor + i as f64,
307                        OutTenant {
308                            left: u.x,
309                            right: u.x + u.width,
310                            type_id: if construction { -part } else { part },
311                            status: 0,
312                            rent_class,
313                            retail_row: None,
314                            draw: exported_film_draw(u),
315                            venue: venue_half(u, idx, i == 1, construction),
316                        },
317                    );
318                }
319            }
320            counts.rooms += 1;
321            continue;
322        }
323
324        let Some(id) = kind_tenant(u.kind) else {
325            continue;
326        };
327        let sub_idx: i64 = match (&u.subtype, u.kind.subtype_list()) {
328            (Some(name), Some(list)) => list
329                .iter()
330                .position(|s| s == name)
331                .map(|i| i as i64)
332                .unwrap_or(-1),
333            _ => -1,
334        };
335        let retail = matches!(u.kind, Kind::Shop | Kind::FastFood | Kind::Restaurant);
336        let retail_row =
337            (retail && retail_rows.len() < TDT_RETAIL_SLOTS).then_some(retail_rows.len() as i64);
338        push_tenant(
339            &mut tenants_by_tdt,
340            u.floor,
341            OutTenant {
342                left: u.x,
343                right: u.x + u.width,
344                type_id: if construction { -id } else { id },
345                status,
346                rent_class,
347                retail_row,
348                draw: 0,
349                venue: None,
350            },
351        );
352        if retail_row.is_some() {
353            retail_rows.push((u.floor + TDT_FLOOR_OFFSET as f64, sub_idx.max(0)));
354        } else if retail {
355            counts.retail_rows_dropped += 1;
356        }
357        counts.rooms += 1;
358    }
359
360    // Coverage for the paving pass, from the records actually emitted.
361    let clamp_tile = |v: f64| -> usize {
362        let v = if v.is_finite() { v } else { 0.0 };
363        v.floor().min(LOT_WIDTH as f64).max(0.0) as usize
364    };
365    for (tdt, records) in &tenants_by_tdt {
366        let floor = tdt.value() - TDT_FLOOR_OFFSET as f64;
367        for r in records {
368            let lo = clamp_tile(r.left);
369            let hi = clamp_tile(r.right).max(lo);
370            if hi == lo {
371                continue;
372            }
373            let set = covered_tiles
374                .entry(FloorKey::of(floor))
375                .or_insert_with(|| vec![false; LOT_WIDTH as usize + 1]);
376            for cell in &mut set[lo..hi] {
377                *cell = true;
378            }
379        }
380    }
381
382    // Paving spans: each floor's extent minus the tiles under a record, one
383    // record per contiguous run; lobby floors type 24, others type 0.
384    const TDT_LOBBY_TYPE: i64 = 24;
385    const TDT_FLOOR_TYPE: i64 = 0;
386    let extent_list: Vec<(FloorKey, (f64, f64))> = extents.iter().map(|(k, v)| (*k, *v)).collect();
387    for (key, ext) in extent_list {
388        let floor = key.value();
389        if !fits_tdt_rows(Kind::Floor, floor, 1.0) {
390            continue;
391        }
392        // A tall ground lobby's raised stories are paved as lobby too: our
393        // reading of the format, where the game writes type 24 on every
394        // tile of a lobby story (to confirm on the PC, #308).
395        let lobby = floor.fract() == 0.0
396            && (is_lobby_floor(floor as i64) || raised.contains(&(floor as i64)));
397        let covered = covered_tiles.get(&key);
398        let lo = clamp_tile(ext.0);
399        let hi = clamp_tile(ext.1).max(lo);
400        let mut run_start: Option<usize> = None;
401        let flush = |run_start: &mut Option<usize>,
402                     end: usize,
403                     tenants: &mut IndexMap<FloorKey, Vec<OutTenant>>| {
404            let Some(start) = *run_start else {
405                return;
406            };
407            push_tenant(
408                tenants,
409                floor,
410                OutTenant {
411                    left: start as f64,
412                    right: end as f64,
413                    type_id: if lobby {
414                        TDT_LOBBY_TYPE
415                    } else {
416                        TDT_FLOOR_TYPE
417                    },
418                    status: if lobby { 0 } else { 2 },
419                    rent_class: 4,
420                    retail_row: None,
421                    draw: 0,
422                    venue: None,
423                },
424            );
425            *run_start = None;
426        };
427        for x_tile in lo..hi {
428            if covered.is_some_and(|set| set[x_tile]) {
429                flush(&mut run_start, x_tile, &mut tenants_by_tdt);
430            } else if run_start.is_none() {
431                run_start = Some(x_tile);
432            }
433        }
434        flush(&mut run_start, hi, &mut tenants_by_tdt);
435    }
436
437    Ok(GatheredTower {
438        tenants_by_tdt,
439        extents,
440        retail_rows,
441        counts,
442        header,
443        ground_lobby_height: if has_ground_lobby {
444            lobby_height.clamp(1, MAX_LOBBY_HEIGHT)
445        } else {
446            0
447        },
448        people_pop,
449        class_pop,
450        rooms,
451    })
452}

457/// The venue table's half for a finished hall or theatre (#1932), a theatre
458/// with no film age counting as a film on its first day; `None` for every
459/// other record.
460fn venue_half(
461    u: &GatheredRoom,
462    room: usize,
463    upper: bool,
464    construction: bool,
465) -> Option<OutVenueHalf> {
466    let film_days = match u.kind {
467        // A theater built today has shown its film no mornings yet.
468        Kind::Cinema => Some(u.film_days.unwrap_or(0)),
469        Kind::PartyHall => None,
470        _ => return None,
471    };
472    (!construction).then_some(OutVenueHalf {
473        room,
474        upper,
475        film_days,
476        film_classic: u.film_classic,
477        auditorium: false,
478    })
479}

481/// The draw byte a theatre writes (#1932): the draw its film had last night,
482/// the age before this morning's step (a theatre that has not seen a morning
483/// shows a new film's draw, and a classic's tops out at 1). Every other room
484/// writes 0.
485fn exported_film_draw(u: &GatheredRoom) -> i64 {
486    if u.kind != Kind::Cinema {
487        return 0;
488    }
489    showing_draw((u.film_days.unwrap_or(0) - 1).max(0), u.film_classic)
