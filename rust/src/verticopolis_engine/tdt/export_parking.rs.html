1//! The parking side of a `.TDT` export (#307): the stall table in the
2//! routing tail, each stall record's index into it, the connected list and
3//! the ramp column in the header.
4//!
5//! A run of the original on the owner's machine (2026-11-15) pinned the
6//! layout: the game rebuilds its connected list from the stall table, so a
7//! stall the table does not name is never connected. Record `i` of the table
8//! is the stall's floor index and logical unit id, then four zero bytes; an
9//! empty record is `FF` then five zeros. A stall's tenant record names its
10//! table record in bytes 6 and 7, low byte first. The connected list holds
11//! table record indexes, ascending, count first. Header bytes 36 and 37 hold
12//! the x of the ramp on B1 (the parking column), or `FFFF` with none.
13
14use super::byte_writer::ByteWriter;
15
16/// Stall and ramp type ids in a tenant record (negative while built).
17const STALL_TYPE: i64 = 11;
18const RAMP_TYPE: i64 = 44;
19/// The TDT floor index of B1, the floor whose ramp starts the column.
20const B1_FLOOR_INDEX: usize = 9;
21/// The header word holding the column's x.
22pub const TDT_HEADER_PARKING_COLUMN_AT: usize = 0x24;
23/// The header word's value with no ramp on B1.
24const NO_PARKING_COLUMN: i64 = 0xffff;
25/// Where the stall table starts in the routing tail written after the
26/// stairs table.
27pub const STALL_TABLE_IN_ROUTING_TAIL: usize = 4_012;
28/// The stall table's records, and so the most stalls a file can name.
29pub const TDT_STALL_SLOTS: usize = 512;
30const STALL_RECORD_SIZE: usize = 6;
31/// The floor byte of an empty stall record.
32const EMPTY_STALL_RECORD: u8 = 0xff;
33/// Where a stall's tenant record holds its table record index.
34const TABLE_INDEX_IN_TENANT_RECORD: usize = 6;
35
36/// One tenant on a floor as the encoder orders them: its clamped left and
37/// right tiles and its type id.
38pub type TenantSpan = (f64, f64, i64);
39
40fn is_stall(type_id: i64) -> bool {
41    type_id.abs() == STALL_TYPE
42}
43
44fn is_ramp(type_id: i64) -> bool {
45    type_id.abs() == RAMP_TYPE
46}
47
48/// For each of a floor's tenants, in order of their left tile, whether it is
49/// a stall chained to a ramp on its own floor: each piece touches the next,
50/// outward from a ramp, in both directions.
51pub fn linked_to_a_ramp(spans: &[TenantSpan]) -> Vec<bool> {
52    let parking: Vec<usize> = (0..spans.len())
53        .filter(|&i| is_stall(spans[i].2) || is_ramp(spans[i].2))
54        .collect();
55    let mut linked = vec![false; spans.len()];
56    for &i in &parking {
57        linked[i] = is_ramp(spans[i].2);
58    }
59    let touches = |a: usize, b: usize| spans[a].1 >= spans[b].0;
60    for pair in parking.windows(2) {
61        let (a, b) = (pair[0], pair[1]);
62        if !linked[b] && linked[a] && touches(a, b) {
63            linked[b] = true;
64        }
65    }
66    for pair in parking.windows(2).rev() {
67        let (a, b) = (pair[0], pair[1]);
68        if !linked[a] && linked[b] && touches(a, b) {
69            linked[a] = true;
70        }
71    }
72    for &i in &parking {
73        linked[i] &= is_stall(spans[i].2);
74    }
75    linked
76}

78/// A stall as the encoder placed it, in the order it wrote them.
79struct PlacedStall {
80    floor: usize,
81    id: usize,
82    record_at: usize,
83    connected: bool,
84}
85
86/// The stalls and the column the encoder met while writing the floors.
87#[derive(Default)]
88pub struct ParkingExport {
89    stalls: Vec<PlacedStall>,
90    column: Option<f64>,
91}
92
93impl ParkingExport {
94    /// Note a tenant the encoder just wrote at `record_at` on TDT floor
95    /// `floor`. A stall without a logical id stays out of the table.
96    pub fn place(
97        &mut self,
98        floor: usize,
99        span: TenantSpan,
100        id: Option<usize>,
101        record_at: usize,
102        connected: bool,
103    ) {
104        let (left, _, type_id) = span;
105        if is_ramp(type_id) && floor == B1_FLOOR_INDEX {
106            // With two ramps on B1 the game names the rightmost.
107            self.column = Some(self.column.map_or(left, |x| x.max(left)));
108        }
109        let Some(id) = id.filter(|_| is_stall(type_id)) else {
110            return;
111        };
112        self.stalls.push(PlacedStall {
113            floor,
114            id,
115            record_at,
116            connected,
117        });
118    }
119
120    /// The stalls the table holds, by record: the connected ones first, so
121    /// a tower with more stalls than records keeps every connected stall it
122    /// can (#307), then the rest, each group in the order written.
123    fn table(&self) -> Vec<&PlacedStall> {
124        let (connected, loose): (Vec<&PlacedStall>, Vec<&PlacedStall>) =
125            self.stalls.iter().partition(|s| s.connected);
126        connected
127            .into_iter()
128            .chain(loose)
129            .take(TDT_STALL_SLOTS)
130            .collect()
131    }
132
133    /// Header bytes 36 and 37: the column's x, or `FFFF` with no B1 ramp.
134    pub fn header_column(&self) -> i64 {
135        self.column.map_or(NO_PARKING_COLUMN, |x| x as i64)
136    }
137
138    /// The parking block: the connected stalls' table indexes, count first.
139    /// The connected stalls lead the table, so the list is its first
140    /// records in order.
141    pub fn write_connected_list(&self, w: &mut ByteWriter) {
142        let connected = self.table().iter().filter(|s| s.connected).count();
143        w.u16(connected as i64);
144        for slot in 0..TDT_STALL_SLOTS {
145            w.u16(if slot < connected { slot as i64 } else { 0 });
146        }
147    }
148
149    /// Fill the stall table of the routing tail starting at `tail_at`, and
150    /// point each stall's tenant record at its record. A stall past the
151    /// table's last record keeps the 0 the encoder wrote.
152    pub fn write_stall_table(&self, w: &mut ByteWriter, tail_at: usize) {
153        let table_at = tail_at + STALL_TABLE_IN_ROUTING_TAIL;
154        let table = self.table();
155        for slot in 0..TDT_STALL_SLOTS {
156            let at = table_at + slot * STALL_RECORD_SIZE;
157            let record: [u8; STALL_RECORD_SIZE] = match table.get(slot) {
158                Some(s) => [s.floor as u8, s.id as u8, 0, 0, 0, 0],
159                None => [EMPTY_STALL_RECORD, 0, 0, 0, 0, 0],
160            };
161            for (k, byte) in record.into_iter().enumerate() {
162                w.set_u8(at + k, byte as i64);
163            }
164        }
165        for (i, s) in table.iter().enumerate() {
166            w.set_u16(s.record_at + TABLE_INDEX_IN_TENANT_RECORD, i as i64);
167        }
168    }
169}
