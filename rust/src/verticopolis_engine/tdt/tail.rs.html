15/// Does the trailing routing region begin exactly at `at`?
16fn routing_tail_starts_at(bytes: &[u8], at: usize) -> bool {
17    const RUN: usize = 64;
18    if at == 0 || at + RUN > bytes.len() {
19        return false;
20    }
21    if bytes[at - 1] == 0xff {
22        return false;
23    }
24    bytes[at..at + RUN].iter().all(|&b| b == 0xff)
25}

27/// Does a structure whose position we know sit where a table ending at `end`
28/// puts it: the stairs table where the game keeps it, where our exports before
29/// generation 3 put it (with the 4-byte gap or, earlier still, without), or the
30/// routing region one stairs table further on?
31fn table_end_anchored(bytes: &[u8], end: usize) -> bool {
32    let distances = [
33        TDT_ELEVATORS_TO_STAIRS,
34        TDT_LEGACY_ELEVATORS_TO_STAIRS,
35        TDT_LEGACY_NO_GAP_ELEVATORS_TO_STAIRS,
36    ];
37    distances.into_iter().any(|distance| {
38        let stairs = end + distance;
39        stairs_table_starts_at(bytes, stairs)
40            || routing_tail_starts_at(bytes, stairs + TDT_STAIR_SLOTS * TDT_STAIR_RECORD_SIZE)
41    })
42}

44/// How a built elevator slot's payload is sized; see `tdtTail.ts`.
45#[derive(Clone, Copy, PartialEq, Eq, Debug)]
46enum PayloadLayout {
47    /// Express shafts carry an entry per eligible floor, the rest per spanned
48    /// floor: what SimTower writes.
49    ExpressEligible,
50    ExpressServiced,
51    Spanned,
52    Serviced,
53}

55/// Stops recorded in a slot's serviced bitmap, counting only floors the
56/// shaft actually spans.
57fn stop_count(serviced: &[u8], bottom_floor: i64, top_floor: i64) -> i64 {
58    let from = bottom_floor.max(0);
59    let to = top_floor.min(serviced.len() as i64 - 1);
60    let mut stops = 0;
61    let mut i = from;
62    while i <= to {
63        if serviced[i as usize] != 0 {
64            stops += 1;
65        }
66        i += 1;
67    }
68    stops
69}

71fn payload_size(
72    layout: PayloadLayout,
73    type_id: i64,
74    bottom_floor: i64,
75    top_floor: i64,
76    serviced: &[u8],
77) -> i64 {
78    // The table walk checks the span before sizing, so the helpers cannot
79    // refuse here; a refusal would be a walker bug.
80    let size = match layout {
81        PayloadLayout::ExpressEligible if type_id == TDT_ELEVATOR_TYPE_EXPRESS => {
82            express_payload_size(bottom_floor, top_floor)
83        }
84        PayloadLayout::ExpressEligible => built_shaft_payload_size(bottom_floor, top_floor),
85        PayloadLayout::ExpressServiced => built_shaft_payload_size_for(
86            type_id,
87            bottom_floor,
88            top_floor,
89            stop_count(serviced, bottom_floor, top_floor),
90        ),
91        PayloadLayout::Spanned => built_shaft_payload_size(bottom_floor, top_floor),
92        PayloadLayout::Serviced => {
93            built_shaft_payload_size(1, stop_count(serviced, bottom_floor, top_floor).max(1))
94        }
95    };
96    size.expect("elevator span checked before sizing")
97}

105const CUT_SHORT: &str = "The elevator table is cut short, so elevators were rebuilt from the floor layout and the save's stairways couldn't be read.";
106const RESYNCED: &str = "Some of the save's elevators are stored in a layout we don't fully understand, so their shafts were found by searching the file. Check that your elevators look right.";
107const NOT_DOCUMENTED: &str = "The elevator table doesn't match the documented layout, so elevators were rebuilt from the floor layout and the save's stairways couldn't be read.";

109/// Walk the 24-slot elevator table, sizing built payloads per `layout`. With
110/// `resync`, a header that does not parse after a built slot is not the end of
111/// the walk: the previous shaft's payload was sized wrong, so the next
112/// plausible header is searched for past that shaft's smallest possible
113/// payload and the walk goes on from there.
114fn read_elevator_table(r: &mut ByteReader, layout: PayloadLayout, resync: bool) -> TableWalk {
115    r.enter_block("elevator table");
116    let mut elevators = vec![];
117    let mut prev_header: Option<usize> = None;
118    let mut slot_after_prev = 0;
119    let mut slot = 0;
120    // Every read below is guarded by the size check first, so the reader
121    // cannot run short mid-slot; `expect` marks that invariant.
122    while slot < TDT_ELEVATOR_SLOTS {
123        if r.remaining() < TDT_ELEVATOR_HEADER_SIZE {
124            return TableWalk {
125                elevators: None,
126                warning: Some(CUT_SHORT),
127                end: r.offset(),
128            };
129        }
130        let header_at = r.offset();
131        let used = r.u8().expect("header sized");
132        let type_id = r.u8().expect("header sized");
133        let capacity = r.u8().expect("header sized");
134        let cars = r.u8().expect("header sized");
135        r.skip(56).expect("header sized");
136        r.skip(2).expect("header sized");
137        let x = r.u16().expect("header sized");
138        let top_floor = r.u8().expect("header sized");
139        let bottom_floor = r.u8().expect("header sized");
140        let serviced = r.bytes(TDT_FLOOR_COUNT as usize).expect("header sized");
141        let mut car_homes = Vec::with_capacity(8);
142        for _ in 0..8 {
143            car_homes.push(r.u8().expect("header sized"));
144        }
145        if used == 0 {
146            slot += 1;
147            continue;
148        }
149        if used != 1
150            || type_id > 2
151            || !(1..=8).contains(&cars)
152            || top_floor < bottom_floor
153            || top_floor >= TDT_FLOOR_COUNT
154        {
155            if let Some(prev) = prev_header.filter(|_| resync) {
156                // The next shaft starts after this one's smallest possible
157                // payload and no later than its largest (a whole-tower span).
158                let past_smallest_payload = prev + TDT_ELEVATOR_HEADER_SIZE + MIN_BUILT_PAYLOAD;
159                let within_largest_payload = prev + TDT_ELEVATOR_HEADER_SIZE + MAX_BUILT_PAYLOAD;
160                // The stretch up to a header is the previous shaft's payload
161                // (whole floors) and any empty slots between the two. Only one
162                // count of empty slots makes that add up, so a header that
163                // leaves none is not the next slot.
164                let gap_fits =
165                    |found: usize| empty_slots_in_gap(found - prev, slot_after_prev).is_some();
166                if let Some(found) = find_next_header(
167                    r.raw(),
168                    past_smallest_payload,
169                    within_largest_payload,
170                    gap_fits,
171                ) {
172                    let empty_slots =
173                        empty_slots_in_gap(found - prev, slot_after_prev).expect("accepted above");
174                    slot = slot_after_prev + empty_slots;
175                    r.seek(found).expect("found inside the file");
176                    continue;
177                }
178            }
179            return TableWalk {
180                elevators: None,
181                warning: Some(NOT_DOCUMENTED),
182                end: r.offset(),
183            };
184        }
185        let payload = payload_size(layout, type_id, bottom_floor, top_floor, &serviced) as usize;
186        if r.remaining() < payload {
187            return TableWalk {
188                elevators: None,
189                warning: Some(CUT_SHORT),
190                end: r.offset(),
191            };
192        }
193        r.skip(payload).expect("payload sized");
194        prev_header = Some(header_at);
195        slot += 1;
196        slot_after_prev = slot;
197        elevators.push(TdtElevator {
198            type_id,
199            capacity,
200            cars,
201            x,
202            top_floor,
203            bottom_floor,
204            serviced,
205            car_homes,
206            record: header_at..r.offset(),
207        });
208    }
209    TableWalk {
210        elevators: Some(elevators),
211        warning: None,
212        end: r.offset(),
213    }
214}

216/// Which payload layout does this file use? Decided once for the whole file;
217/// see `chooseLayout` in `tdtTail.ts` for the reasoning behind each step.
218fn choose_layout(bytes: &[u8], table_start: usize) -> PayloadLayout {
219    match stamped_generation(bytes) {
220        Some(TDT_STAMP_SPANNED_GENERATION) => return PayloadLayout::Spanned,
221        Some(TDT_STAMP_ELIGIBLE_GENERATION | TDT_STAMP_GENERATION) => {
222            return PayloadLayout::ExpressEligible
223        }
224        _ => {}
225    }
226    let walk = |layout: PayloadLayout| -> TableWalk {
227        let mut r = ByteReader::new(bytes);
228        r.skip(table_start).expect("table start inside the file");
229        read_elevator_table(&mut r, layout, false)
230    };
231    let anchored = |end: usize| table_end_anchored(bytes, end);
232    // A layout that sizes a payload wrong lands later headers on payload bytes,
233    // which read as empty slots, so the walk that finds the most shafts has
234    // the right layout. Ties go to the order below.
235    let layouts = [
236        PayloadLayout::ExpressEligible,
237        PayloadLayout::ExpressServiced,
238        PayloadLayout::Spanned,
239        PayloadLayout::Serviced,
240    ];
241    let walks = layouts.map(&walk);
242    // The legacy serviced-stops layout is read only when its table ends where
243    // the rest of the file says it should; it fits too many broken files.
244    let shafts = |layout: PayloadLayout, w: &TableWalk| {
245        let believable = layout != PayloadLayout::Serviced || anchored(w.end);
246        w.elevators.as_ref().filter(|_| believable).map(Vec::len)
247    };
248    let Some(most) = layouts
249        .iter()
250        .zip(walks.iter())
251        .filter_map(|(layout, w)| shafts(*layout, w))
252        .max()
253    else {
254        return PayloadLayout::ExpressServiced;
255    };
256    let ties: Vec<(PayloadLayout, &TableWalk)> = layouts
257        .into_iter()
258        .zip(walks.iter())
259        .filter(|(layout, w)| shafts(*layout, w) == Some(most))
260        .collect();
261    if let Some((layout, _)) = ties.iter().find(|(_, w)| anchored(w.end)) {
262        return *layout;
263    }
264    let ambiguous = |w: &TableWalk| {
265        w.elevators.as_ref().is_some_and(|es| {
266            es.iter().any(|e| {
267                stop_count(&e.serviced, e.bottom_floor, e.top_floor)
268                    != e.top_floor - e.bottom_floor + 1
269            })
270        })
271    };
272    ties.iter()
273        .find(|(layout, w)| *layout != PayloadLayout::ExpressServiced || !ambiguous(w))
274        .or(ties.first())
275        .map(|(layout, _)| *layout)
276        .expect("a walk found the most shafts")
277}

279/// The per-class people in the finance block, trusted only when they add up to
280/// the total beside them and the total is not zero: a file from before the
281/// block was filled holds zeros, which say nothing.
282fn finance_population(finance: &[u8]) -> Option<[i64; TDT_POPULATION_CLASSES]> {
283    let word = |at: usize| {
284        i64::from(u32::from_le_bytes(
285            finance[at..at + 4].try_into().expect("four bytes"),
286        ))
287    };
288    let mut pops = [0; TDT_POPULATION_CLASSES];
289    for (class, pop) in pops.iter_mut().enumerate() {
290        *pop = word(TDT_FINANCE_POPULATION_AT + class * 4);
291    }
292    let total = word(TDT_FINANCE_TOTAL_AT);
293    (total > 0 && pops.iter().sum::<i64>() == total).then_some(pops)
294}

296/// `walkTolerantTail`: the blocks after the floor map, in file order.
297pub fn walk_tolerant_tail(r: &mut ByteReader) -> TdtTail {
298    let mut tail = TdtTail::default();
299
300    r.enter_block("people block");
301    if r.remaining() < 4 {
302        tail.warnings.push(
303            "The file ends right after the floor map, so no people or transport data is present."
304                .into(),
305        );
306        return tail;
307    }
308    tail.people_at = Some(r.offset());
309    let people_count = r.u32().expect("four bytes remain");
310    if people_count > TDT_MAX_PEOPLE {
311        tail.warnings.push(
312            "The people table claims an impossible head count, so the rest of the file was skipped."
313                .into(),
314        );
315        return tail;
316    }
317    let people_bytes = people_count as usize * TDT_PERSON_RECORD_SIZE;
318    if r.remaining() < people_bytes {
319        tail.warnings.push(
320            "The people table runs past the end of the file, so the rest of the file was skipped."
321                .into(),
322        );
323        tail.people_count = Some(people_count);
324        return tail;
325    }
326    r.skip(people_bytes).expect("people block sized");
327    tail.people_count = Some(people_count);
328
329    r.enter_block("retail table");
330    if r.remaining() < TDT_RETAIL_SLOTS * TDT_RETAIL_RECORD_SIZE {
331        tail.warnings.push(
332            "The retail table is missing or cut short, so the save's transport data couldn't be reached."
333                .into(),
334        );
335        return tail;
336    }
337    let mut occupied = 0;
338    for at in 0..TDT_RETAIL_SLOTS {
339        let row: Vec<u8> = (0..TDT_RETAIL_RECORD_SIZE)
340            .map(|_| r.u8().expect("retail table sized") as u8)
341            .collect();
342        if row[0] != 0xff {
343            occupied += 1;
344            tail.retail.push(TdtRetailRow {
345                row: at,
346                floor: row[0],
347                slot: row[1],
348                served: row[RETAIL_SERVED_BYTE],
349                style: (row[RETAIL_CONSTANT_BYTE] == RETAIL_ROW_CONSTANT)
350                    .then_some(row[RETAIL_STYLE_BYTE]),
351            });
352        }
353    }
354    tail.retail_rows = Some(occupied);
355
356    let table_start = r.offset();
357    let layout = choose_layout(r.raw(), table_start);
358    let mut table = read_elevator_table(r, layout, false);
359    if table.elevators.is_none() && table.warning == Some(NOT_DOCUMENTED) {
360        // Some real saves size a shaft's payload by a rule none of the three
361        // layouts has. Walk again, finding each header by its shape.
362        r.seek(table_start).expect("table start inside the file");
363        let retry = read_elevator_table(r, layout, true);
364        if retry.elevators.is_some() {
365            tail.warnings.push(RESYNCED.to_string());
366            table = retry;
367        }
368    }
369    let Some(elevators) = table.elevators else {
370        tail.warnings
371            .push(table.warning.unwrap_or(CUT_SHORT).to_string());
372        return tail;
373    };
374    tail.elevators = Some(elevators);
375    let after_elevators = r.offset();
376
377    // After the elevators come the statistic words, the finance block, two
378    // blocks nobody has decoded, the parking block, the clinic list and the
379    // stairs table. The people counts sit at the start of the finance block.
380    let blocks_at = after_elevators;
381    let raw = r.raw();
382    let finance_at = blocks_at + TDT_STATS_SIZE;
383    let mut stairs_search_from = after_elevators;
384    r.enter_block("finance block");
385    let people_end = finance_at + TDT_FINANCE_TOTAL_AT + 4;
386    if raw.len() >= people_end {
387        stairs_search_from = people_end;
388        tail.finance_population = finance_population(&raw[finance_at..people_end]);
389    } else {
390        tail.warnings.push(
391            "The save ends right after its elevators, so its finance and parking data could not be read."
392                .into(),
393        );
394    }
395
396    r.enter_block("stairs table");
397    let generation = stamped_generation(raw);
398    let (stairs_at, stairs) = locate_stairs_table(raw, blocks_at, stairs_search_from, generation);
399    tail.stairs = Some(stairs);
400    if let Some(at) = stairs_at {
401        tail.venues = super::venues::read_venue_table(raw, at);
402    }
403
404    r.enter_block("parking block");
405    // Our exports before generation 3 keep the parking count right after a
406    // 132-byte finance block; the game's files keep it after the statistics,
407    // finance and the two blocks nobody has decoded. An unstamped file is read
408    // the older way only when the game's place holds nothing and the older one
409    // holds a count a parking block could (the older place is a finance line in
410    // a game save, and a game save has no stalls there when the line is large).
411    let game_parking_at =
412        finance_at + TDT_FINANCE_SIZE + TDT_UNKNOWN_FLAGS_SIZE + TDT_UNKNOWN_BYTES_SIZE;
413    let count_at = |at: usize| {
414        (raw.len() >= at + 2).then(|| usize::from(u16::from_le_bytes([raw[at], raw[at + 1]])))
415    };
416    let older_export = match generation {
417        Some(g) => g < TDT_STAMP_GENERATION,
418        None => {
419            count_at(game_parking_at) == Some(0)
420                && count_at(blocks_at + TDT_FINANCE_SIZE).is_some_and(|n| (1..=512).contains(&n))
421        }
422    };
423    let parking_at = if older_export {
424        blocks_at + TDT_FINANCE_SIZE
425    } else {
426        game_parking_at
427    };
428    if raw.len() >= parking_at + TDT_PARKING_SIZE && raw.len() >= people_end {
429        tail.parking_connected = Some(i64::from(u16::from_le_bytes([
430            raw[parking_at],
431            raw[parking_at + 1],
432        ])));
433    } else if raw.len() >= people_end {
434        tail.warnings.push(
435            "The save ends inside its parking data, so the connected-stall count could not be read."
436                .into(),
437        );
438    }
439    tail
440}

442/// Clinic ids are small slot numbers; a list holding larger words is not one.
443const CLINIC_ID_BOUND: usize = 16;
444
445/// Does the 22-byte clinic list that ends at `table` agree with the clinic
446/// count in the header? It does in every save the game wrote that has a
447/// clinic, which is how a table in the game's position is told from a table
448/// our older exports wrote 160 bytes earlier.
449fn clinic_list_matches_header(raw: &[u8], table: usize) -> bool {
450    if raw.len() < TDT_HEADER_CLINIC_COUNT_AT + 2
451        || table < TDT_CLINIC_LIST_SIZE
452        || table > raw.len()
453    {
454        return false;
455    }
456    let word = |at: usize| usize::from(u16::from_le_bytes([raw[at], raw[at + 1]]));
457    let list = table - TDT_CLINIC_LIST_SIZE;
458    let count = word(list);
459    (1..=TDT_CLINIC_MAX).contains(&count)
460        && count == word(TDT_HEADER_CLINIC_COUNT_AT)
461        && (0..TDT_CLINIC_MAX).all(|i| word(list + 2 + 2 * i) < CLINIC_ID_BOUND)
462}

464/// The stairs table and where it starts. A file of ours from generation 3 on, or one whose clinic
465/// list matches its header, keeps the table where the game does, and a table
466/// that fits there is taken as it stands. Anything else is found by the
467/// signature scan, which also reads the tables our older exports wrote 160
468/// bytes earlier (the game's position would read their end-of-parking bytes
469/// and clinic list as a flight, so it is not tried for them).
470fn locate_stairs_table(
471    raw: &[u8],
472    blocks_at: usize,
473    scan_from: usize,
474    generation: Option<i64>,
475) -> (Option<usize>, Vec<TdtStair>) {
476    let game = blocks_at + TDT_ELEVATORS_TO_STAIRS;
477    let at_game_offset =
478        generation == Some(TDT_STAMP_GENERATION) || clinic_list_matches_header(raw, game);
479    if at_game_offset {
480        if let Some(stairs) = stairs_in_table_at(raw, game) {
481            return (Some(game), stairs);
482        }
483    }
484    locate_stairs_at(raw, scan_from).map_or((None, vec![]), |(at, stairs)| (Some(at), stairs))
485}
