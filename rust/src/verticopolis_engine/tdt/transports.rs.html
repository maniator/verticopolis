16/// A transport as the importer writes it into the save.
17#[derive(Clone, Debug, PartialEq)]
18pub struct OutTransport {
19    pub id: i64,
20    pub kind: Kind,
21    pub x: i64,
22    pub width: i64,
23    pub bottom: i64,
24    pub top: i64,
25    pub cars: i64,
26    pub car_positions: Vec<i64>,
27    /// The waiting floor of each car the save names (the per-car home bytes
28    /// of the elevator record). Written as the shaft's `schedule.homeFloors`
29    /// so idle cars go back to them after their first trip, as in the 1994
30    /// game. `None` for synthesized shafts, whose save names no homes.
31    pub home_floors: Option<Vec<i64>>,
32    /// `None` leaves `skipFloors` out of the save, as the synthesized
33    /// standard and service shafts do.
34    pub skip_floors: Option<Vec<i64>>,
35}

37impl OutTransport {
38    pub fn to_json(&self) -> Value {
39        let mut m = serde_json::Map::new();
40        m.insert("id".into(), json!(self.id));
41        m.insert("kind".into(), json!(self.kind.as_str()));
42        m.insert("x".into(), json!(self.x));
43        m.insert("width".into(), json!(self.width));
44        m.insert("bottom".into(), json!(self.bottom));
45        m.insert("top".into(), json!(self.top));
46        m.insert("cars".into(), json!(self.cars));
47        m.insert("carPositions".into(), json!(self.car_positions));
48        m.insert("carDir".into(), json!(vec![0; self.car_positions.len()]));
49        m.insert("load".into(), json!(0));
50        if let Some(h) = &self.home_floors {
51            m.insert("schedule".into(), json!({ "homeFloors": h }));
52        }
53        if let Some(s) = &self.skip_floors {
54            m.insert("skipFloors".into(), json!(s));
55        }
56        Value::Object(m)
57    }
58}

69/// True when placing a transport would overlap one already placed; stacked
70/// walkways of the same footprint may share their landing floor.
71fn overlaps_placed(
72    placed: &[OutTransport],
73    kind: Kind,
74    x: i64,
75    width: i64,
76    bottom: i64,
77    top: i64,
78) -> bool {
79    let is_walkway = kind.is_walkway();
80    for t in placed {
81        if x >= t.x + t.width || t.x >= x + width {
82            continue;
83        }
84        if bottom > t.top || t.bottom > top {
85            continue;
86        }
87        if is_walkway
88            && t.kind.is_walkway()
89            && t.x == x
90            && t.width == width
91            && (bottom == t.top || top == t.bottom)
92        {
93            continue;
94        }
95        return true;
96    }
97    false
98}

100/// `transportsFromDecoded`.
101/// `lobby_height` is the tall ground lobby's height from the header (1 for
102/// none): a flight out of the ground floor that climbs exactly that many
103/// stories comes in whole, as the one tall flight it is (#308). Every other
104/// multi-story record comes in as one flight per story.
105pub fn transports_from_decoded(
106    elevators: &[TdtElevator],
107    stairs: &[TdtStair],
108    first_id: i64,
109    lobby_height: i64,
110) -> DecodedTransports {
111    let mut out: Vec<OutTransport> = vec![];
112    let mut dropped_shafts = 0;
113    let mut adjusted_shafts = 0;
114    let mut dropped_flights = 0;
115    for e in elevators {
116        let Some(kind) = elevator_kind(e.type_id) else {
117            dropped_shafts += 1;
118            continue;
119        };
120        let raw_bottom = e.bottom_floor - TDT_FLOOR_OFFSET;
121        let raw_top = e.top_floor - TDT_FLOOR_OFFSET;
122        if raw_top <= raw_bottom {
123            dropped_shafts += 1;
124            continue;
125        }
126        let bottom = raw_bottom.max(MIN_FLOOR);
127        let mut top = raw_top.min(MAX_FLOOR);
128        if top <= bottom {
129            dropped_shafts += 1;
130            continue;
131        }
132        let mut trimmed = bottom != raw_bottom || top != raw_top;
133        if top - bottom > kind.max_span() {
134            top = bottom + kind.max_span();
135            trimmed = true;
136        }
137        let width = kind.facility().width;
138        let x = e.x.min(LOT_WIDTH - width).max(0);
139        if overlaps_placed(&out, kind, x, width, bottom, top) {
140            dropped_shafts += 1;
141            continue;
142        }
143        if trimmed {
144            adjusted_shafts += 1;
145        }
146        let cars = e.cars.min(kind.max_cars()).max(1);
147        let mut skip_floors = vec![];
148        // The floors the cars stop at, ascending, built in the same pass.
149        let mut stops = vec![bottom];
150        for fl in bottom + 1..top {
151            if e.serviced[(fl + TDT_FLOOR_OFFSET) as usize] == 0 {
152                skip_floors.push(fl);
153            } else {
154                stops.push(fl);
155            }
156        }
157        stops.push(top);
158        // Each car starts on its waiting floor and returns there when idle.
159        // A home the shaft does not stop at snaps to the nearest stop (the
160        // lower one on a tie), as an authored home does on a stop edit.
161        let car_positions: Vec<i64> = (0..cars as usize)
162            .map(|i| {
163                let home = e.car_homes[i] - TDT_FLOOR_OFFSET;
164                *stops
165                    .iter()
166                    .min_by_key(|&&f| (f - home).abs())
167                    .expect("bottom is a stop")
168            })
169            .collect();
170        let home_floors = Some(car_positions.clone());
171        out.push(OutTransport {
172            id: first_id + out.len() as i64,
173            kind,
174            x,
175            width,
176            bottom,
177            top,
178            cars,
179            car_positions,
180            home_floors,
181            skip_floors: Some(skip_floors),
182        });
183    }
184    let mut walkways = 0;
185    for s in stairs {
186        if s.type_id > 5 {
187            continue;
188        }
189        let kind = if s.type_id % 2 == 1 {
190            Kind::Stairs
191        } else {
192            Kind::Escalator
193        };
194        let stories = tdt_stair_stories(s.type_id);
195        let width = kind.facility().width;
196        let x = s.x.min(LOT_WIDTH - width).max(0);
197        let base = s.floor - TDT_FLOOR_OFFSET;
198        let whole = stories > 1 && base == 1 && stories == lobby_height;
199        let climb = if whole { stories } else { 1 };
200        for i in 0..stories / climb {
201            let bottom = base + i;
202            if bottom < MIN_FLOOR || bottom + climb > MAX_FLOOR {
203                continue;
204            }
205            if walkways >= 64 {
206                dropped_flights += 1;
207                continue;
208            }
209            if overlaps_placed(&out, kind, x, width, bottom, bottom + climb) {
210                dropped_flights += 1;
211                continue;
212            }
213            walkways += 1;
214            out.push(OutTransport {
215                id: first_id + out.len() as i64,
216                kind,
217                x,
218                width,
219                bottom,
220                top: bottom + climb,
221                cars: 0,
222                car_positions: vec![],
223                home_floors: None,
224                skip_floors: None,
225            });
226        }
227    }
228    DecodedTransports {
229        transports: out,
230        dropped_shafts,
231        adjusted_shafts,
232        dropped_flights,
233    }
234}

236/// Built extent [left, right) per floor, in ascending floor order (the
237/// order the importer's paving map is filled in).
238pub type BuiltExtents = BTreeMap<i64, (i64, i64)>;

240/// The fallback layout and the shafts it had no safe column for.
241pub struct Synthesized {
242    pub transports: Vec<OutTransport>,
243    /// Shafts left out because every column held a room or an earlier shaft on
244    /// a floor they serve; a shaft over a room would be an invalid tower.
245    pub left_out: i64,
246}

248/// `synthesizeTransports`: the fallback layout from the floor map alone.
249pub fn synthesize_transports(
250    built_extents: &BuiltExtents,
251    claimed: &HashMap<i64, Vec<u8>>,
252    hotel_floors: &[i64],
253    staff_floors: &[i64],
254    first_id: i64,
255) -> Synthesized {
256    if built_extents.is_empty() {
257        return Synthesized {
258            transports: vec![],
259            left_out: 0,
260        };
261    }
262    let mut bottom = i64::MAX;
263    let mut top = i64::MIN;
264    let mut min_left = i64::MAX;
265    let mut max_right = i64::MIN;
266    for (&floor, &(left, right)) in built_extents {
267        bottom = bottom.min(floor);
268        top = top.max(floor);
269        min_left = min_left.min(left);
270        max_right = max_right.max(right);
271    }
272    let center = round((min_left + max_right) as f64 / 2.0) as i64;
273
274    struct Spec {
275        kind: Kind,
276        bottom: i64,
277        top: i64,
278        skip_floors: Option<Vec<i64>>,
279    }
280    let mut specs: Vec<Spec> = vec![];
281    let mut covered = i64::MIN;
282    let ground_top = (bottom + 30).min(top);
283    if ground_top > bottom {
284        specs.push(Spec {
285            kind: Kind::ElevatorStandard,
286            bottom,
287            top: ground_top,
288            skip_floors: None,
289        });
290        covered = ground_top;
291    }
292    let mut anchor = LOBBY_INTERVAL;
293    while anchor < top {
294        let band_bottom = anchor.max(bottom);
295        let band_top = (band_bottom + 30).min(top);
296        if !(band_top <= covered || band_top <= band_bottom) {
297            specs.push(Spec {
298                kind: Kind::ElevatorStandard,
299                bottom: band_bottom,
300                top: band_top,
301                skip_floors: None,
302            });
303            covered = band_top;
304        }
305        anchor += LOBBY_INTERVAL;
306    }
307    if top >= 30 {
308        let ex_bottom = bottom.max(1);
309        if top > ex_bottom {
310            let skip: Vec<i64> = (ex_bottom + 1..top)
311                .filter(|&fl| !is_lobby_floor(fl))
312                .collect();
313            specs.push(Spec {
314                kind: Kind::ElevatorExpress,
315                bottom: ex_bottom,
316                top,
317                skip_floors: Some(skip),
318            });
319        }
320    }
321    if !hotel_floors.is_empty() {
322        let staff_range: Vec<i64> = hotel_floors
323            .iter()
324            .chain(staff_floors.iter())
325            .copied()
326            .chain(std::iter::once(1))
327            .collect();
328        let mut lo = staff_range.iter().copied().min().unwrap_or(1).max(bottom);
329        let hi = staff_range.iter().copied().max().unwrap_or(1).min(top);
330        while lo < hi {
331            let t = (lo + 30).min(hi);
332            specs.push(Spec {
333                kind: Kind::ElevatorService,
334                bottom: lo,
335                top: t,
336                skip_floors: None,
337            });
338            lo = t;
339        }
340    }
341    specs.truncate(24);
342    let total_width: i64 = specs.iter().map(|s| s.kind.facility().width).sum();
343    let mut x = round(center as f64 - total_width as f64 / 2.0)
344        .min((LOT_WIDTH - total_width) as f64)
345        .max(0.0) as i64;
346    let mut transports: Vec<OutTransport> = vec![];
347    let mut left_out = 0;
348    for s in specs {
349        let width = s.kind.facility().width;
350        let cars = 8;
351        // The strip of shafts goes where the tower's middle is. A shaft whose
352        // column holds a room, or another shaft, moves to the nearest column
353        // that is free over every floor it serves; with none free it is left out.
354        let free = |at: i64| {
355            at >= 0
356                && at + width <= LOT_WIDTH
357                && !(s.bottom..=s.top).any(|floor| {
358                    claimed.get(&floor).is_some_and(|row| {
359                        row[at as usize..(at + width) as usize]
360                            .iter()
361                            .any(|&c| c != 0)
362                    })
363                })
364                && !transports.iter().any(|t| {
365                    at < t.x + t.width && t.x < at + width && s.bottom <= t.top && t.bottom <= s.top
366                })
367        };
368        let placed = (0..=LOT_WIDTH)
369            .flat_map(|step| [x + step, x - step])
370            .find(|&at| free(at));
371        x += width;
372        let Some(placed) = placed else {
373            left_out += 1;
374            continue;
375        };
376        transports.push(OutTransport {
377            id: first_id + transports.len() as i64,
378            kind: s.kind,
379            x: placed,
380            width,
381            bottom: s.bottom,
382            top: s.top,
383            cars,
384            car_positions: (0..cars).map(|i| (s.bottom + i).min(s.top)).collect(),
385            home_floors: None,
386            skip_floors: s.skip_floors,
387        });
388    }
389    Synthesized {
390        transports,
391        left_out,
392    }
