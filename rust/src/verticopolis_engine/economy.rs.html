107/// What the catalog publishes as a facility's spend per customer: a venue that
108/// has subtypes and earns a daily income in `mode` (the shop, fast food and
109/// restaurant, plus the food hall, amusements and boutique bay in Modern).
110/// The inspector's patronage block gates on exactly this.
111pub fn catalog_spend_per_customer(mode: GameMode, kind: Kind) -> Option<f64> {
112    let has_subtypes = kind.subtype_list().is_some();
113    let earns_in_mode = commercial_daily_income(mode, kind).is_some();
114    if has_subtypes && earns_in_mode {
115        retail_spend_per_customer(kind)
116    } else {
117        None
118    }
119}

182    /// `collectRent`: quarterly lease income.
183    pub fn collect_rent(&mut self) {
184        let mut sums: IndexMap<Kind, (f64, i64)> = IndexMap::new();
185        for u in &self.tower.units {
186            if !matches!(u.kind, Kind::Office | Kind::FitnessClub | Kind::Clinic) {
187                continue;
188            }
189            if !u.is_tenanted() || !self.tower.is_floor_served(u.floor) {
190                continue;
191            }
192            let b = sums.entry(u.kind).or_insert((0.0, 0));
193            b.0 += rent_of(u.kind, u.rent, u.no_rate);
194            b.1 += 1;
195        }
196        let scale = self
197            .mode
198            .quarterly_rent_scale(self.clock.calendar.quarter_days);
199        for (kind, (sum, n)) in sums {
200            let amt = jsmath::round(sum * scale);
201            if amt <= 0.0 {
202                continue;
203            }
204            self.money += amt;
205            self.record_money(ledger_cat_for(kind).unwrap_or("upkeep"), amt);
206            let msg = lease_message(kind, amt, n);
207            self.emit(&msg, LogKind::Money);
208        }
209    }
210}

212/// The quarterly lease line (`LEASE_MESSAGE` in `EconomySystem.ts`).
213pub(crate) fn lease_message(kind: Kind, amt: f64, n: i64) -> String {
214    let plural = if n > 1 { "s" } else { "" };
215    match kind {
216        Kind::Office => format!(
217            "Quarterly office rent collected: ${} ({n} offices).",
218            with_thousands(amt)
219        ),
220        Kind::FitnessClub => format!(
221            "Fitness Club membership dues collected: ${} ({n} club{plural}).",
222            with_thousands(amt)
223        ),
224        _ => format!(
225            "Clinic lease collected: ${} ({n} clinic{plural}).",
226            with_thousands(amt)
227        ),
228    }
229}

244    #[test]
245    fn rent_lines_word_their_counts_as_typescript_does() {
246        // The TypeScript templates (EconomySystem.ts LEASE_MESSAGE, rentalIncome.ts).
247        assert_eq!(
248            lease_message(Kind::Office, 1520000.0, 152),
249            "Quarterly office rent collected: $1,520,000 (152 offices)."
250        );
251        // The TypeScript office line never drops its plural.
252        assert_eq!(
253            lease_message(Kind::Office, 10000.0, 1),
254            "Quarterly office rent collected: $10,000 (1 offices)."
255        );
256        assert_eq!(
257            crate::build::treasure_message(452311.0),
258            "💰 Excavation crews unearthed buried treasure worth $452,311!"
259        );
260        assert_eq!(
261            lease_message(Kind::FitnessClub, 30000.0, 1),
262            "Fitness Club membership dues collected: $30,000 (1 club)."
263        );
264        assert_eq!(
265            lease_message(Kind::FitnessClub, 60000.0, 2),
266            "Fitness Club membership dues collected: $60,000 (2 clubs)."
267        );
268        assert_eq!(
269            lease_message(Kind::Clinic, 1500.0, 1),
270            "Clinic lease collected: $1,500 (1 clinic)."
271        );
272        assert_eq!(
273            lease_message(Kind::Clinic, 3000.0, 2),
274            "Clinic lease collected: $3,000 (2 clinics)."
275        );
276        assert_eq!(
277            rental_message(950.0, 1),
278            "Monthly rent collected: $950 (1 rental)."
