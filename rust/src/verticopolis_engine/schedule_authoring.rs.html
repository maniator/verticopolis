1//! The Auto-tune proposal for an elevator's schedule: a read-only query that
2//! turns the shaft's own measured telemetry into the active-car rows and
3//! staging the Modern dialog offers. A port of `autoTuneSchedule`
4//! (the retired TypeScript `src/engine/scheduleAuthoring.ts`) with the dialog's warm-up gate and
5//! origin aim (`src/ui/uiElevatorSchedule.ts`) moved into the engine, so the
6//! host reads what is ready and what to propose instead of recomputing it.
7//!
8//! Nothing here writes: no tower, telemetry, rng, cash or event stream is
9//! touched, and the proposal is only ever applied by the player through the
10//! ordinary schedule command (one undo step). No wording lives here; the host
11//! renders the buttons and the copy.
12//!
13//! The tuning is a simple rule: each hour gets the fewest cars that carry
14//! that hour's measured load with room, never below one car.
15//! It is not a validated wait-time optimization, and nothing here claims it
16//! is.
17

29/// Sampled hours (nonzero slots) a day's curve needs before it counts as
30/// measured. One busy hour must not arm Auto-tune against 23 empty slots.
31pub const WARMED_MIN_HOURS: usize = 6;

33/// How the query ended.
34#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
35#[serde(rename_all = "kebab-case")]
36pub enum ProposalStatus {
37    /// At least one day type is warm; `proposal` carries the rows.
38    Available,
39    /// Neither day type has a warm curve yet.
40    InsufficientData,
41    /// The tower's mode offers no Auto-tune (Classic). Readiness is still
42    /// reported.
43    UnavailableInMode,
44    /// The shaft holds no cars (a loaded save can), so there is nothing to
45    /// tune. Readiness is still reported.
46    NoCars,
47    /// The id names a stair or escalator.
48    NotAnElevator,
49    /// The id names no shaft.
50    MissingShaft,
51    /// The working draft is not a schedule the engine reads.
52    InvalidDraft,
53}

55/// One day type's readiness. A warm day is the one Auto-tune tunes, and its
56/// measured 24-hour `curve` (0 to 1 per hour of day) is handed back so the host
57/// draws and advises from the ring the engine tuned from, which stays current
58/// while a dialog is open. A cold day has no curve.
59#[derive(Clone, Debug, Default, PartialEq, Serialize)]
60#[serde(rename_all = "camelCase")]
61pub struct DayReadiness {
62    pub warm: bool,
63    pub sampled_hours: usize,
64    pub curve: Option<Vec<f64>>,
65}

67/// The rows and staging to offer. A day type that is not warm keeps the

101/// The hours of a day's ring that have been sampled, a measured zero
102/// included (#474): the count the warm gate reads.
103fn sampled_hours(sampled: &[bool]) -> usize {
104    sampled.iter().filter(|s| **s).count()
105}
106
107/// The ring when the day is warm, for a caller that only reads it.
108pub(crate) fn warm_curve<'a>((curve, sampled): (&'a [f64], &[bool])) -> Option<&'a [f64]> {
109    (sampled_hours(sampled) >= WARMED_MIN_HOURS).then_some(curve)
110}
111
112pub(crate) fn readiness((curve, sampled): (&[f64], &[bool])) -> DayReadiness {
113    let sampled = sampled_hours(sampled);
114    let warm = sampled >= WARMED_MIN_HOURS;
115    DayReadiness {
116        warm,
117        sampled_hours: sampled,
118        curve: warm.then(|| curve.to_vec()),
119    }
120}

122/// The share of capacity at which the cars running an hour count as full at
123/// a sample: riders step off at each stop, so a busy car rarely reads 1.0.
124const SATURATED: f64 = 0.9;
125
126/// Each hour's cars from its load fraction: the fewest that carry the hour's
127/// riders with room ([`cars_for`]), at least one, at most the installed cars.
128fn tuned_row(curve: &[f64], cars: i64) -> Vec<i64> {
129    (0..SCHEDULE_HOURS)
130        .map(|h| cars_for(curve.get(h).copied().unwrap_or(0.0) * cars as f64, cars))
131        .collect()
132}
133
134/// The fewest cars, from 1 to `cars`, that carry `load` cars' worth of
135/// riders below [`SATURATED`] each. A proposal of just the load (its rounded
136/// share) left those cars full, and full cars measure no more than they
137/// carry, so a shaft cut to one car never won a car back. Now a tuned row
138/// has room by construction, so tuning it again proposes the same row, and
139/// an hour whose cars ran full measures at least `SATURATED` of each and is
140/// proposed one more car each tune until it has room.
141fn cars_for(load: f64, cars: i64) -> i64 {
142    let mut n = 1;
143    while n < cars && load / n as f64 >= SATURATED {
144        n += 1;
145    }
146    n
147}
148
149/// A tuned row with every hour the ring has not sampled set back to the
150/// draft's count (every car where the draft has no row).
151fn keep_unmeasured_hours(
152    mut row: Vec<i64>,
153    draft: Option<&[i64]>,
154    sampled: &[bool],
155    cars: i64,
156) -> Vec<i64> {
157    for (h, n) in row.iter_mut().enumerate() {
158        if sampled.get(h) != Some(&true) {
159            *n = draft.and_then(|d| d.get(h)).copied().unwrap_or(cars);
160        }
161    }
162    row
163}

165/// The origin floors summed over the day's 24 slots, in first-seen order.
166fn day_origin_totals(ring: &[IndexMap<i64, f64>]) -> IndexMap<i64, f64> {
167    let mut totals: IndexMap<i64, f64> = IndexMap::new();
168    for slot in ring {
169        for (&floor, &n) in slot {
170            *totals.entry(floor).or_insert(0.0) += n;
171        }
172    }
173    totals
174}

176/// The busiest boarding floor among `served`, ties to the lower floor; `None`
177/// with no mass.
178fn peak_origin_floor(totals: &IndexMap<i64, f64>, served: &[i64]) -> Option<i64> {
179    let mut best: Option<(i64, f64)> = None;
180    let mut total = 0.0;
181    for (&floor, &n) in totals {
182        if served.binary_search(&floor).is_err() {
183            continue;
184        }
185        total += n;
186        let better = match best {
187            None => true,
188            Some((bf, bn)) => n > bn || (n == bn && floor < bf),
189        };
190        if better {
191            best = Some((floor, n));
192        }
193    }
194    if total > 0.0 {
195        best.map(|(floor, _)| floor)
196    } else {
197        None
198    }
199}

201/// The staging seed: the lower half of the fleet at the base lobby (the only
202/// car of a one-car shaft included), the upper half up-tower, aimed at the
203/// busiest measured boarding floor when it sits above the base.
204fn seeded_homes(geo: &ShaftGeometry, aim: Option<i64>) -> Vec<i64> {
205    let up = match aim {
206        Some(a) if a > geo.base_lobby() => a.clamp(geo.bottom, geo.top),
207        _ => geo.up_tower_target(),
208    };
209    geo.seeded_homes(up)
210}

212#[doc(hidden)]
213impl Simulation {
214    /// The Auto-tune proposal for shaft `id` against the player's working
215    /// `draft` (the schedule as the dialog holds it, homes left out when the
216    /// player has not staged by hand). `weekend_view` names the day type the
217    /// player is looking at, which picks the day whose boarding geography
218    /// aims the staging seed. Read-only.
219    pub fn auto_tune_proposal(
220        &self,
221        id: i64,
222        draft: Option<&Schedule>,
223        weekend_view: bool,
224    ) -> AutoTuneProposal {
225        let Some(i) = self.tower.transport_index(id) else {
226            return AutoTuneProposal::refused(ProposalStatus::MissingShaft);
227        };
228        let t = &self.tower.transports[i];
229        if !t.kind.is_elevator() {
230            return AutoTuneProposal::refused(ProposalStatus::NotAnElevator);
231        }
232        let empty = HourlyByDay::default();
233        let rings = self.telemetry.hourly.get(&id).unwrap_or(&empty);
234        let (wd, we) = (readiness(rings.day(false)), readiness(rings.day(true)));
235        // Readiness is the shaft's, whatever the mode offers: the Classic grid
236        // still shows a measured curve, it just gets no proposal.
237        let unavailable = if !self.mode.schedule_assist() {
238            Some(ProposalStatus::UnavailableInMode)
239        } else if t.cars <= 0 {
240            Some(ProposalStatus::NoCars)
241        } else {
242            None
243        };
244        if let Some(status) = unavailable {
245            return AutoTuneProposal {
246                status,
247                weekday: wd,
248                weekend: we,
249                proposal: None,
250            };
251        }
252        if !wd.warm && !we.warm {
253            return AutoTuneProposal {
254                status: ProposalStatus::InsufficientData,
255                weekday: wd,
256                weekend: we,
257                proposal: None,
258            };
259        }
260        let twins = self.twins_of(i, &self.elevator_stop_sets());
261        let (weekday, weekend) = self.tuned_rows(t, draft, &twins).unwrap_or_default();
262
263        let geo = self.shaft_geometry(t);
264        let authored = draft
265            .and_then(|d| d.home_floors.as_ref())
266            .filter(|h| !h.is_empty());
267        let homes_seeded = authored.is_none();
268        let homes = match authored {
269            Some(h) => h.clone(),
270            None => {
271                // The up-tower half aims above the base lobby, where every
272                // crowd rider coming in boards (#477), so only floors above
273                // it compete for the aim.
274                let base = geo.base_lobby();
275                let above = &geo.served[geo.served.partition_point(|&f| f <= base)..];
276                let aim = aim_on_weekend(weekend_view, wd.warm, we.warm).and_then(|weekend| {
277                    let origins = self.telemetry.origins.get(&id)?;
278                    origin_aim(origins, weekend, above)
279                });
280                seeded_homes(&geo, aim)
281            }
282        };
283        let home_floors = homes
284            .into_iter()
285            .take(t.cars as usize)
286            .map(|f| geo.snap_to_served(f.clamp(geo.bottom, geo.top)))
287            .collect();
288        AutoTuneProposal {
289            status: ProposalStatus::Available,
290            weekday: wd,
291            weekend: we,
292            proposal: Some(ProposedSchedule {
293                weekday,
294                weekend,
295                home_floors,
296                homes_seeded,
297            }),
298        }
299    }
300}

302/// The rows of one shaft Auto-tune proposes: weekday, then weekend.
303type TunedRows = (Option<Vec<i64>>, Option<Vec<i64>>);
304
305impl Simulation {
306    /// The active-car rows Auto-tune gives `t` against `draft`: a warm day's
307    /// from the load `t` and its `twins` measured, a cold day's the draft's
308    /// own. `None` when the shaft has no cars or neither day is warm. The
309    /// proposal and the morning pass share it; the pass needs no staging.
310    pub(crate) fn tuned_rows(
311        &self,
312        t: &Transport,
313        draft: Option<&Schedule>,
314        twins: &[i64],
315    ) -> Option<TunedRows> {
316        let empty = HourlyByDay::default();
317        let rings = self.telemetry.hourly.get(&t.id).unwrap_or(&empty);
318        let wd_warm = warm_curve(rings.day(false)).is_some();
319        let we_warm = warm_curve(rings.day(true)).is_some();
320        if t.cars <= 0 || !(wd_warm || we_warm) {
321            return None;
322        }
323        let cars = t.cars;
324        // A day that is not warm keeps the draft's own row, held to the cars
325        // installed in case the draft was written for a bigger fleet.
326        let draft_row = |weekend: bool| {
327            let row = draft.and_then(|d| {
328                if weekend {
329                    d.weekend.as_ref()
330                } else {
331                    d.weekday.as_ref()
332                }
333            });
334            row.map(|r| r.iter().map(|&n| n.clamp(0, cars)).collect::<Vec<i64>>())
335        };
336        // A warm day still keeps the hours its ring has not measured (the
337        // hours before a reload, a new shaft's first night) at the draft's
338        // count: read as idle, they would drop to one car, a rush included.
339        let tuned = |weekend: bool| {
340            let (ring, sampled) = rings.day(weekend);
341            let row = tuned_row(&self.twin_load(twins, ring, weekend), cars);
342            keep_unmeasured_hours(row, draft_row(weekend).as_deref(), sampled, cars)
343        };
344        let weekday = if wd_warm {
345            Some(tuned(false))
346        } else {
347            draft_row(false)
348        };
349        let weekend = if we_warm {
350            Some(tuned(true))
351        } else {
352            draft_row(true)
353        };
354        Some((weekday, weekend))
355    }
356
357    /// Each elevator's stops as a floor set, with its index in
358    /// `tower.transports`, ascending: read once for a caller that looks for
359    /// twins, so a pass over many shafts compares sets, not stop lists.
360    /// Stairs and escalators are left out, so the twin search never walks
361    /// a Modern tower's links.
362    pub(crate) fn elevator_stop_sets(&self) -> Vec<(usize, FloorBits)> {
363        let set = |(i, t): (usize, &Transport)| {
364            t.kind.is_elevator().then(|| {
365                let mut floors = FloorBits::default();
366                for f in self.tower.stops_of(t) {
367                    // Every stop is a buildable floor, which has a slot.
368                    if let Some(slot) = floor_slot(f) {
369                        floors.insert(slot);
370                    }
371                }
372                (i, floors)
373            })
374        };
375        self.tower
376            .transports
377            .iter()
378            .enumerate()
379            .filter_map(set)
380            .collect()
381    }
382
383    /// The ids of the twins of the transport at index `i`: elevators of the
384    /// same kind stopping at every floor it stops at, so it could carry
385    /// their riders. `sets` is [`Self::elevator_stop_sets`]; each pair is
386    /// one set comparison.
387    pub(crate) fn twins_of(&self, i: usize, sets: &[(usize, FloorBits)]) -> Vec<i64> {
388        let Ok(at) = sets.binary_search_by_key(&i, |&(j, _)| j) else {
389            return Vec::new();
390        };
391        let (t, mine) = (&self.tower.transports[i], &sets[at].1);
392        let twin = |&(j, ref theirs): &(usize, FloorBits)| {
393            let other = &self.tower.transports[j];
394            (j != i && other.kind == t.kind && theirs.covers(mine)).then_some(other.id)
395        };
396        sets.iter().filter_map(twin).collect()
397    }
398
399    /// Each hour's load to tune a shaft by: its `own`, or its `twins'`
400    /// ([`Self::twins_of`]) when they carried more. Modern riders route
401    /// around a shaft cut to one car, so its own reading falls with each cut
402    /// and never wins its cars back (owner, 2026-11-17); the twins' reading
403    /// is what it would carry. Each twin counts on a day it has measured
404    /// (`warm_curve`), and the twins' load is their mean.
405    fn twin_load(&self, twins: &[i64], own: &[f64], weekend: bool) -> Vec<f64> {
406        let (mut sum, mut counted) = (vec![0.0; SCHEDULE_HOURS], 0.0);
407        for id in twins {
408            let Some(rings) = self.telemetry.hourly.get(id) else {
409                continue;
410            };
411            let Some(curve) = warm_curve(rings.day(weekend)) else {
412                continue;
413            };
414            for (h, load) in sum.iter_mut().zip(curve) {
415                *h += load;
416            }
417            counted += 1.0;
418        }
419        (0..SCHEDULE_HOURS)
420            .map(|h| {
421                let mine = own.get(h).copied().unwrap_or(0.0);
422                if counted > 0.0 {
423                    mine.max(sum[h] / counted)
424                } else {
425                    mine
426                }
427            })
428            .collect()
429    }
430}

432/// Which day type's boarding geography aims the staging: the one on screen
433/// when it is warm, else the other warm day; `None` when neither is warm.
434fn aim_on_weekend(weekend_view: bool, wd_warm: bool, we_warm: bool) -> Option<bool> {
435    match (weekend_view, wd_warm, we_warm) {
436        (_, false, false) => None,
437        (true, _, true) | (false, false, true) => Some(true),
438        _ => Some(false),
439    }
440}

442/// The staging aim: the busiest boarding floor in `served` (ascending) over
443/// the whole day's rings.
444fn origin_aim(origins: &OriginRings, weekend: bool, served: &[i64]) -> Option<i64> {
445    let ring = if weekend {
446        &origins.weekend
447    } else {
448        &origins.weekday
449    };
450    peak_origin_floor(&day_origin_totals(ring), served)
451}
