23impl UnitState {
24    pub fn as_str(self) -> &'static str {
25        match self {
26            UnitState::Construction => "construction",
27            UnitState::Empty => "empty",
28            UnitState::Occupied => "occupied",
29            UnitState::MovingIn => "moving_in",
30            UnitState::Vacating => "vacating",
31            UnitState::Asleep => "asleep",
32            UnitState::Dirty => "dirty",
33            UnitState::Infested => "infested",
34            UnitState::Fire => "fire",
35            UnitState::Gutted => "gutted",
36        }
37    }
38
39    /// The state a save names, the inverse of [`UnitState::as_str`].
40    pub fn from_name(name: &str) -> Option<UnitState> {
41        Some(match name {
42            "construction" => UnitState::Construction,
43            "empty" => UnitState::Empty,
44            "occupied" => UnitState::Occupied,
45            "moving_in" => UnitState::MovingIn,
46            "vacating" => UnitState::Vacating,
47            "asleep" => UnitState::Asleep,
48            "dirty" => UnitState::Dirty,
49            "infested" => UnitState::Infested,
50            "fire" => UnitState::Fire,
51            "gutted" => UnitState::Gutted,
52            _ => return None,
53        })
54    }

89/// The facts a unit rarely carries, each unset until something sets it: a
90/// move-out notice (`vacate_*`), the cinema's film policy, its film's age and
91/// its ticket price, a retail room's four daily tallies, a room's opening time (`complete_at*`) and a hotel room's
92/// dirty days.
93#[derive(Clone, Debug, Default, PartialEq)]
94pub struct Extras {
95    pub vacate_reason: Option<&'static str>,
96    pub vacate_at: Option<f64>,
97    pub film_policy: Option<&'static str>,
98    /// The days a theatre's film has shown, counted each morning (#1932); a
99    /// theatre that has not seen a morning yet reads it as 0.
100    pub film_days: Option<u8>,
101    /// Whether a theatre shows a classic film, whose draw never climbs past
102    /// a fading film's (#1932). Modern calls it a revival.
103    pub film_classic: bool,
104    /// Whether a Modern theater shows a blockbuster (#1950).
105    pub film_blockbuster: bool,
106    /// A Modern theater's ticket price step, unset at Normal (#1950).
107    pub ticket_price: Option<&'static str>,
108    /// A Modern theater's second screen, stored only while it shows
109    /// something other than the first (#1950); unset, it mirrors screen 1.
110    pub second_screen: Option<ScreenFilm>,
111    pub patronage_today: Option<f64>,
112    pub patronage_yest: Option<f64>,
113    pub profit_today: Option<f64>,
114    pub profit_yest: Option<f64>,
115    pub complete_at: Option<f64>,
116    /// A `completeAt` the save carried that is not a number (a string, a
117    /// boolean, null). The TypeScript keeps and re-saves it as is, and the
118    /// completion check compares the clock against `Number(value)`, so a
119    /// string that reads as a number completes then and anything else never
120    /// completes. Kept beside `complete_at` so the hash and the behavior both
121    /// match.
122    pub complete_at_raw: Option<Value>,
123    pub dirty_days: Option<i64>,
124    /// How many stories a lobby tile rises, 2 or 3, unset for a one-story
125    /// lobby and for every other kind (#308, #2068). Set once when the tile
126    /// is laid and never changed. The save writes it as `lobbyHeights` runs,
127    /// not on the unit.
128    pub lobby_stories: Option<u8>,
129}

151impl Extras {

158/// A unit's extras to write: assign through it like the record itself. A
159/// unit that had none gets a record while this lives, and drops it again when
160/// the last fact is cleared, so a unit's extras are never an empty box.
161pub struct ExtrasMut<'a>(&'a mut Option<Box<Extras>>);
162
163impl std::ops::Deref for ExtrasMut<'_> {
164    type Target = Extras;
165    fn deref(&self) -> &Extras {
166        self.0.as_deref().unwrap_or(&NO_EXTRAS)
167    }
168}
169
170impl std::ops::DerefMut for ExtrasMut<'_> {
171    fn deref_mut(&mut self) -> &mut Extras {
172        self.0.get_or_insert_default()
173    }
174}
175
176impl Drop for ExtrasMut<'_> {
177    fn drop(&mut self) {
178        if self.0.as_deref().is_some_and(|e| *e == Extras::default()) {
179            *self.0 = None;
180        }
181    }
182}

195    /// `serializeUnit` (sim/coerce.ts).
196    pub fn serialize(&self) -> Value {
197        let f = self.kind.facility();
198        let mut m = Map::new();
199        m.insert("id".into(), json!(self.id));
200        m.insert("kind".into(), json!(self.kind.as_str()));
201        m.insert("floor".into(), json!(self.floor));
202        m.insert("x".into(), json!(self.x));
203        if !(self.width == 1 && f.width == 1 && self.kind.is_structural()) {
204            m.insert("width".into(), json!(self.width));
205        }
206        if self.state != UnitState::Empty {
207            m.insert("state".into(), json!(self.state.as_str()));
208        }
209        if self.satisfaction != 1.0 {
210            m.insert("satisfaction".into(), json!(self.satisfaction));
211        }
212        if self.occupants != 0 && !self.kind.counts_visitors() {
213            m.insert("occupants".into(), json!(self.occupants));
214        }
215        if self.ever_occupied {
216            m.insert("everOccupied".into(), json!(true));
217        }
218        if self.pending_income != 0.0 {
219            m.insert("pendingIncome".into(), json!(self.pending_income));
220        }
221        if self.label != f.name {
222            m.insert("label".into(), json!(self.label));
223        }
224        if let Some(r) = self.residents {
225            m.insert("residents".into(), json!(r));
226        }
227        if let Some(v) = self.visitors_yest {
228            m.insert("visitorsYest".into(), json!(v));
229        }
230        if let Some(r) = self.rent {
231            m.insert("rent".into(), json!(r));
232        }
233        if self.no_rate {
234            m.insert("noRate".into(), json!(true));
235        }
236        let extras = self.extras();
237        if let Some(v) = extras.vacate_reason {
238            m.insert("vacateReason".into(), json!(v));
239        }
240        if let Some(v) = extras.vacate_at {
241            m.insert("vacateAt".into(), json!(v));
242        }
243        if let Some(v) = extras.film_policy {
244            m.insert("filmPolicy".into(), json!(v));
245        }
246        if let Some(v) = extras.film_days {
247            m.insert("filmDays".into(), json!(v));
248        }
249        if extras.film_classic {
250            m.insert("filmClassic".into(), json!(true));
251        }
252        if extras.film_blockbuster {
253            m.insert("filmBlockbuster".into(), json!(true));
254        }
255        if let Some(v) = extras.ticket_price {
256            m.insert("ticketPrice".into(), json!(v));
257        }
258        if let Some(screen) = extras.second_screen {
259            m.insert("screen2".into(), screen.to_json());
260        }
261        if let Some(v) = self.subtype {
262            m.insert("subtype".into(), json!(v));
263        }
264        if self.kind.subtype_list().is_some() {
265            if let Some(v) = extras.patronage_today {
266                m.insert("patronageToday".into(), json!(v));
267            }
268            if let Some(v) = extras.patronage_yest {
269                m.insert("patronageYest".into(), json!(v));
270            }
271            if let Some(v) = extras.profit_today {
272                m.insert("profitToday".into(), json!(v));
273            }
274            if let Some(v) = extras.profit_yest {
275                m.insert("profitYest".into(), json!(v));
276            }
277        }
278        if let Some(v) = extras.complete_at {
279            m.insert("completeAt".into(), json!(v));
280        } else if let Some(raw) = &extras.complete_at_raw {
281            m.insert("completeAt".into(), raw.clone());
282        }
283
284        if let Some(d) = extras.dirty_days {
285            if d != 0 && self.state == UnitState::Dirty {
286                m.insert("dirtyDays".into(), json!(d));
287            }
288        }
289        Value::Object(m)
290    }
291}

314    /// into a car, a walker starting up the stairs or an escalator. It only
315    /// counts up, and only an undo reads it: a shaft ridden since a step was
316    /// made has been in use, so the step does not take it out or take its
317    /// cars or floors away at their full price. Never saved: a load starts
318    /// it at zero, and forgets the steps that read it.
319    pub rides: u64,
320}

322impl Transport {
323    /// `serializeTransport` (sim/coerce.ts).
324    pub fn serialize(&self) -> Value {
325        let mut m = Map::new();
326        m.insert("id".into(), json!(self.id));
327        m.insert("kind".into(), json!(self.kind.as_str()));
328        m.insert("x".into(), json!(self.x));
329        m.insert("width".into(), json!(self.width));
330        m.insert("bottom".into(), json!(self.bottom));
331        m.insert("top".into(), json!(self.top));
332        m.insert("cars".into(), json!(self.cars));
333        m.insert("carPositions".into(), json!(self.car_positions));
334        m.insert("carDir".into(), json!(self.car_dir));
335        if let Some(l) = &self.car_load {
336            m.insert("carLoad".into(), json!(l));
337        }
338        m.insert("load".into(), json!(self.load));
339        if let Some(s) = &self.skip_floors {
340            m.insert("skipFloors".into(), json!(s));
341        }
342        if self.preserve_basement_stops {
343            m.insert("preserveBasementStops".into(), json!(true));
344        }
345        if let Some(s) = &self.schedule {
346            if !s.is_empty() {
347                m.insert("schedule".into(), s.to_json());
348            }
349        }
350        Value::Object(m)
351    }

353    /// Every floor from `bottom` to `top` the shaft stops at, lowest first:
354    /// the floors [`Transport::stops_at`] answers true for. It marks the
355    /// skipped floors once and then walks the span, so a shaft with `s`
356    /// skips over a span of `n` floors costs `n + s` rather than the `n * s`
357    /// of asking `stops_at` floor by floor. A skip that is not a whole
358    /// floor inside the span matches nothing, as in `stops_at`.
359    pub(crate) fn stop_floors(&self) -> Vec<i64> {
360        if self.top < self.bottom {
361            return Vec::new();
362        }
363        let skips = self.skip_floors.as_deref().unwrap_or(&[]);
364        if skips.is_empty() {
365            return (self.bottom..=self.top).collect();
366        }
367        let span = (self.top - self.bottom + 1) as usize;
368        let mut skipped = vec![false; span];
369        for &s in skips {
370            let whole_floor_in_span =
371                s.fract() == 0.0 && s >= self.bottom as f64 && s <= self.top as f64;
372            if whole_floor_in_span {
373                skipped[(s as i64 - self.bottom) as usize] = true;
374            }
375        }
376        (self.bottom..=self.top)
377            .zip(skipped)
378            .filter_map(|(fl, skip)| (!skip).then_some(fl))
379            .collect()
380    }
381
382    pub fn stops_at(&self, fl: i64) -> bool {
383        fl >= self.bottom
384            && fl <= self.top
385            && !self
386                .skip_floors
387                .as_ref()
388                .is_some_and(|s| s.contains(&(fl as f64)))
389    }
390}
