1//! The crowd in a save: everyone in the tower as they stand, so a reload
2//! keeps the people walking, waiting, riding, eating and cleaning instead of
3//! starting an empty building that fills again from the lobby.
4//!
5//! A save carries the people and the few crowd values a step reads back
6//! (the crowd's random stream, its next id, the spawn accumulator, the
7//! frustration level, the commute-stress averages, the last step's hour, the
8//! lines and the stood-down cars the Modern router reads next, and the
9//! walk-ins booked for the rest of the hour). Everything else the crowd keeps is either
10//! rebuilt from the people on load (`adopt.rs`: who
11//! rides each car, the venue and meal head counts, the housekeeping crews
12//! and the rooms they hold) or rebuilt each step (routes, landings, the
13//! open cars).
14//!
15//! A save from before the crowd was saved has no `crowd` key and loads with
16//! an empty crowd, as every save did. A build from before this one ignores
17//! the key. A person the loader cannot place (a hand-edited save with an
18//! unknown state, a route that does not fit its legs, a floor off the lot) is
19//! left out rather than trusted, so a forged save cannot send the motion code
20//! past the end of a route.
21

41/// The most people a save restores. A working tower holds at most the
42/// walk-ins' ceiling, the 140-person sample and its housekeeping crews, far
43/// below this; the cap only bounds what a hand-edited save can make the
44/// loader keep.
45pub(crate) const MAX_SAVED_PEOPLE: usize = 10_000;
46
47/// The most walk-in bookings a save restores: an hour's bookings for a
48/// tower at full demand stay well under it.
49pub(crate) const MAX_SAVED_WALK_INS: usize = 20_000;
50
51/// How far off the lot a saved position may lie, in tiles, on either side:
52/// a walk from anywhere in it ends in a bounded time.
53const OFF_LOT_MARGIN: f64 = LOT_WIDTH as f64;
54
55/// The longest route a saved person may carry, in floors. A route visits
56/// each shaft at most once, and a tower holds at most 88 shafts.
57const MAX_ROUTE_FLOORS: usize = 128;

59impl PState {
60    fn parse(s: &str) -> Option<PState> {
61        Some(match s {
62            "toShaft" => PState::ToShaft,
63            "waiting" => PState::Waiting,
64            "riding" => PState::Riding,
65            "climbing" => PState::Climbing,
66            "toDest" => PState::ToDest,
67            "dwelling" => PState::Dwelling,
68            _ => return None,
69        })
70    }
71
72    /// Whether a person in this state is partway along a leg of their route,
73    /// so the route must hold the leg's far floor.
74    fn on_a_leg(self) -> bool {
75        matches!(
76            self,
77            PState::ToShaft | PState::Waiting | PState::Riding | PState::Climbing
78        )
79    }
80}

143    /// The person a save wrote, or `None` when the entry is not one the
144    /// motion code can carry on from (see the module notes).
145    pub(crate) fn from_save(v: &Value) -> Option<Person> {
146        let o = v.as_object()?;
147        let int = |key: &str| o.get(key).and_then(Value::as_f64).and_then(whole);
148        let finite = |key: &str| o.get(key).and_then(Value::as_f64).filter(|n| n.is_finite());
149        let flag = |key: &str| o.get(key) == Some(&Value::Bool(true));
150        let opt_int = |key: &str| -> Option<Option<i64>> {
151            match o.get(key) {
152                None | Some(Value::Null) => Some(None),
153                Some(v) => v.as_f64().and_then(whole).map(Some),
154            }
155        };
156        let opt_finite = |key: &str| -> Option<Option<f64>> {
157            match o.get(key) {
158                None | Some(Value::Null) => Some(None),
159                Some(v) => v.as_f64().filter(|n| n.is_finite()).map(Some),
160            }
161        };
162        let on_lot = |f: i64| (MIN_FLOOR..=MAX_FLOOR).contains(&f);
163        // A position across the lot (with a margin either side); a height
164        // between the lowest and the highest floor.
165        let across = |key: &str| {
166            let span = -OFF_LOT_MARGIN..=LOT_WIDTH as f64 + OFF_LOT_MARGIN;
167            finite(key).filter(|x| span.contains(x))
168        };
169        let height = |key: &str| {
170            let span = MIN_FLOOR as f64..=MAX_FLOOR as f64;
171            finite(key).filter(|y| span.contains(y))
172        };
173        let ints = |key: &str| -> Option<Vec<i64>> {
174            o.get(key)?
175                .as_array()?
176                .iter()
177                .map(|n| n.as_f64().and_then(whole))
178                .collect()
179        };
180
181        let id = int("id").filter(|&id| id > 0)?;
182        let state = PState::parse(o.get("state")?.as_str()?)?;
183        let floors = ints("floors")?;
184        let shafts = ints("shafts")?;
185        // A route is its floors and the shafts between them: one floor more
186        // than shafts, every floor on the lot.
187        if floors.is_empty()
188            || floors.len() > MAX_ROUTE_FLOORS
189            || floors.len() != shafts.len() + 1
190            || !floors.iter().all(|&f| on_lot(f))
191        {
192            return None;
193        }
194        let leg = usize::try_from(int("leg")?).ok()?;
195        if leg > shafts.len() || (state.on_a_leg() && leg >= shafts.len()) {
196            return None;
197        }
198        let floor = int("floor").filter(|&f| on_lot(f))?;
199        let origin_floor = int("originFloor").filter(|&f| on_lot(f))?;
200        // Only a rider holds a seat in a car.
201        let car_index = opt_int("carIndex")?.filter(|&c| c >= 0);
202        if (state == PState::Riding) != car_index.is_some() {
203            return None;
204        }
205        let errand = match o.get("errand") {
206            None | Some(Value::Null) => None,
207            Some(e) => Some(Errand {
208                place: ErrandPlace::parse(e.get("place")?.as_str()?)?,
209                inbound: e.get("inbound")?.as_bool()?,
210            }),
211        };
212        let staff_trip = match o.get("staffTrip") {
213            None | Some(Value::Null) => None,
214            Some(t) => Some(StaffTrip::parse(t.as_str()?)?),
215        };
216        // A housekeeper works for a crew and is somewhere in their working
217        // day; nobody else carries either, or a room to clean.
218        let staff = flag("staff");
219        let crew_id = opt_int("crewId")?;
220        let clean_unit_id = opt_int("cleanUnitId")?;
221        let works = crew_id.is_some() && staff_trip.is_some();
222        if staff != works || (!staff && clean_unit_id.is_some()) {
223            return None;
224        }
225        let x = across("x")?;
226        // A commuting worker carries their office and which way they go.
227        let commute = match o.get("commute") {
228            None | Some(Value::Null) => None,
229            Some(c) => Some(Commute {
230                workplace_id: c.get("officeId")?.as_f64().and_then(whole)?,
231                inbound: c.get("inbound")?.as_bool()?,
232            }),
233        };
234        Some(Person {
235            id,
236            seed: seed_for(id),
237            state,
238            floor,
239            fy: height("fy")?,
240            x,
241            floors,
242            origin_floor,
243            shafts,
244            leg,
245            shaft_id: opt_int("shaftId")?,
246            car_index,
247            dest_x: across("destX")?,
248            wait: finite("wait")?,
249            trip_wait: finite("tripWait")?,
250            age: finite("age")?,
251            linger: finite("linger")?,
252            staff,
253            clean_unit_id,
254            meal_venue_id: opt_int("mealVenueId")?,
255            venue_unit_id: opt_int("venueUnitId")?,
256            counted_hotel_guest: flag("countedHotelGuest"),
257            origin_unit_id: opt_int("originUnitId")?,
258            dwell_seconds_left: opt_finite("dwellSecondsLeft")?,
259            returning: flag("returning"),
260            routine: o
261                .get("routine")
262                .and_then(Value::as_str)
263                .and_then(routine_name),
264            linger_for: opt_finite("lingerFor")?,
265            from_metro: flag("fromMetro"),
266            gave_up: flag("gaveUp"),
267            crew_id,
268            staff_trip,
269            walk_in: flag("walkIn"),
270            errand,
271            start_x: across("startX").unwrap_or(x),
272            commuter: commute.is_some(),
273            commute,
274        })
275    }
276}

283impl SaveEntries for Person {
284    fn save_entries<E>(
285        &self,
286        mut put: impl FnMut(&'static str, EntryValue<'_>) -> Result<(), E>,
287    ) -> Result<(), E> {
288        put("id", EntryValue::Int(self.id))?;
289        put("state", EntryValue::Text(self.state.as_str()))?;
290        put("floor", EntryValue::Int(self.floor))?;
291        put("fy", EntryValue::Float(self.fy))?;
292        put("x", EntryValue::Float(self.x))?;
293        put("startX", EntryValue::Float(self.start_x))?;
294        put("floors", EntryValue::Ints(&self.floors))?;
295        put("originFloor", EntryValue::Int(self.origin_floor))?;
296        put("shafts", EntryValue::Ints(&self.shafts))?;
297        // A leg index is far below `i64::MAX`.
298        put("leg", EntryValue::Int(self.leg as i64))?;
299        if let Some(v) = self.shaft_id {
300            put("shaftId", EntryValue::Int(v))?;
301        }
302        if let Some(v) = self.car_index {
303            put("carIndex", EntryValue::Int(v))?;
304        }
305        put("destX", EntryValue::Float(self.dest_x))?;
306        put("wait", EntryValue::Float(self.wait))?;
307        put("tripWait", EntryValue::Float(self.trip_wait))?;
308        put("age", EntryValue::Float(self.age))?;
309        put("linger", EntryValue::Float(self.linger))?;
310        let flags = [
311            ("staff", self.staff),
312            ("countedHotelGuest", self.counted_hotel_guest),
313            ("returning", self.returning),
314            ("fromMetro", self.from_metro),
315            ("gaveUp", self.gave_up),
316            ("walkIn", self.walk_in),
317        ];
318        for (key, set) in flags {
319            if set {
320                put(key, EntryValue::True)?;
321            }
322        }
323        let ids = [
324            ("cleanUnitId", self.clean_unit_id),
325            ("mealVenueId", self.meal_venue_id),
326            ("venueUnitId", self.venue_unit_id),
327            ("originUnitId", self.origin_unit_id),
328            ("crewId", self.crew_id),
329        ];
330        for (key, id) in ids {
331            if let Some(id) = id {
332                put(key, EntryValue::Int(id))?;
333            }
334        }
335        if let Some(v) = self.dwell_seconds_left {
336            put("dwellSecondsLeft", EntryValue::Float(v))?;
337        }
338        if let Some(v) = self.linger_for {
339            put("lingerFor", EntryValue::Float(v))?;
340        }
341        if let Some(v) = self.routine {
342            put("routine", EntryValue::Text(v))?;
343        }
344        if let Some(t) = self.staff_trip {
345            put("staffTrip", EntryValue::Text(t.as_str()))?;
346        }
347        if let Some(e) = self.errand {
348            let place = Scalar::Text(e.place.as_str());
349            put(
350                "errand",
351                EntryValue::Pair("place", place, "inbound", Scalar::Bool(e.inbound)),
352            )?;
353        }
354        if let Some(c) = self.commute {
355            // The key keeps its first name; it holds a venue's id too.
356            let office = Scalar::Int(c.workplace_id);
357            put(
358                "commute",
359                EntryValue::Pair("officeId", office, "inbound", Scalar::Bool(c.inbound)),
360            )?;
361        }
362        Ok(())
363    }
364}

366/// The crowd as the save writes it: [`Crowd::save_json`]'s object, each
367/// person written straight to the save (`save_entry`), and the marks beside
368/// its desks (`CrowdMarks`).
369pub(crate) struct SavedCrowd<'a>(pub &'a Crowd, pub CrowdMarks);
370
371impl Serialize for SavedCrowd<'_> {
372    fn serialize<S: Serializer>(&self, s: S) -> Result<S::Ok, S::Error> {
373        let c = self.0;
374        let mut m = s.serialize_map(None)?;
375        for (key, value) in c.save_head() {
376            m.serialize_entry(&key, &value)?;
377        }
378        c.commuters_write(&mut m, self.1)?;
379        m.serialize_entry("walkIns", &Each(&c.walk_ins, walk_in_json))?;
380        m.serialize_entry("people", &WrittenAll(&c.people))?;
381        m.end()
382    }
383}

385impl Crowd {
386    /// The crowd's save values ahead of the walk-ins and the people: the
387    /// random stream and counters a step reads back, the hour the last step
388    /// ran in. The commuters' desks and bookings follow them.
389    fn save_head(&self) -> Map<String, Value> {
390        let mut m = Map::new();
391        m.insert("nextId".into(), json!(self.next_id));
392        m.insert("rng".into(), json!(self.rng.seed()));
393        m.insert("spawnAcc".into(), json!(self.spawn_acc));
394        m.insert("frustration".into(), json!(self.frustration));
395        m.insert("commuteStress".into(), self.commute_stress_json());
396        if let Some(hour) = self.hour {
397            m.insert("hour".into(), json!(hour));
398        }
399        if self.weekend {
400            m.insert("weekend".into(), json!(true));
401        }
402        // A line nobody stands in holds only a remembered wait, which
403        // `waitsSeen` carries.
404        let rows: Vec<[i64; 3]> = (self.shaft_lines.sorted().into_iter())
405            .filter(|(_, line)| line.riders > 0)
406            .map(|(&(shaft, floor), line)| [shaft, floor, line.riders])
407            .collect();
408        if !rows.is_empty() {
409            m.insert("lines".into(), json!(rows));
410        }
411        if !self.waits_seen.is_empty() {
412            m.insert("waitsSeen".into(), self.waits_seen_json());
413        }
414        if !self.closed_cars.is_empty() {
415            m.insert("closedCars".into(), json!(self.closed_cars));
416        }
417        m
418    }
419
420    /// The crowd's save object: the same keys and values [`SavedCrowd`]
421    /// writes.
422    pub(crate) fn save_json(&self, marks: CrowdMarks) -> Value {
423        let mut m = self.save_head();
424        self.commuters_save(&mut m, marks);
425        let walk_ins = self.walk_ins.iter().map(walk_in_json).collect();
426        m.insert("walkIns".into(), Value::Array(walk_ins));
427        let people = self.people.iter().map(Person::save_form).collect();
428        m.insert("people".into(), Value::Array(people));
429        Value::Object(m)
430    }
431
432    /// Takes the people and values a save's `crowd` object carries. An entry
433    /// that does not read is skipped. The crowd is kept in id order (a
434    /// housekeeper is found by a binary search on it, `staff_shift::index_of`),
435    /// as every save writes it, so a person whose id is not above the last
436    /// one kept is left out, and ids stay unique.
437    pub(crate) fn restore(&mut self, v: &Value) {
438        let Some(o) = v.as_object() else { return };
439        let num = |key: &str| o.get(key).and_then(Value::as_f64).filter(|n| n.is_finite());
440        if let Some(state) = num("rng")
441            .and_then(whole)
442            .and_then(|s| u32::try_from(s).ok())
443        {
444            self.rng.restore(state);
445        }
446        self.spawn_acc = num("spawnAcc").unwrap_or(0.0);
447        self.frustration = num("frustration").unwrap_or(0.0);
448        // The hour the last step ran in, which the next step's hourly work
449        // reads before it sets it again.
450        self.hour = num("hour").and_then(whole).filter(|h| (0..24).contains(h));
451        self.weekend = o.get("weekend") == Some(&Value::Bool(true));
452        self.restore_lines(o.get("lines"));
453        self.restore_closed_cars(o.get("closedCars"));
454        self.restore_waits_seen(o.get("waitsSeen"));
455        self.commuters_restore(o);
456        if let Some(rows) = o.get("commuteStress") {
457            // A malformed list keeps the empty averages a load starts with.
458            let _ = self.seed_commute_stress(rows);
459        }
460        if let Some(list) = o.get("walkIns").and_then(Value::as_array) {
461            for w in list.iter().take(MAX_SAVED_WALK_INS) {
462                let at = w
463                    .get("at")
464                    .and_then(Value::as_f64)
465                    .filter(|n| n.is_finite());
466                let unit = w.get("unitId").and_then(Value::as_f64).and_then(whole);
467                if let (Some(at), Some(unit_id)) = (at, unit) {
468                    self.walk_ins.push(WalkIn { at, unit_id });
469                }
470            }
471            self.sort_walk_ins();
472        }
473        if let Some(list) = o.get("people").and_then(Value::as_array) {
474            for p in list.iter().filter_map(Person::from_save) {
475                if self.people.len() == MAX_SAVED_PEOPLE {
476                    break;
477                }
478                if self.people.last().is_none_or(|last| p.id > last.id) {
479                    self.people.push(p);
480                }
481            }
482        }
483        // The view's floor index, so the first frame after a load reads it.
484        self.by_floor.rebuild(&self.people);
485        // The next person gets an id above everyone's in the tower.
486        let past_ids = self.people.last().map_or(1, |p| p.id + 1);
487        let saved = num("nextId").and_then(whole).unwrap_or(1);
488        self.next_id = saved.max(past_ids).max(1);
489    }
490}
