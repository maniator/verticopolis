19/// `CongestionBindingClass` (sim/gripe.ts): the transport class whose shaft
20/// binds a floor's congestion reading; `Walkways` is the stairs-and-escalators
21/// tie, `None` the transportless defensive case.
22#[derive(Clone, Copy, Debug, PartialEq, Eq)]
23pub enum CongestionBindingClass {
24    Elevator,
25    Stairs,
26    Escalator,
27    Walkways,
28    None,
29}
30
31/// `BINDING_TIE_EPS`: two ratios within this band count as tied.
32const BINDING_TIE_EPS: f64 = 1e-9;
33
34const HOUSEHOLD_WEIGHTS: [i64; 4] = [4, 6, 2, 1];
35
36/// `householdPrice`.
37pub fn household_price(base: f64, residents: Option<i64>) -> f64 {
38    match residents {
39        None => base,
40        Some(r) => jsmath::round((base * r as f64) / CLASSIC_HOUSEHOLD),
41    }
42}
43
44/// `rollHousehold`.
45pub fn roll_household(rng: &mut crate::rng::Rng) -> i64 {
46    let total: i64 = HOUSEHOLD_WEIGHTS.iter().sum();
47    let mut roll = rng.int(1, total);
48    for i in 0..HOUSEHOLD_SIZES.len() {
49        roll -= HOUSEHOLD_WEIGHTS[i];
50        if roll <= 0 {
51            return HOUSEHOLD_SIZES[i];
52        }
53    }
54    3
55}

59    /// `servingTransportKindsAt(floor)`: (elevator, stairs, escalator), the
60    /// passenger kinds with a shaft stopping at the floor; staff-only
61    /// service elevators never count.
62    pub fn serving_transport_kinds_at(&self, floor: i64) -> (bool, bool, bool) {
63        let mut kinds = (false, false, false);
64        for t in &self.tower.transports {
65            if t.kind.is_staff_only_transport() || !t.stops_at(floor) {
66                continue;
67            }
68            if t.kind.is_elevator() {
69                kinds.0 = true;
70            } else if t.kind == Kind::Stairs {
71                kinds.1 = true;
72            } else if t.kind == Kind::Escalator {
73                kinds.2 = true;
74            }
75        }
76        kinds
77    }

79    /// `bindingTransportClassAt(floor)`: the class that binds the floor's
80    /// congestion reading, from the spatial model's own attribution. A
81    /// walkway flips the wording only when strictly worse than every serving
82    /// elevator beyond `BINDING_TIE_EPS`; two tied walkway kinds that both
83    /// clear the band read as the combined class. A floor without an
84    /// attribution entry, or with every class at 0, falls back to the
85    /// serving-kinds classification.
86    pub fn binding_transport_class_at(&self, floor: i64) -> CongestionBindingClass {
87        self.binding_class_from(
88            self.spatial_congestion_attribution_by_floor().get(&floor),
89            floor,
90        )
91    }

93    /// `bindingTransportClassAt` on an attribution entry the caller already
94    /// built (`None` for a floor without one), so a reader holding the
95    /// congestion maps does not build them again.
96    pub(crate) fn binding_class_from(
97        &self,
98        att: Option<&CongestionAttribution>,
99        floor: i64,
100    ) -> CongestionBindingClass {
101        if let Some(att) = att {
102            let walk_max = att.stairs.max(att.escalator);
103            if walk_max > att.elevator + BINDING_TIE_EPS {
104                let stairs_bind = att.stairs > att.elevator + BINDING_TIE_EPS;
105                let escalator_bind = att.escalator > att.elevator + BINDING_TIE_EPS;
106                if stairs_bind
107                    && escalator_bind
108                    && (att.stairs - att.escalator).abs() <= BINDING_TIE_EPS
109                {
110                    return CongestionBindingClass::Walkways;
111                }
112                return if att.stairs >= att.escalator {
113                    CongestionBindingClass::Stairs
114                } else {
115                    CongestionBindingClass::Escalator
116                };
117            }
118            if att.elevator > 0.0 {
119                return CongestionBindingClass::Elevator;
120            }
121        }
122        match self.serving_transport_kinds_at(floor) {
123            (true, _, _) => CongestionBindingClass::Elevator,
124            (false, true, true) => CongestionBindingClass::Walkways,
125            (false, true, false) => CongestionBindingClass::Stairs,
126            (false, false, true) => CongestionBindingClass::Escalator,
127            (false, false, false) => CongestionBindingClass::None,
128        }
129    }

131    /// `congestionChurnNote(floor)`: the buy-back line's note naming the
132    /// transport that binds the floor's reading, with the lever that clears it.
133    fn congestion_churn_note(&self, floor: i64) -> String {
134        let binding = self.binding_transport_class_at(floor);
135        let noun = match binding {
136            CongestionBindingClass::Elevator => "elevators",
137            CongestionBindingClass::Walkways => "stairs and escalators",
138            CongestionBindingClass::Stairs => "stairs",
139            CongestionBindingClass::Escalator => "escalators",
140            CongestionBindingClass::None => "vertical transport",
141        };
142        let lever = if binding == CongestionBindingClass::Elevator {
143            "add cars"
144        } else {
145            "add capacity"
146        };
147        format!(" A new owner will buy in, but the crowded {noun} will wear them down too until you {lever}.")
148    }

150    /// `rerollSubtype(id)`: a fresh canon subtype drawn off the current one
151    /// (from the list minus the current entry, so no rejection sampling), or
152    /// from the whole list when the unit has none; `None` for a unit without
153    /// a subtype list of at least two.
154    pub fn reroll_subtype(&mut self, id: i64) -> Option<&'static str> {
155        let (kind, current) = {
156            let u = self.tower.get_unit(id)?;
157            (u.kind, u.subtype)
158        };
159        let list = kind.subtype_list()?;
160        if list.len() < 2 {
161            return None;
162        }
163        let current_idx = current.and_then(|c| list.iter().position(|&s| s == c));
164        let idx = match current_idx {
165            None => self.rng.int(0, list.len() as i64 - 1) as usize,
166            Some(cur) => {
167                let mut i = self.rng.int(0, list.len() as i64 - 2) as usize;
168                if i >= cur {
169                    i += 1;
170                }
171                i
172            }
173        };
174        let next = list[idx];
175        self.tower.set_unit_subtype(id, Some(next));
176        self.tower.bump_meal_overlay_revision();
177        Some(next)
178    }

180    /// `vacate(u, reason)` on the unit at index `i`.
181    pub fn vacate(&mut self, i: usize, reason: &'static str) {
182        let (kind, floor, ever_occupied, rent, residents, no_rate) = {
183            let u = &self.tower.units[i];
184            (
185                u.kind,
186                u.floor,
187                u.ever_occupied,
188                u.rent,
189                u.residents,
190                u.no_rate,
191            )
192        };
193        // The congestion note's binding class is read before the unit
194        // empties below, as the TypeScript does: the departing tenant's own
195        // load is part of what bound the floor's reading. Gated on the
196        // conditions the emit needs (only a bought-back sold condo on the
197        // market carries the note), so no other eviction pays the map build.
198        let cong_note =
199            if reason == "congestion" && kind == Kind::Condo && ever_occupied && !no_rate {
200                self.congestion_churn_note(floor)
201            } else {
202                String::new()
203            };
204        let mut buyback = 0.0;
205        if kind == Kind::Condo && ever_occupied {
206            let sale_price = rent.unwrap_or_else(|| rent_config(Kind::Condo).unwrap().default);
207            buyback = household_price(sale_price, residents);
208            self.money -= buyback;
209            self.record_money("condos", -buyback);
210        }
211        let mode = self.mode;
212        let (u, at) = self.tower.marked(i);
213        u.set_state(at, UnitState::Empty);
214        u.set_occupants(at, 0);
215        if !kind.is_hotel() {
216            u.ever_occupied = false;
217        }
218        u.set_residents(at, None);
219        if kind == Kind::Condo {
220            if let Some(r) = u.rent {
221                u.rent = Some(
222                    match mode
223                        .rent_model(Kind::Condo)
224                        .expect("every mode prices a condo")
225                    {
226                        RentModel::Band(band) => r.min(band.max).max(band.min),
227                        RentModel::Ladder(rungs) => snap_to_ladder(&rungs, r),
228                    },
229                );
230            }
231        }
232        u.extras_mut().vacate_reason = None;
233        u.extras_mut().vacate_at = None;
234        self.tower
235            .set_unit_label_at(i, kind.facility().name.to_string());
236        let name = kind.facility().name;
237        // The trailing note asks the move-in gate directly, on the emptied
238        // unit: a No Rate unit stays off the market, a unit the gate still
239        // holds stays empty until the cause is fixed, and only a unit that
240        // re-sells carries the congestion churn note.
241        let buyback_note = if buyback > 0.0 {
242            if no_rate {
243                " It is off the market (No Rate); set a rate to sell it again.".to_string()
244            } else {
245                let probe = self.tower.units[i].clone();
246                let mut ctx = self.build_satisfaction_context(true);
247                if self.would_evict_fresh_tenant(&probe, &mut ctx) {
248                    " It stays empty until you fix the cause.".to_string()
249                } else {
250                    cong_note
251                }
252            }
253        } else {
254            String::new()
255        };
256        let msg = if buyback > 0.0 {
257            format!(
258                "The owner left {} on {} ({}). You bought it back for ${}.{}",
259                name,
260                self.floor_label(floor),
261                vacate_reason_text(reason),
262                with_thousands(buyback),
263                buyback_note
264            )
265        } else {
266            format!(
267                "A tenant left {} on {} ({}).",
268                name,
269                self.floor_label(floor),
270                vacate_reason_text(reason)
271            )
272        };
273        self.emit(&msg, LogKind::Bad);
274    }

276    /// `rollCondoRelocations` (Modern only).
277    pub fn roll_condo_relocations(&mut self) {
278        let days = (VACATE_NOTICE_MINUTES / (24.0 * 60.0)).ceil();
279        let scale = self.clock.calendar.maint_period_days as f64 / 30.0;
280        for i in 0..self.tower.units.len() {
281            let (kind, state, ever_occupied, residents, floor) = {
282                let u = &self.tower.units[i];
283                (u.kind, u.state, u.ever_occupied, u.residents, u.floor)
284            };
285            if kind != Kind::Condo || state != UnitState::Occupied || !ever_occupied {
286                continue;
287            }
288            let chance = self.mode.condo_relocation_chance(residents) * scale;
289            if chance <= 0.0 || !self.rng.chance(chance) {
290                continue;
291            }
292            let (u, at) = self.tower.marked(i);
293            u.set_state(at, UnitState::Vacating);
294            u.extras_mut().vacate_reason = Some("relocation");
295            u.extras_mut().vacate_at = Some(self.clock.minutes + VACATE_NOTICE_MINUTES);
296            let msg = format!(
297                "A household in Condominium on {} is relocating. They leave in under {} day(s); you buy the unit back and re-list it.",
298                self.floor_label(floor),
299                days
300            );
301            self.emit(&msg, LogKind::Bad);
302        }
303    }

305    /// `reportMoveIns`.
306    pub fn report_move_ins(&mut self) {
307        let m = self.move_ins_today;
308        let mut parts: Vec<String> = Vec::new();
309        if m.offices > 0 {
310            parts.push(format!(
311                "{} office{} leased",
312                m.offices,
313                if m.offices > 1 { "s" } else { "" }
314            ));
315        }
316        if m.condos > 0 {
317            parts.push(format!(
318                "{} condo{} sold",
319                m.condos,
320                if m.condos > 1 { "s" } else { "" }
321            ));
322        }
323        if m.rooms > 0 {
324            parts.push(format!(
325                "{} hotel room{} booked",
326                m.rooms,
327                if m.rooms > 1 { "s" } else { "" }
328            ));
329        }
330        if m.fitness > 0 {
331            parts.push(format!(
332                "{} fitness club{} leased",
333                m.fitness,
334                if m.fitness > 1 { "s" } else { "" }
335            ));
336        }
337        if m.clinic > 0 {
338            parts.push(format!(
339                "{} clinic{} leased",
340                m.clinic,
341                if m.clinic > 1 { "s" } else { "" }
342            ));
343        }
344        if m.rentals > 0 {
345            parts.push(format!(
346                "{} rental{} leased",
347                m.rentals,
348                if m.rentals > 1 { "s" } else { "" }
349            ));
350        }
351        if !parts.is_empty() {
352            self.emit(
353                &format!("New tenants: {}.", parts.join(", ")),
354                LogKind::Good,
355            );
356        }
357        self.move_ins_today = Default::default();
358    }
