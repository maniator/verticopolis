23pub const CAR_FLOORS_PER_MINUTE: f64 = 0.8;
24const DWELL_MINUTES: f64 = 0.8;
25/// How close, in floors, a car counts as standing at a stop: the stop
26/// search looks past a stop this near, so a car that ends a move this near
27/// its target lands on it instead of skipping it.
28const AT_STOP_FLOORS: f64 = 0.05;
29
30/// Stops per word of a stop bit set: stop `k` of the shaft in hand is bit
31/// `k % 64` of word `k / 64`.
32const WORD: usize = 64;
33
34#[derive(Default)]
35pub struct ElevatorDispatch {
36    /// Transient per-car dwell timers, by shaft id. An ordered map, so the
37    /// prune's `retain` iterates no hashed collection.
38    car_dwell: BTreeMap<i64, Vec<f64>>,
39    /// Transient per-car crowd callers seen at the car's stop, by shaft id,
40    /// so Standard Floor Departure counts only newcomers.
41    car_seen: BTreeMap<i64, Vec<f64>>,
42    /// Transient per-car minutes waited at the car's current stop, by shaft
43    /// id, so Modern's Standard Floor Departure stops at its limit.
44    car_held: BTreeMap<i64, Vec<f64>>,
45    /// Waiting passengers per floor. An ordered map, so a floor lookup
46    /// compares small integer keys where a hash map ran SipHash on each.
47    /// Every read and write is by floor, and `retain` decays each entry on
48    /// its own, so no result depends on the order.
49    waiting: BTreeMap<i64, f64>,
50    /// The boarding tally since the last drain (#465): shaft id to origin
51    /// floor to boarded count. Counting only, never read back into behavior.
52    board_tally: HashMap<i64, IndexMap<i64, f64>>,
53    /// Scratch for the waiting pass, one slot per served floor: that
54    /// floor's running estimate this step, `None` until a room touches it.
55    acc: Vec<Option<f64>>,
56    /// Scratch for `move_cars`, bit sets over the stops of the shaft in
57    /// hand: the TypeScript's `calls` (a live hall call, the crowd's or the
58    /// statistical estimate's) and `claimed` (a car is heading to the stop or
59    /// dwelling at it), and the current car's cab stops.
60    call_bits: Vec<u64>,
61    claimed: Vec<u64>,
62    cab_stops: Vec<u64>,
63    /// The cars the schedule has stood down, as (shaft id, car index),
64    /// rebuilt by every `move_cars`. A person waiting at a floor boards only a
65    /// car outside this set, so nobody boards a car that is parked or about
66    /// to park (the crowd keeps its own copy).
67    closed_cars: Vec<(i64, i64)>,
68    /// The cars standing at a stop with their doors open after this
69    /// `move_cars`, as (shaft id, car index, floor), ascending: a car that
70    /// arrived, one still dwelling, or one parked at its home floor. People
71    /// step on and off only these cars, so nobody boards or leaves a car
72    /// that is passing a floor.
73    open_cars: Vec<(i64, i64, i64)>,
74    /// Scratch of zero words, the calls a stood-down car answers: none.
75    no_calls: Vec<u64>,
76    /// Scratch for Modern's best car: the on-shift cars as the choice sees
77    /// them, and each called stop's chosen car.
78    best: BestCars,
79    /// A rules profile run in place of the tower's mode. Only tests set it
80    /// (`DispatchRules::plain_scan`); the game always reads the mode.
81    pub(crate) profile: Option<DispatchRules>,
82}

89    /// `drainBoardings()`: hand over the tally and clear it.
90    pub fn drain_boardings(&mut self) -> HashMap<i64, IndexMap<i64, f64>> {
91        std::mem::take(&mut self.board_tally)
92    }

94    /// The cars a waiting person may not board: stood down by the schedule.
95    /// Ascending, as (shaft id, car index): `move_cars` sorts the list once it
96    /// has walked every shaft, since a save may list its shafts out of id
97    /// order, so a binary search finds a car.
98    pub fn closed_cars(&self) -> &[(i64, i64)] {
99        &self.closed_cars
100    }

102    /// Hands the crowd this step's closed cars by swapping the two lists, so
103    /// no step allocates. `move_cars` clears the list it gets back.
104    pub fn swap_closed_cars(&mut self, crowd_list: &mut Vec<(i64, i64)>) {
105        std::mem::swap(&mut self.closed_cars, crowd_list);
106    }

108    /// The cars standing at a stop with their doors open, as (shaft id, car
109    /// index, floor), ascending, so a binary search finds a car at a floor.
110    pub fn open_cars(&self) -> &[(i64, i64, i64)] {

112    }

120    pub fn waiting_at(&self, floor: i64) -> f64 {
121        self.waiting.get(&floor).copied().unwrap_or(0.0)
122    }

124    fn prune_removed_shafts(&mut self, tower: &Tower) {
125        // The shaft index answers "is this id live" without building a set.
126        let live = |id: &i64| tower.transport_index(*id).is_some();
127        self.car_dwell.retain(|id, _| live(id));
128        self.car_seen.retain(|id, _| live(id));
129        self.car_held.retain(|id, _| live(id));
130    }
131
132    /// `accumulate(tower, dt, rush)`: once per outer step. Returns the tower's
133    /// `totalPopulation`, which the pass over the rooms already sums, so the
134    /// crowd's spawn step reuses it instead of walking the rooms again.
135    pub fn accumulate(&mut self, tower: &Tower, dt: f64, rush: f64) -> i64 {
136        self.prune_removed_shafts(tower);
137        self.accumulate_waiting(tower, dt, rush)
138    }
139
140    fn accumulate_waiting(&mut self, tower: &Tower, dt: f64, rush: f64) -> i64 {
141        self.waiting.retain(|_, n| {
142            let v = *n - dt * 0.03;
143            if v <= 0.0 {
144                false
145            } else {
146                *n = v;
147                true
148            }
149        });
150        let floors = tower.dispatch_floors();
151        let served = &floors.served;
152        self.acc.clear();
153        self.acc.resize(served.len(), None);
154        // `totalPopulation` rides along: the same rooms in the same order,
155        // so one pass over the rooms instead of two.
156        let mut pop = 0;
157        for u in tower.room_units() {
158            if u.is_present() {
159                pop += u.census_count();
160            }
161            // A venue's visitors ride as real people, so they are not waiting.
162            if u.kind.counts_visitors() || u.occupants <= 0 {
163                continue;
164            }
165            let Ok(slot) = served.binary_search(&u.floor) else {
166                continue;
167            };
168            // Each floor folds its rooms in room order, starting from the
169            // value the map held, exactly as the per-room map updates did.
170            let w = self.acc[slot]
171                .get_or_insert_with(|| self.waiting.get(&u.floor).copied().unwrap_or(0.0));
172            *w = (*w + u.occupants as f64 * rush * dt * 0.012).min(25.0);
173        }
174        for (slot, w) in self.acc.iter().enumerate() {
175            if let Some(w) = *w {
176                self.waiting.insert(served[slot], w);
177            }
178        }
179        debug_assert_eq!(pop, tower.total_population());
180        if pop > 0 {
181            for &fl in &floors.lobbies {
182                let w = self.waiting.entry(fl).or_insert(0.0);
183                *w = (*w + pop as f64 * rush * dt * 0.0015).min(25.0);
184            }
185        }
186        pop
187    }

189    /// `moveCars(tower, dt, crowdCalls, clock)`.
190    pub fn move_cars(
191        &mut self,
192        tower: &mut Tower,
193        dt: f64,
194        calls: &ElevatorCalls,
195        hour: i64,
196        is_weekend: bool,
197    ) {
198        let floors = tower.dispatch_floors();
199        let n = tower.transports.len();
200        let rules = self
201            .profile
202            .unwrap_or_else(|| DispatchRules::for_mode(tower.mode));
203        self.closed_cars.clear();
204        self.open_cars.clear();
205        for ti in 0..n {
206            let (kind, id) = {
207                let t = &tower.transports[ti];
208                (t.kind, t.id)
209            };
210            if !kind.is_elevator() {
211                continue;
212            }
213            let staff_only = kind.is_staff_only_transport();
214            let hall = calls.hall.get(&id);
215            let cabs = calls.cab.get(&id);
216            let stops = tower.stops_of(&tower.transports[ti]);
217            if stops.is_empty() {
218                continue;
219            }
220            debug_assert!(stops.windows(2).all(|w| w[0] < w[1]));
221            let idle_floor = stops
222                .iter()
223                .copied()
224                .find(|s| floors.lobbies.binary_search(s).is_ok())
225                .unwrap_or(stops[0]);
226            let t = &mut tower.transports[ti];
227            // Dispatch only reads the schedule, so it borrows it.
228            let sched = t.schedule.as_ref();
229            let active_count = Schedule::active_car_count(sched, is_weekend, hour, t.cars) as usize;
230            let shaft_dwell = Schedule::dwell_minutes_for(sched, DWELL_MINUTES);
231            let response = Schedule::waiting_response_for(sched);
232            let span = (t.top - t.bottom) as f64;
233            let cars = t.cars as usize;
234            let wait_limit = rules.departure_limit.map(|m| m * shaft_dwell);
235            let dwell = self.car_dwell.entry(id).or_default();
236            let seen = self.car_seen.entry(id).or_default();
237            let held = self.car_held.entry(id).or_default();
238            if dwell.len() != cars || seen.len() != cars || held.len() != cars {
239                (*dwell, *seen, *held) = (vec![0.0; cars], vec![0.0; cars], vec![0.0; cars]);
240            }
241            if t.car_load.as_ref().is_none_or(|l| l.len() != cars) {
242                t.car_load = Some(vec![0.0; cars]);
243            }
244            let car_load = t.car_load.as_mut().unwrap();
245            let cap = kind.car_capacity();
246
247            let hall_at = |fl: i64| hall.and_then(|h| h.get(&fl)).copied().unwrap_or(0.0);
248            // `calls` and `claimed` as bit sets over the stops. `stops` is
249            // ascending, so a floor's stop is a binary search away, and a
250            // floor that is not a stop is never asked about.
251            let words = stops.len().div_ceil(WORD);
252            let (call_bits, claimed) = (&mut self.call_bits, &mut self.claimed);
253            call_bits.clear();
254            call_bits.resize(words, 0);
255            claimed.clear();
256            claimed.resize(words, 0);
257            self.no_calls.clear();
258            self.no_calls.resize(words, 0);
259            for (k, &fl) in stops.iter().enumerate() {
260                // A crowd call already raises the stop, so the estimate is
261                // only looked up when it would decide.
262                if hall_at(fl) >= 1.0
263                    || (!staff_only && self.waiting.get(&fl).copied().unwrap_or(0.0) >= 1.0)
264                {
265                    set_bit(call_bits, k);
266                }
267            }
268            let stop_index = |floor: i64| stops.binary_search(&floor).ok();
269            if rules.best_car {
270                // The on-shift cars as they begin the step; each called
271                // stop's best car is worked out when a sweep first asks.
272                self.best.start(stops.len(), response, span);
273                for j in 0..active_count.min(cars) {
274                    let cab = cabs.and_then(|c| c.get(&(j as i64)));
275                    let (pos, dir) = (t.car_positions[j], t.car_dir[j]);
276                    self.best
277                        .push(CarView::of(pos, dir, dwell[j], cab, shaft_dwell));
278                }
279            }
280            for i in 0..cars {
281                let unset = idle_home(rules.spread_homes, &stops, idle_floor, i, cars);
282                let car_home = Schedule::home_floor_for(sched, i, unset).clamp(t.bottom, t.top);
283                // A car the schedule has stood down takes no new riders and
284                // answers no hall calls, but still takes the riders aboard to
285                // their floors; it parks once the cab is empty.
286                let stood_down = i >= active_count;
287                if stood_down {
288                    self.closed_cars.push((id, i as i64));
289                }
290                let riders_aboard = cabs
291                    .and_then(|c| c.get(&(i as i64)))
292                    .is_some_and(|c| !c.is_empty());
293                if stood_down && !riders_aboard {
294                    car_load[i] = 0.0;
295                    dwell[i] = 0.0;
296                    let cur = t.car_positions[i];
297                    let home = car_home as f64;
298                    if (cur - home).abs() < 0.05 {
299                        t.car_positions[i] = home;
300                        t.car_dir[i] = 0;
301                    } else {
302                        let step = dt * CAR_FLOORS_PER_MINUTE;
303                        let dir = if home > cur { 1 } else { -1 };
304                        let np = if (home - cur).abs() <= step {
305                            home
306                        } else {
307                            cur + dir as f64 * step
308                        };
309                        t.car_positions[i] = np.max(t.bottom as f64).min(t.top as f64);
310                        t.car_dir[i] = if np == home { 0 } else { dir };
311                    }
312                    continue;
313                }
314                let mut car_dt = dt;
315                if dwell[i] > 0.0 && rules.departure_restarts && !stood_down {
316                    // Standard Floor Departure: whoever arrives while the car
317                    // waits boards and starts the wait over, up to the limit.
318                    let fl = jsmath::round(t.car_positions[i]) as i64;
319                    if let Some(k) = stop_index(fl) {
320                        let mut b = Boarding {
321                            waiting: &mut self.waiting,
322                            tally: &mut self.board_tally,
323                        };
324                        let (room, live) = (cap - car_load[i], has_bit(call_bits, k));
325                        let got =
326                            b.arrivals(id, fl, (hall_at(fl), &mut seen[i]), room, staff_only, live);
327                        if got > 0.0 {
328                            car_load[i] += got;
329                            let left = wait_limit.map_or(shaft_dwell, |l| l - held[i]);
330                            dwell[i] = shaft_dwell.min(left).max(0.0);
331                        }
332                    }
333                }
334                if dwell[i] > 0.0 {
335                    let pause = dwell[i].min(car_dt);
336                    dwell[i] -= pause;
337                    held[i] += pause;
338                    car_dt -= pause;
339                    if car_dt <= 0.0 {
340                        let floor = jsmath::round(t.car_positions[i]) as i64;
341                        if let Some(k) = stop_index(floor) {
342                            set_bit(claimed, k);
343                        }
344                        // Still dwelling: the doors stay open all step.
345                        self.open_cars.push((id, i as i64, floor));
346                        continue;
347                    }
348                }
349                let v = car_dt * CAR_FLOORS_PER_MINUTE;
350                let mut pos = t.car_positions[i];
351                let was_parked = t.car_dir[i] == 0;
352                let mut dir = if t.car_dir[i] == 0 { 1 } else { t.car_dir[i] };
353                let cab_stops = cabs
354                    .and_then(|c| c.get(&(i as i64)))
355                    .filter(|c| !c.is_empty())
356                    .map(|c| fill_cab_stops(&stops, c, &mut self.cab_stops));
357                let reach = match response {
358                    Some(r) if was_parked && rules.parked_reach_gate => (span - r).max(0.0),
359                    _ => f64::INFINITY,
360                };
361                // Modern: a call is another car's when that car is its best.
362                // Classic's Waiting Car Response: a parked car leaves a call
363                // to a car already heading there within that many floors.
364                let best = &self.best;
365                let covered = |k: usize| {
366                    if rules.best_car {
367                        return best.owner(k, stops[k]) != i;
368                    }
369                    let r = response.filter(|_| rules.parked_car_yields && was_parked);
370                    r.is_some_and(|r| {
371                        let floor = stops[k] as f64;
372                        moving_car_covers(&t.car_positions, &t.car_dir, i, active_count, floor, r)
373                    })
374                };
375                let demand = Demand {
376                    stops: &stops,
377                    calls: if stood_down {
378                        &self.no_calls[..]
379                    } else {
380                        &call_bits[..]
381                    },
382                    claimed: &claimed[..],
383                    cab: cab_stops,
384                };
385                let pick = |dir: i64| first_answering(&demand, pos, dir, reach, covered);
386                // The sweep looks only ahead, so a car parked on a stop whose
387                // call it owns boards the estimate there now, as on arrival.
388                let own_stop =
389                    (rules.parked_owner_boards && was_parked && !stood_down && !staff_only)
390                        .then(|| stop_index(jsmath::round(pos) as i64))
391                        .flatten()
392                        .filter(|&k| {
393                            let (fl, unclaimed) = (stops[k], !has_bit(claimed, k));
394                            let on_it = (fl as f64 - pos).abs() <= AT_STOP_FLOORS;
395                            let estimate = self.waiting.get(&fl).copied().unwrap_or(0.0);
396                            on_it && unclaimed && estimate >= 1.0 && !covered(k)
397                        });
398                let mut target = own_stop.or_else(|| pick(dir));
399                if target.is_none() {
400                    dir = -dir;
401                    target = pick(dir);
402                }
403                if let Some(k) = target {
404                    set_bit(claimed, k);
405                }
406                let target = match target {
407                    Some(k) => stops[k] as f64,
408                    None => {
409                        let tg = car_home as f64;
410                        if (pos - tg).abs() < 0.05 {
411                            t.car_dir[i] = 0;
412                            car_load[i] = 0.0;
413                            // Parked at home: it settles on the floor and
414                            // opens for whoever waits there.
415                            t.car_positions[i] = tg;
416                            self.open_cars.push((id, i as i64, car_home));
417                            continue;
418                        }
419                        tg
420                    }
421                };
422                // A car that would end this move within `AT_STOP_FLOORS` of
423                // its target lands on it now: left short, the next search
424                // would look past the stop and the car would glide by it.
425                if (target - pos).abs() <= v + AT_STOP_FLOORS {
426                    pos = target;
427                    dwell[i] = shaft_dwell;
428                    held[i] = 0.0;
429                    seen[i] = hall_at(target as i64);
430                    // Arrived: the doors open for the rest of the step.
431                    self.open_cars.push((id, i as i64, target as i64));
432                    car_load[i] = (car_load[i] - (car_load[i] * 0.45).ceil()).max(0.0);
433                    let tf = target as i64;
434                    let w = if staff_only {
435                        hall_at(tf)
436                    } else {
437                        self.waiting.get(&tf).copied().unwrap_or(0.0)
438                    };
439                    // A stood-down car lets nobody on.
440                    let room = if stood_down { 0.0 } else { cap - car_load[i] };
441                    let live = stop_index(tf).is_some_and(|k| has_bit(call_bits, k));
442                    let mut b = Boarding {
443                        waiting: &mut self.waiting,
444                        tally: &mut self.board_tally,
445                    };
446                    let board = b.board(id, tf, w, room, staff_only, live);
447                    if board > 0.0 {
448                        car_load[i] += board;
449                        t.rides += 1;
450                    }
451                    if pos >= t.top as f64 {
452                        dir = -1;
453                    } else if pos <= t.bottom as f64 {
454                        dir = 1;
455                    }
456                } else {
457                    dir = if target > pos { 1 } else { -1 };
458                    pos += dir as f64 * v;
459                }
460                t.car_positions[i] = pos.max(t.bottom as f64).min(t.top as f64);
461                t.car_dir[i] = dir;
462            }
463        }
464        // The crowd searches the list, and the shafts come in saved order,
465        // which need not be id order.
466        self.closed_cars.sort_unstable();
467        self.open_cars.sort_unstable();
468    }
