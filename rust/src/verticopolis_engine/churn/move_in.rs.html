16    /// `attemptMoveIns`.
17    pub fn attempt_move_ins(&mut self) {
18        let weekend = self.clock.is_weekend();
19        let parking = self.parking_move_in_reading();
20        let parking_penalty = if parking.office_short { 0.5 } else { 1.0 };
21        let home_parking_factor = parking.home_factor;
22        let served_set = self.tower.served_floors();
23        let mut reach_memo: FastMap<i64, bool> = FastMap::default();
24        let mut sat_ctx: Option<SatisfactionContext> = None;
25        let evening = self.clock.is_evening();
26        for i in self.tower.room_indices() {
27            let u = self.tower.units[i].clone();
28            if u.state != UnitState::Empty || u.no_rate {
29                continue;
30            }
31            let f = u.kind.facility();
32            if f.population == 0 && !u.kind.is_hotel() {
33                continue;
34            }
35            if !served_set.contains(&u.floor) {
36                continue;
37            }
38            let key = self.tower.seg_at(u.floor, Some(u.x as f64));
39            let reachable = match reach_memo.get(&key) {
40                Some(&h) => h,
41                None => {
42                    let h = self.position_reachable(u.floor, u.x);
43                    reach_memo.insert(key, h);
44                    h
45                }
46            };
47            if !reachable {
48                continue;
49            }
50            if u.kind == Kind::Condo
51                || u.kind == Kind::Office
52                || u.kind.is_lease_amenity()
53                || u.kind.is_rental()
54            {
55                if sat_ctx.is_none() {
56                    sat_ctx = Some(self.build_satisfaction_context(true));
57                }
58                if self.would_evict_fresh_tenant(&u, sat_ctx.as_mut().unwrap()) {
59                    continue;
60                }
61            }
62            let demand = self.demand_factor(&u);
63            let mut filled = false;
64            match u.kind {
65                Kind::Office => {
66                    if !weekend && self.rng.chance(0.25 * demand * parking_penalty) {
67                        filled = true;
68                    }
69                }
70                Kind::Condo => {
71                    if self.rng.chance(0.18 * demand * home_parking_factor) {
72                        filled = true;
73                    }
74                }
75                Kind::FitnessClub | Kind::Clinic => {
76                    if self.rng.chance(0.22 * demand) {
77                        self.move_in(i);
78                        if let Some(ctx) = sat_ctx.as_mut() {
79                            if u.kind == Kind::FitnessClub && served_set.contains(&u.floor) {
80                                note_halo_venue(&mut ctx.halo_floors, self.mode, &u);
81                            }
82                        }
83                    }
84                }
85                Kind::RentalStudio | Kind::RentalApartment => {
86                    let fill_rate = if u.kind == Kind::RentalStudio {
87                        0.22
88                    } else {
89                        0.16
90                    };
91                    if self.rng.chance(fill_rate * demand * home_parking_factor) {
92                        filled = true;
93                    }
94                }
95                k if k.is_hotel() => {
96                    if evening && self.rng.chance(0.5 * demand) {
97                        let (w, at) = self.tower.marked(i);
98                        w.set_state(at, UnitState::Asleep);
99                        w.ever_occupied = true;
100                        self.move_ins_today.rooms += 1;
101                        if let Some(ctx) = sat_ctx.as_mut() {
102                            if let Some(dm) = ctx.demand_map.as_mut() {
103                                let w = self.tower.units[i].clone();
104                                fold_origin_demand(dm, self, &w);
105                            }
106                        }
107                    }
108                }
109                _ => {}
110            }
111            if filled {
112                self.move_in(i);
113                if let Some(ctx) = sat_ctx.as_mut() {
114                    if let Some(dm) = ctx.demand_map.as_mut() {
115                        let w = self.tower.units[i].clone();
116                        fold_origin_demand(dm, self, &w);
117                    }
118                }
119            }
120        }
121    }

123    /// `moveIn(u)`.
124    pub fn move_in(&mut self, i: usize) {
125        let kind = self.tower.units[i].kind;
126        {
127            let (u, at) = self.tower.marked(i);
128            u.set_state(at, UnitState::Occupied);
129            u.satisfaction = 1.0;
130            u.extras_mut().vacate_reason = None;
131            u.extras_mut().vacate_at = None;
132        }
133        if kind == Kind::Condo && !self.tower.units[i].ever_occupied {
134            let asking = {
135                let u = &self.tower.units[i];
136                rent_of(u.kind, u.rent, u.no_rate)
137            };
138            let (price, residents) = if self.mode.variant_households() {
139                let r = roll_household(&mut self.rng);
140                (household_price(asking, Some(r)), Some(r))
141            } else {
142                (asking, None)
143            };
144            let floor = self.tower.units[i].floor;
145            let (u, at) = self.tower.marked(i);
146            u.ever_occupied = true;
147            if residents.is_some() {
148                u.set_residents(at, residents);
149            }
150            u.rent = Some(asking);
151            self.money += price;
152            self.record_money("condos", price);
153            self.move_ins_today.condos += 1;
154            let who = match residents {
155                Some(r) => format!(" to a household of {r}"),
156                None => String::new(),
157            };
158            let msg = format!(
159                "Condominium on {} sold{} for ${}.",
160                self.floor_label(floor),
161                who,
162                with_thousands(price)
163            );
164            self.emit(&msg, LogKind::Money);
165        }
166        match kind {
167            Kind::Office => {
168                let name = self.company_name();
169                self.tower.units[i].ever_occupied = true;
170                self.tower.set_unit_label_at(i, name);
171                self.move_ins_today.offices += 1;
172            }
173            Kind::FitnessClub => {
174                let name = self.gym_name();
175                self.tower.units[i].ever_occupied = true;
176                self.tower.set_unit_label_at(i, name);
177                self.move_ins_today.fitness += 1;
178            }
179            Kind::Clinic => {
180                let name = self.clinic_name();
181                self.tower.units[i].ever_occupied = true;
182                self.tower.set_unit_label_at(i, name);
183                self.move_ins_today.clinic += 1;
184            }
185            Kind::RentalStudio | Kind::RentalApartment => {
186                let residents = if kind == Kind::RentalApartment {
187                    Some(roll_household(&mut self.rng))
188                } else {
189                    None
190                };
191                let (u, at) = self.tower.marked(i);
192                u.ever_occupied = true;
193                if residents.is_some() {
194                    u.set_residents(at, residents);
195                }
196                self.move_ins_today.rentals += 1;
197            }
198            _ => {}
199        }
200    }
