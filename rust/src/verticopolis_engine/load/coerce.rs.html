8pub(super) fn coerce_log(v: Option<&Value>) -> Vec<LogEntry> {
9    let Some(arr) = v.and_then(Value::as_array) else {
10        return Vec::new();
11    };
12    let mut out: Vec<LogEntry> = Vec::new();
13    for e in arr.iter().rev() {
14        if out.len() >= LOG_RING_CAP {
15            break;
16        }
17        let Some(o) = e.as_object() else { continue };
18        let Some(text) = o.get("text").and_then(Value::as_str) else {
19            continue;
20        };
21        let mut t: Vec<u16> = text.encode_utf16().collect();
22        if t.len() > LOG_TEXT_CAP {
23            t.truncate(LOG_TEXT_CAP);
24        }
25        if t.len() == LOG_TEXT_CAP && t.last().is_some_and(|&c| (0xD800..=0xDBFF).contains(&c)) {
26            t.pop();
27        }
28        let text = String::from_utf16_lossy(&t);
29        let minute = match finite_num(o.get("minute")) {
30            Some(m) => m.max(0.0),
31            None => 0.0,
32        };
33        let kind = match o.get("kind").and_then(Value::as_str) {
34            Some("good") => LogKind::Good,
35            Some("bad") => LogKind::Bad,
36            Some("money") => LogKind::Money,
37            _ => LogKind::Info,
38        };
39        out.push(LogEntry { minute, text, kind });
40    }
41    out.reverse();
42    out
43}

48pub(super) fn coerce_view(v: Option<&Value>) -> Option<Value> {
49    let o = v?.as_object()?;
50    let tile = finite_num(o.get("tile"))?;
51    let floor = finite_num(o.get("floor"))?;
52    let mut out = Map::new();
53    out.insert("tile".into(), json!(tile.min(LOT as f64).max(0.0)));
54    out.insert(
55        "floor".into(),
56        json!(floor.min(MAX_FLOOR as f64).max(MIN_FLOOR as f64)),
57    );
58    match o.get("zoom") {
59        None | Some(Value::Null) => {}
60        Some(z) => {
61            let z = finite_num(Some(z))?;
62            out.insert(
63                "zoom".into(),
64                json!(z.min(VIEW_ZOOM_MAX).max(VIEW_ZOOM_MIN)),
65            );
66        }
67    }
68    Some(Value::Object(out))
69}

71pub(super) fn coerce_dirty_days(state: UnitState, kind: Kind, raw: Option<&Value>) -> Option<i64> {
72    if state != UnitState::Dirty || !kind.is_hotel() {
73        return None;
74    }
75    let raw = raw?;
76    let n = match raw.as_f64() {
77        Some(x) if x.is_finite() => x.floor().max(0.0) as i64,
78        _ => 0,
79    };
80    Some(n.min(crate::housekeeping::INFEST_DAYS - 1))
81}

83/// A theater's film age (#1932): a whole number of days, held to a byte as
84/// the original keeps it. Any other kind, or a value that is not a finite
85/// number, carries none.
86pub(super) fn film_days(kind: Kind, raw: Option<&Value>) -> Option<u8> {
87    if kind != Kind::Cinema {
88        return None;
89    }
90    let days = raw?.as_f64().filter(|x| x.is_finite())?;
91    Some(days.floor().clamp(0.0, f64::from(u8::MAX)) as u8)
92}

107/// A theater's second screen (#1950), read with the keys of its first. Any
108/// other room, and a value that is not an object, carry none, and a theater
109/// without one shows its first screen's film on both. A mode that runs one
110/// screen drops it once the tower is assembled (`assemble.rs`).
111pub(super) fn second_screen(kind: Kind, raw: Option<&Value>) -> Option<ScreenFilm> {
112    if kind != Kind::Cinema {
113        return None;
114    }
115    let screen = raw?.as_object()?;
116    let tier = if cinema_flag(kind, screen.get("filmClassic")) {
117        FilmTier::Classic
118    } else if cinema_flag(kind, screen.get("filmBlockbuster")) {
119        FilmTier::Blockbuster
120    } else {
121        FilmTier::New
122    };
123    Some(ScreenFilm {
124        tier,
125        days: film_days(kind, screen.get("filmDays")),
126        policy: film_policy(screen.get("filmPolicy")),
127    })
128}

130/// Whether a theater shows a classic film (#1932): `true` on a theater, and
131/// anything else reads as a new film.
132pub(super) fn cinema_flag(kind: Kind, raw: Option<&Value>) -> bool {
133    kind == Kind::Cinema && raw == Some(&Value::Bool(true))
134}

136/// A Modern theater's ticket price step (#1950): a step Modern lists, other
137/// than Normal, on a theater. Normal, an unknown step, a Classic save and any
138/// other room read as none.
139pub(super) fn ticket_price(kind: Kind, modern: bool, raw: Option<&Value>) -> Option<&'static str> {
140    if kind != Kind::Cinema || !modern {
141        return None;
142    }
143    let step = GameMode::Modern.ticket_price(raw?.as_str()?)?.key;
144    (step != NORMAL_TICKET_PRICE).then_some(step)
145}

184    /// A save's camera view crosses the load boundary as written, with or
185    /// without a zoom (a TDT import carries none, and a null zoom is absent).
186    #[test]
187    fn a_view_loads_as_written() {
188        let stamped = json!({ "tile": 123.5, "floor": 42.25, "zoom": 1.7 });
189        assert_eq!(view(stamped.clone()), Some(stamped));
190        let zoomless = json!({ "tile": 10, "floor": 5 });
191        assert_eq!(
192            view(zoomless.clone()),
193            Some(json!({ "tile": 10.0, "floor": 5.0 }))
194        );
195        assert_eq!(
196            view(json!({ "tile": 10, "floor": 5, "zoom": null })),
197            Some(json!({ "tile": 10.0, "floor": 5.0 }))
198        );
199        assert_eq!(
200            coerce_view(None),
201            None,
202            "a save without a view loads viewless"
203        );
204    }
205
206    /// A malformed view is dropped whole, so the camera falls back to the
207    /// tower's center: the wrong container, a missing or non-numeric member,
208    /// or a member a JSON writer turned to null (NaN and Infinity).
209    #[test]
210    fn a_malformed_view_is_dropped_whole() {
211        for bad in [
212            json!("over there"),
213            json!(42),
214            json!({}),
215            json!({ "tile": "12", "floor": 5 }),
216            json!({ "tile": null, "floor": 5 }),
217            json!({ "tile": 12, "floor": null }),
218            json!({ "tile": 12, "floor": 5, "zoom": "big" }),
219        ] {
220            assert_eq!(view(bad.clone()), None, "{bad}");
221        }
222    }
