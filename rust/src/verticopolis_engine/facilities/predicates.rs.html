7/// Whether shop and restaurant customers come in person from outside the
8/// tower (`crowd::walk_ins`). Off for now (owner, 2026-11-14): at full
9/// demand they jammed the elevators of towers built before 3.12.0 and
10/// tenants moved out over congestion. While it is off, those venues count
11/// their customers from demand as 3.11 did, with nobody shown arriving.
12pub const WALK_INS_ON: bool = false;
13
14/// A per-thread stand-in for [`WALK_INS_ON`] in tests.
15#[cfg(test)]
16fn walk_ins_switch() -> &'static std::thread::LocalKey<std::cell::Cell<bool>> {
17    thread_local! {
18        static ON: std::cell::Cell<bool> = const { std::cell::Cell::new(WALK_INS_ON) };
19    }
20    &ON
21}
22
23/// Turns walk-ins on or off for the calling test's thread, so their own
24/// tests keep running while [`WALK_INS_ON`] is off.
25#[cfg(test)]
26pub fn set_walk_ins_for_test(on: bool) {
27    walk_ins_switch().with(|w| w.set(on));
28}
29
30#[cfg(test)]
31fn walk_ins_on() -> bool {
32    walk_ins_switch().with(std::cell::Cell::get)
33}
34
35#[cfg(not(test))]
36fn walk_ins_on() -> bool {
37    WALK_INS_ON
38}

40/// How many hours before a venue opens its crew is called in
41/// (`Kind::staffed_at`).

47    /// `transportCarCapacity`.
48    pub fn car_capacity(self) -> f64 {
49        match self {
50            Kind::ElevatorStandard => 21.0,
51            Kind::ElevatorService => 17.0,
52            Kind::ElevatorExpress => 42.0,
53            Kind::Escalator => 30.0,
54            Kind::Stairs => 8.0,
55            _ => 0.0,
56        }
57    }

63    pub fn tdt_header_capacity(self) -> i64 {
64        match self {

113    #[doc(hidden)]
114    /// A shop or restaurant whose customers come in from outside the tower
115    /// at the rate its demand sets (`crowd::walk_ins`): every kind with a
116    /// spend per customer but the daycare, whose children are not shoppers.
117    /// Each walk-in comes through the lobby or the Metro and rides the cars
118    /// like anyone else, and the day's patronage counts the ones who got
119    /// there. None does while [`WALK_INS_ON`] is off.
120    pub fn takes_walk_ins(self) -> bool {
121        walk_ins_on()
122            && matches!(
123                self,
124                Kind::FastFood
125                    | Kind::Restaurant
126                    | Kind::FoodHall
127                    | Kind::Amusements
128                    | Kind::BoutiqueBay
129                    | Kind::Nightclub
130                    | Kind::Spa
131                    | Kind::SkyBar
132                    | Kind::Shop
133            )
134    }

136    #[doc(hidden)]
137    /// A kind whose head count mirrors the visitors inside it (an
138    /// attendance venue, a [`Kind::is_meal_venue`] or a kind that
139    /// [`Kind::takes_walk_ins`]). The count lives
140    /// in the unsaved `customers_in`, which a load counts again from the
141    /// visitors the save's crowd carries (`crowd::save`) and an undo starts
142    /// at 0, and the dispatch estimate skips it, since those visitors ride
143    /// as real people.
144    pub fn counts_visitors(self) -> bool {
145        self.attendance_cap().is_some() || self.is_meal_venue() || self.takes_walk_ins()
146    }

148    #[doc(hidden)]
149    /// The staff a room of this kind keeps at work while it is open, who
150    /// come in and go home as people in a Modern tower (`crowd::commuters`);
151    /// 0 for a kind with no staff of its own. The original draws no venue
152    /// staff, so these are our own.
153    pub fn staff_size(self) -> i64 {
154        match self {
155            Kind::Shop | Kind::BoutiqueBay => 2,
156            Kind::FastFood | Kind::Amusements | Kind::Spa | Kind::SkyBar => 3,
157            Kind::Restaurant | Kind::Nightclub | Kind::Daycare => 4,
158            Kind::Cinema | Kind::PartyHall => 4,
159            Kind::FoodHall | Kind::AquaticCenter => 6,
160            _ => 0,
161        }
162    }
163
164    #[doc(hidden)]
165    /// Whether this kind keeps staff of its own ([`Kind::staff_size`]).
166    pub fn has_staff(self) -> bool {
167        self.staff_size() > 0
168    }
169
170    #[doc(hidden)]
171    /// Whether a venue's staff should be at work at `hour`: while it is
172    /// open, and from four hours before it opens (`CREW_CALL_AHEAD`), so a
173    /// shop that opens at 10 has its crew ride in at 6, before the office
174    /// rush. A break of four hours or less, such as a restaurant's between
175    /// lunch and dinner, keeps them in.
176    pub fn staffed_at(self, hour: i64) -> bool {
177        self.has_staff()
178            && (0..=CREW_CALL_AHEAD).any(|d| self.is_open_at((hour + d).rem_euclid(24)))
179    }
180
181    #[doc(hidden)]
182    /// Whether a room of this kind in a `mode` tower opens only once its
183    /// staff have come in (`crowd::commuters`): a venue with staff, in a
184    /// tower whose staff commute in person.
185    pub fn waits_for_staff(self, mode: crate::clock::GameMode) -> bool {
186        self.has_staff() && mode.venue_staff_commute_in_person()
187    }

190    /// A kind whose workers come in and go home as people in a `mode` tower
191    /// (`crowd::commuters`): its head count is the workers who got there,
192    /// and the dispatch estimate skips it, since they ride the cars
193    /// themselves. Offices in Modern today.

196    }

198    #[doc(hidden)]
199    /// A home whose residents leave for the day and come home as people in a
200    /// `mode` tower (`crowd::commuters`): a condo, apartment or studio. Its
201    /// head count is the residents at home, and the dispatch estimate skips
202    /// it, since they ride the cars themselves. Modern homes today.
203    pub fn lives_in_person(self, mode: crate::clock::GameMode) -> bool {
204        (self == Kind::Condo || self.is_rental()) && mode.residents_commute_in_person()
205    }

212    pub fn stays_in_person(self, mode: crate::clock::GameMode) -> bool {
213        self.is_hotel() && mode.hotel_guests_arrive_in_person()

216    #[doc(hidden)]

219    /// ([`Kind::commutes_in_person`]), a home whose residents do

276    #[doc(hidden)]
277    /// `isOpenAt`.
278    pub fn is_open_at(self, hour: i64) -> bool {
279        match self {
280            Kind::FastFood => (7..22).contains(&hour),
281            Kind::Restaurant => (11..14).contains(&hour) || (17..23).contains(&hour),
282            Kind::FoodHall => (10..22).contains(&hour),
283            Kind::Shop => (10..21).contains(&hour),
284            Kind::Amusements => (10..24).contains(&hour),
285            Kind::Nightclub => !(2..20).contains(&hour),
286            Kind::BoutiqueBay => (10..21).contains(&hour),
287            Kind::Spa => (9..21).contains(&hour),
288            Kind::SkyBar => (16..24).contains(&hour),
289            Kind::Daycare => (7..19).contains(&hour),
290            Kind::Cinema => (12..24).contains(&hour),
291            Kind::PartyHall => (17..24).contains(&hour),
292            Kind::AquaticCenter => (8..22).contains(&hour),
293            _ => true,
294        }
295    }

297    #[doc(hidden)]
298    /// `openHoursPerDay`.
299    pub fn open_hours_per_day(self) -> i64 {
300        let h = (0..24).filter(|&hr| self.is_open_at(hr)).count() as i64;
301        if h == 0 {
302            1
303        } else {
304            h
305        }
306    }
